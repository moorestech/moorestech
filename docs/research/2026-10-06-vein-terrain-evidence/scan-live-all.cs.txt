using UnityEngine;
using System.Linq;
using System.Text;
var waters=Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None).Where(r=>r.enabled && r.sharedMaterials.Any(m=>m!=null && m.shader.name=="BK/Water")).ToArray();
var report=new StringBuilder(); report.AppendLine("waters="+waters.Length);
foreach(var t in Terrain.activeTerrains){
var d=t.terrainData; var hs=d.GetHeights(0,0,d.heightmapResolution,d.heightmapResolution);
var count=0;var landCount=0;var best=-1f; var example="";
for(int z=0;z<d.heightmapResolution;z+=4) for(int x=0;x<d.heightmapResolution;x+=4){
var wx=t.transform.position.x+x*d.size.x/(d.heightmapResolution-1);var wz=t.transform.position.z+z*d.size.z/(d.heightmapResolution-1);var y=t.transform.position.y+hs[z,x]*d.size.y;
if(y>=4.3f)continue;
count++;
var ax=Mathf.Min(d.alphamapWidth-1,Mathf.RoundToInt((float)x/(d.heightmapResolution-1)*(d.alphamapWidth-1)));var az=Mathf.Min(d.alphamapHeight-1,Mathf.RoundToInt((float)z/(d.heightmapResolution-1)*(d.alphamapHeight-1)));
var a=d.GetAlphamaps(ax,az,1,1);var sand=a[0,0,0];var max=0f;var layer=0;
for(int l=1;l<d.alphamapLayers;l++)if(a[0,0,l]>max){max=a[0,0,l];layer=l;}
if(max>0.5f)landCount++;
if(max<=best)continue; best=max;example=$"({wx},{y},{wz}) sand={sand} dominant={d.terrainLayers[layer].name} weight={max}";
}
report.AppendLine($"{t.name} lowNonzeroSamples={count} nonBeachDominant={landCount} example={example}");
}
System.IO.File.WriteAllText("/tmp/moores-vein-terrain-20261006/live-low-ground.txt",report.ToString());return report.ToString();
