> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** X-5c（Web ホストと Web 資産の撤去）が `feature/uitk` に入った後、NuGetForUnity の `packages.config` の推移閉包を計算し直して「どこからも要らなくなったパッケージ」を確かめ、client の 55 件（`Microsoft.AspNetCore.*` 24・`Microsoft.Extensions.*` 23〈DI は client では NuGet でない〉・その他 8〈`Microsoft.Net.Http.Headers`・Security 2・`DiagnosticSource`・`Channels`・MCP 2・`ServerSentEvents`〉）と server の 13 件（MCP 系）を、`packages.config` の行と `Assets/Packages/<id>.<ver>/`（.meta 込み）ごと消し、`Client.Tests/Tests.asmdef` の AspNetCore 4 DLL の名指しを同じコミットで外す。消す前後で compile（client と server の両プロジェクト）・テスト・ラダー 6 本・見た目の不変（W6-6）が同じであることを証拠で残し、`feature/uitk` へ取り込む。

**Architecture:** 撤去レーン（`単位`。Mac の Editor 1 本）。本番コードは 1 行も書かない。消す集合は人が選ばず、scratchpad に置く道具 `x5d_nuget_prune.py`（このファイルの Task 1 に全文。コミットしない）が「`manuallyInstalled` の根から Web の根 `Microsoft.AspNetCore` と MCP の根 `ModelContextProtocol` を除いた根の推移閉包」で導き、計画の期待（55／13）と一致し、消す DLL の名前を残る DLL・asmdef・rsp・link.xml・.cs のどれも参照していないときだけ `git rm` する。消すのは Editor を閉じた状態で行い、起動し直して NuGetForUnity が再解決しても消したものが戻らないことを見る。server の単体プロジェクトの compile は、PR ごとに走る既存の CI `Platform Compile Check`（`.github/workflows/platform-compile.yml:65` `projectPath: moorestech_server/`。StandaloneWindows64・StandaloneOSX。`workflow_dispatch` でも流せる）を経路にし、Mac では server を開かない（Editor は常に client の 1 本）。

# X-5d の NuGet 整理の道具（repo にコミットしない。scratchpad に置き、worktree の根で流す）
# 1) packages.config の manuallyInstalled から Web と MCP の根を除いた根で nuspec の dependencies の推移閉包を取り、消す集合を導いて計画の期待と比べる

PRIVATE_REPO = "moorestech_client/Assets/PersonalAssets/moorestech-client-private"


def hidden(path):
    return any(p.startswith(".") or p.endswith("~") for p in path.split("/")[1:])


def read_config(project):
    path = f"{project}/Assets/packages.config"
    root = ET.parse(path).getroot()
    return path, {p.get("id"): (p.get("version"), p.get("manuallyInstalled") == "true") for p in root.findall("package")}


def read_deps(project, pkgs):
    deps = {}
    for pid, (ver, _) in pkgs.items():
        specs = glob.glob(f"{project}/Assets/Packages/{pid}.{ver}/*.nuspec")
        if len(specs) != 1:
            sys.exit(f"NG {project}: nuspec が 1 本でない {pid}.{ver}: {specs}")
        t = ET.parse(specs[0]).getroot()
        ns = {"n": t.tag.split("}")[0].strip("{")}
        deps[pid] = {d.get("id") for d in t.findall(".//n:dependencies//n:dependency", ns)}
    return deps


def closure(roots, deps):
    seen, stack = set(), list(roots)
    while stack:
        x = stack.pop()
        if x not in seen:
            seen.add(x)
            stack += [d for d in deps.get(x, ()) if d in deps]
    return seen


def dll_names(project, pkgs, ids):
    return {os.path.basename(f)[:-4] for pid in ids
            for f in glob.glob(f"{project}/Assets/Packages/{pid}.{pkgs[pid][0]}/**/*.dll", recursive=True)}


def scan_dlls(project, skip_dirs, names):
    hits, needles = [], [(n, b"\0" + n.encode() + b"\0") for n in names]
    for base in [f"{project}/{d}" for d in ("Assets", "Packages", "LocalPackages", "Library/PackageCache")]:
        for f in glob.glob(f"{base}/**/*.dll", recursive=True):
            if hidden(f) or any(f.startswith(s + "/") for s in skip_dirs):
                continue
            data, own = open(f, "rb").read(), os.path.basename(f)[:-4]
            hits += [f"{f} -> {n}" for n, b in needles if n != own and b in data]
    return hits


def code_hits(f, text):
    out = []
    for i, line in enumerate(text.splitlines(), 1):
        s = line.strip()
        if s.startswith(("//", "*", "/*")) or not CS_PATTERN.search(line):
            continue
        out.append(f"{f}:{i}: {s}")
    return out


def ls_files(repo, specs):
    out = subprocess.run(["git", "-C", repo, "ls-files", "-z", "--"] + specs, capture_output=True, text=True, check=True).stdout
    return [f if repo == "." else f"{repo}/{f}" for f in out.split("\0") if f]


def scan_text(project, names):
    asmdef, code = [], []
    exts = (".asmdef", ".rsp", "link.xml", ".cs")
    # client は server の Scripts も client の DLL で compile する（manifest の tech.moores.server）
    bases = [project, "moorestech_server/Assets/Scripts"] if project == "moorestech_client" else [project]
    tracked = ls_files(".", [f"{b}/*{ext}" for b in bases for ext in exts])
    private = []
    if project == "moorestech_client":
        if not os.path.exists(f"{PRIVATE_REPO}/.git"):
            sys.exit(f"NG: {PRIVATE_REPO} が無い（走査できないので消さない）")
        private = ls_files(PRIVATE_REPO, [f"*{ext}" for ext in exts])
    cache = [f for f in glob.glob(f"{project}/Library/PackageCache/**/*.cs", recursive=True) if not hidden(f)]
    print(f"scan: tracked {len(tracked)} / private asmdef {sum(f.endswith('.asmdef') for f in private)} cs {sum(f.endswith('.cs') for f in private)} / PackageCache cs {len(cache)}")
    for f in tracked + private + cache:
        if "/Assets/Packages/" in f:
            continue
        text = open(f, encoding="utf-8", errors="replace").read()
        if f.endswith(".cs"):
            code += code_hits(f, text)
        else:
            asmdef += [f"{f} -> {n}" for n in sorted(names) if f"{n}.dll" in text]
    return asmdef, code


def main():

    ap.add_argument("--side", choices=["client", "server"], required=True)
    ap.add_argument("--apply", action="store_true")
    ap.add_argument("--names-out", required=True)
    args = ap.parse_args()
    project = f"moorestech_{args.side}"
    path, pkgs = read_config(project)
    deps = read_deps(project, pkgs)
    roots = {p for p, (_, manual) in pkgs.items() if manual} - DROP_ROOTS
    keep = closure(roots, deps)
    remove = set(pkgs) - keep
    ok = True
    print(f"{args.side}: 全 {len(pkgs)} / 根 {sorted(roots)} / 残す {len(keep)} / 消す {len(remove)}")
    if remove != EXPECTED[args.side] or len(keep) != EXPECTED_KEEP_COUNT[args.side]:
        ok = False
        print(f"NG: 期待との差 余分={sorted(remove - EXPECTED[args.side])} 不足={sorted(EXPECTED[args.side] - remove)}")
    names = dll_names(project, pkgs, remove)
    with open(args.names_out, "w", encoding="utf-8") as w:
        w.writelines(f"{n}\n" for n in sorted(names | remove))
    print(f"names: DLL {len(names)} / id {len(remove)} / 和 {len(names | remove)} -> {args.names_out}")
    for h in scan_dlls(project, [f"{project}/Assets/Packages/{p}.{pkgs[p][0]}" for p in remove], names):
        ok = False
        print(f"NG dll: 残る DLL が消す DLL を参照 {h}")
    asmdef, code = scan_text(project, names)
    for h in asmdef:
        # asmdef の名指しは同じコミットで外す。--apply の前に外し終えていなければ止める
        ok = ok and not args.apply
        print(f"{'NG' if args.apply else 'PENDING'} asmdef: {h}")
    for h in code:
        allowed = h in TEXT_ALLOW
        ok = ok and allowed
        print(f"{'ALLOW' if allowed else 'NG'} cs: {h}")
    if not ok:
        sys.exit("NG: 消さない（上の NG を読む）")
    print("OK: 閉包・期待・参照の検査が一致")
    if not args.apply:
        return
    lines = open(path, encoding="utf-8-sig").read().splitlines(keepends=True)
    kept = [l for l in lines if not any(f'<package id="{p}" ' in l for p in remove)]
    with open(path, "w", encoding="utf-8-sig", newline="") as w:
        w.writelines(kept)
    targets = [t for p in sorted(remove) for t in (f"{project}/Assets/Packages/{p}.{pkgs[p][0]}", f"{project}/Assets/Packages/{p}.{pkgs[p][0]}.meta")]
    subprocess.run(["git", "rm", "-r", "-q", "--"] + targets, check=True)
    print(f"{args.side}: packages.config {len(lines)} -> {len(kept)} 行 / git rm {len(targets)} 個（ディレクトリ {len(remove)}・.meta {len(remove)}）")


if __name__ == "__main__":
