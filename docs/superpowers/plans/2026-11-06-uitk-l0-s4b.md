# UITK L0-s4b（段階 4 の純ロジック移植: ブロック UI・研究・列車）Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** Web 版のブロック UI・研究・列車の純ロジック（vitest で固定された判断）を、`Client.Ui.State` の C# と EditMode NUnit へ 1 対 1 で写し、段階 4 の画面レーン（S4-alpha・S4-machine・S4-special・S4-research・S4-train）が使う判断の部品を先に揃える。

**Architecture:** Editor の要らない枠で C# とテストを書き溜め、最後の 1 タスクだけ Editor 枠を借りて compile とテストを通す（実行計画 §1.2「末」）。写し先は `Client.Ui.State/BlockInventory/Logic/<領域>/`・`Research/Logic/`・`Train/Logic/`。ロジックの入力は Web の bridge 型の「ロジックが読む部分」だけを State 側の不変型で持ち、表示文は既存の `UiText`（辞書キー＋名前付き引数）で返す。画面・ViewModel・コマンド・ゲームデータからの写し替えは段階 4 の画面レーンが持つ。

**Tech Stack:** C# 9（Unity 6000.3.8f1・`Client.Ui.State` asmdef）、NUnit（EditMode・`Client.Tests`）、uloop CLI、`Mooresmaster.Localization.Generated.LocalizationKeys`、`Core.Master.ItemId`

```csharp
namespace Client.Ui.State.Text
{
    /// <summary>
    /// 表示用の数値書式。de などで「,」にならないよう InvariantCulture を明示する（plan.md §3.9）。状態層の純ロジックも使うので State に置く
    /// Display number formatting with explicit InvariantCulture so locales like de never print ',' (plan.md §3.9); lives in State because state-layer logic uses it too
    /// </summary>
    public static class UiNumberFormat
    {
        // Web の toFixed と同じく、桁区切りなしで小数点は「.」
        // Like the Web's toFixed: no grouping and '.' as the decimal point
        public static string Fixed(double value, int decimals)
        {
            var rounded = Math.Round(value, decimals, MidpointRounding.AwayFromZero);
            return rounded.ToString("F" + decimals, CultureInfo.InvariantCulture);
        }

        public static string Integer(long value)
        {
            return value.ToString(CultureInfo.InvariantCulture);
        }

        // Web の Math.round と同じく .5 は正の無限大側へ丸める（C# 既定の偶数丸めとは違う）
        // Like the Web's Math.round, .5 rounds toward positive infinity (unlike C#'s default banker's rounding)
        public static long RoundHalfUp(double value)
        {
            return (long)Math.Floor(value + 0.5);
        }
    }
```

```csharp
using System;

namespace Client.Ui.State.BlockInventory.Logic.Detail
{
    /// <summary>
    /// ブロック詳細の数値計算。Web の detailLogic.ts の計算部分の写し
    /// Numeric helpers for block details; a port of the arithmetic in Web detailLogic.ts
    /// </summary>
    public static class BlockDetailMath
    {
        // uGUI CommonMachineBlockStateDetail.PowerRate と同式。要求 0 は充足 1 とみなす
        // Same formula as uGUI CommonMachineBlockStateDetail.PowerRate; a zero request counts as fully met
        public static double ComputePowerRate(double currentPower, double requestPower)
        {
            return requestPower == 0 ? 1 : currentPower / requestPower;
        }

        // レシピ秒が 0 以下なら毎分個数は出さない（Web は null）
        // No per-minute figure when the recipe time is zero or less (null on the Web)
        public static bool TryGetItemsPerMinute(int outputCount, double recipeTimeSeconds, out double itemsPerMinute)
        {
            itemsPerMinute = recipeTimeSeconds <= 0 ? 0 : outputCount * 60 / recipeTimeSeconds;
            return recipeTimeSeconds > 0;
        }

        // 0..total-1 の全スロット添字。全スロットを並べるビューはこの定義を使う
        // All slot indices 0..total-1; views listing every slot use this single definition
        public static int[] AllSlotIndices(int total)
        {
            var indices = new int[Math.Max(0, total)];
            for (var i = 0; i < indices.Length; i++) indices[i] = i;
            return indices;
        }

        // 残燃料/満燃料を 0..1 に収める。分母 0 は 0（uGUI 発電機の燃料バー相当）
        // Clamp remaining/full fuel to 0..1; a zero denominator gives 0 (mirrors the uGUI generator fuel bar)
        public static double FuelRatio(double remainingFuelTime, double currentFuelTime)
        {
            if (currentFuelTime <= 0) return 0;
            return Math.Min(1, Math.Max(0, remainingFuelTime / currentFuelTime));
        }
    }
}
```

```csharp
    /// <summary>
    /// 機械のスロット構成（入力・出力・モジュールの数）と、統合 index の帯への分割
    /// A machine's slot layout (input, output and module counts) and the split of combined indices into bands
    /// </summary>
    public readonly struct MachineSlotLayout
    {
        public int Input { get; }
        public int Output { get; }
        public int Module { get; }

        public MachineSlotLayout(int input, int output, int module)
        {
            Input = input;
            Output = output;
            Module = module;
        }

        // 統合 index を 入力→出力→モジュール に割る（uGUI のスロット構成順）。総数を超える添字は作らない
        // Split combined indices into input→output→module (uGUI order); never yields indices past the total
        public MachineSlotBands Split(int totalSlots)
        {
            var all = BlockDetailMath.AllSlotIndices(totalSlots);
            var inputEnd = Input;
            var outputEnd = Input + Output;
            var moduleEnd = Input + Output + Module;
            return new MachineSlotBands(Slice(0, inputEnd), Slice(inputEnd, outputEnd), Slice(outputEnd, moduleEnd));

            #region Internal

            List<int> Slice(int start, int end)
            {
                // Web の Array.slice と同じく範囲外は切り詰める
                // Truncate out-of-range bounds like Web Array.slice
                // silent-ok: 構成と総数の食い違いは Web と同じく欠けた帯として描く（判断記録 D-19） / 欠けた帯は S4-machine の参照キャプチャとの並べ比べと録画で見える
                var from = Math.Min(start, all.Length);
                var to = Math.Min(end, all.Length);
                var band = new List<int>();
                for (var i = from; i < to; i++) band.Add(all[i]);
                return band;
            }

            #endregion
        }
    }
```

```csharp
using System;
using System.Collections.Generic;
using Client.Localization;
using Client.Ui.State.Text;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.State.BlockInventory.Logic.Belt
{
    public enum BeltWindowStep
    {
        Previous = -1,
        Next = 1,
    }

    // 行の文。毎分の搬出数（小数 1 桁）は計測済みの行にだけある
    // The row text; exits per minute (one decimal) exist only on a measured row
    public abstract class BeltThroughputRowDisplay
    {
        public UiText Text { get; }

        protected BeltThroughputRowDisplay(UiText text)
        {
            Text = text;
        }
    }

    public sealed class MeasuringBeltRowDisplay : BeltThroughputRowDisplay
    {
        public MeasuringBeltRowDisplay(UiText text) : base(text)
        {
        }
    }

    public sealed class MeasuredBeltRowDisplay : BeltThroughputRowDisplay
    {
        public string ItemsPerMinute { get; }

        public MeasuredBeltRowDisplay(UiText text, string itemsPerMinute) : base(text)
        {
            ItemsPerMinute = itemsPerMinute;
        }
    }

    /// <summary>
    /// ベルトの搬出統計の表示判断（Web beltThroughputLogic.ts の写し）
    /// Display decisions for belt exit statistics (port of Web beltThroughputLogic.ts)
    /// </summary>
    public static class BeltThroughputLogic
    {
        // 既定の窓は 180tick（9 秒）。バニラ 3 速度の搬出周期（20・9・3tick）の最小公倍数（ユーザー裁定 2026-10-20）
        // The default window is 180 ticks (9 s), the LCM of the three vanilla output periods (user ruling 2026-10-20)
        public const int DefaultBeltWindowTicks = 180;

        // 棒グラフは 1 秒平均を 9 本（ユーザー裁定 2026-10-20）
        // The bar chart shows nine one-second averages (user ruling 2026-10-20)
        public const int RecentSecondBarCount = 9;

        private const int TicksPerSecond = 20;
        private const int TicksPerMinute = TicksPerSecond * 60;

        // 選んだ長さの窓。一覧に無ければ先頭の窓へ落とし、一覧が空なら選べない
        // The window of the chosen length; falls back to the first, and none can be chosen from an empty list
        public static bool TrySelectBeltWindow(IReadOnlyList<BeltThroughputWindow> windows, int ticks, out BeltThroughputWindow selected)
        {
            foreach (var window in windows)
            {
                if (window.Ticks != ticks) continue;
                selected = window;
                return true;
            }
            selected = windows.Count > 0 ? windows[0] : null;
            return windows.Count > 0;
        }

        // 一覧の中で 1 つ隣の窓へ移る。端では止まり、一覧に無い長さからは先頭扱いで動く
        // Move to the neighbouring window; stops at the ends, and an unlisted length counts as the first
        public static int CycleBeltWindow(IReadOnlyList<BeltThroughputWindow> windows, int currentTicks, BeltWindowStep step)
        {
            if (windows.Count == 0) return currentTicks;
            var currentIndex = 0;
            for (var i = 0; i < windows.Count; i++)
            {
                if (windows[i].Ticks == currentTicks) currentIndex = i;
            }
            var nextIndex = Math.Min(windows.Count - 1, Math.Max(0, currentIndex + (int)step));
            return windows[nextIndex].Ticks;
        }

        public static BeltThroughputRowDisplay RowDisplay(BeltThroughputWindow window)
        {
            var seconds = new LocalizationTextParam("seconds", FormatCount(window.Ticks / (double)TicksPerSecond));
            if (!(window is MeasuredBeltWindow measured)) return new MeasuringBeltRowDisplay(new UiText(LocalizationKeys.Ui.BlockInventory.BeltThroughputMeasuring, new[] { seconds }));

            // 停止中（最大 0）は割合を出せないので 0% にする
            // A stopped belt (max 0) has no ratio, so it shows 0%
            var percent = measured.MaxCount > 0 ? UiNumberFormat.RoundHalfUp(measured.ExitCount / measured.MaxCount * 100) : 0;
            var text = new UiText(LocalizationKeys.Ui.BlockInventory.BeltThroughputWindow, new[]
            {
                seconds,
                new LocalizationTextParam("count", UiNumberFormat.Integer(measured.ExitCount)),
                new LocalizationTextParam("max", FormatCount(measured.MaxCount)),
                new LocalizationTextParam("percent", UiNumberFormat.Integer(percent)),
            });
            var itemsPerMinute = UiNumberFormat.Fixed(measured.ExitCount * (double)TicksPerMinute / measured.Ticks, 1);
            return new MeasuredBeltRowDisplay(text, itemsPerMinute);
        }

        // 古い順の 1 秒ごとの搬出数を、最大に対する高さ（0..1）の棒 barCount 本に写す。まだ無い秒は先頭側を null で埋める
        // Map oldest-first per-second counts onto barCount bar heights (0..1); missing seconds are null at the front
        public static double?[] RecentSecondBars(IReadOnlyList<int> counts, double maxCountPerSecond, int barCount)
        {
            var bars = new double?[barCount];
            var recentCount = Math.Min(barCount, counts.Count);
            var offset = barCount - recentCount;
            for (var i = 0; i < recentCount; i++)
            {
                // 停止中（最大 0）は 0。整数の粗さで最大を超えた秒は満杯に丸める
                // A stopped belt (max 0) gives 0; a second above the maximum through integer coarseness rounds to full
                var count = counts[counts.Count - recentCount + i];
                bars[offset + i] = maxCountPerSecond > 0 ? Math.Min(1, count / maxCountPerSecond) : 0;
            }
            return bars;
        }

        // 整数はそのまま、端数は小数 1 桁（90tick=4.5 秒。速度によっては最大数も端数になる）
        // Whole numbers as they are, fractions to one decimal (90 ticks = 4.5 s; some speeds give a fractional maximum)
        private static string FormatCount(double value)
        {
            return value == Math.Floor(value) ? UiNumberFormat.Integer((long)value) : UiNumberFormat.Fixed(value, 1);
        }
    }
}
```

```csharp
using System.Collections.Generic;
using Client.Ui.State.BlockInventory.Logic.Common;
using Core.Master;

namespace Client.Ui.State.BlockInventory.Logic.Machine
{
    /// <summary>
    /// ホストが配信するスロット束縛から、描くスロットと各スロットのゴーストを導く（ADR 0042 R7/R8・2026-08-30 裁定。Web machineSlotGhosts.ts の写し）
    /// Derives the slots to draw and each slot's ghost from the host-published binding (ADR 0042 R7/R8, 2026-08-30 ruling; port of Web machineSlotGhosts.ts)
    /// </summary>
    public static class MachineSlotGhosts
    {
        // 束縛が無い機械は全実スロットを描く（R11）。束縛があれば束縛スロット＋中身の残る束縛外スロットだけ（R7）
        // An unbound machine draws every real slot (R11); with a binding, only bound slots plus occupied unbound ones (R7)
        public static MachineSlotView BuildMachineSlotView(MachineBindingDetail machine, IReadOnlyList<BlockItemSlotState> itemSlots, IReadOnlyList<BlockFluidSlotState> fluidSlots)
        {
            var bands = machine.Layout.Split(itemSlots.Count);
            if (machine.SlotBindings.Count == 0 && machine.TankBindings.Count == 0)
            {
                return new MachineSlotView(ItemsWithoutGhost(bands.Input), ItemsWithoutGhost(bands.Output), FluidsWithoutGhost());
            }

            // 同じスロットへの束縛が重なったら後勝ち（Web の Map と同じ）
            // When bindings overlap on one slot the later one wins (same as the Web Map)
            var itemGhosts = new Dictionary<int, MachineItemGhost>();
            foreach (var binding in machine.SlotBindings) itemGhosts[binding.Slot] = new MachineItemGhost(binding.ItemId, binding.Count, binding.AcceptedItemIds);
            var fluidGhosts = new Dictionary<int, MachineFluidGhost>();
            foreach (var binding in machine.TankBindings) fluidGhosts[binding.Tank] = new MachineFluidGhost(binding.FluidId, binding.Amount);

            return new MachineSlotView(ItemBand(bands.Input), ItemBand(bands.Output), FluidBand());

            #region Internal

            List<BoundItemSlot> ItemsWithoutGhost(IReadOnlyList<int> band)
            {
                var slots = new List<BoundItemSlot>();
                foreach (var index in band) slots.Add(BoundItemSlot.WithoutGhost(index));
                return slots;
            }

            List<BoundFluidSlot> FluidsWithoutGhost()
            {
                var slots = new List<BoundFluidSlot>();
                for (var index = 0; index < fluidSlots.Count; index++) slots.Add(BoundFluidSlot.WithoutGhost(index));
                return slots;
            }

            // 束縛スロットを帯の順に並べ、続けて中身の残る束縛外スロット（返却しきれなかった残置分）を並べる
            // Bound slots in band order first, then occupied unbound slots (leftovers the refund could not return)
            List<BoundItemSlot> ItemBand(IReadOnlyList<int> band)
            {
                var slots = new List<BoundItemSlot>();
                foreach (var index in band)
                {
                    if (itemGhosts.TryGetValue(index, out var ghost)) slots.Add(BoundItemSlot.WithGhost(index, ghost));
                }
                foreach (var index in band)
                {
                    if (!itemGhosts.ContainsKey(index) && itemSlots[index].HasRealItem) slots.Add(BoundItemSlot.WithoutGhost(index));
                }
                return slots;
            }

            List<BoundFluidSlot> FluidBand()
            {
                var slots = new List<BoundFluidSlot>();
                for (var index = 0; index < fluidSlots.Count; index++)
                {
                    if (fluidGhosts.TryGetValue(index, out var ghost)) slots.Add(BoundFluidSlot.WithGhost(index, ghost));
                }
                for (var index = 0; index < fluidSlots.Count; index++)
                {
                    if (!fluidGhosts.ContainsKey(index) && fluidSlots[index].HasRealFluid) slots.Add(BoundFluidSlot.WithoutGhost(index));
                }
                return slots;
            }

            #endregion
        }

        // 入力帯で itemId を受け入れる束縛スロット。Shift 移動の宛先を束縛先へ絞るのに使う
        // Bound input-band slots accepting itemId; used to narrow Shift-move destinations to bound slots
        // 製作上限では絞らない。上限超えはサーバーの拒否と既存の拒否通知に任せる
        // Not narrowed by the craft cap; the server refusal and its existing notice handle that
        public static List<int> BoundMachineInputSlotsForItem(MachineBindingDetail machine, int totalItemSlots, ItemId itemId)
        {
            var inputBand = new HashSet<int>(machine.Layout.Split(totalItemSlots).Input);
            var slots = new List<int>();
            foreach (var binding in machine.SlotBindings)
            {
                if (!inputBand.Contains(binding.Slot) || !AcceptedItems.Contains(binding.AcceptedItemIds, itemId)) continue;
                slots.Add(binding.Slot);
            }
            return slots;
        }

        // 実物入りの束縛入力だけ所持/必要を出す。空はゴーストが語り、受け入れ外の残置品は通常の個数バッジに任せる
        // Only filled bound inputs show held/required; empties use the ghost and unaccepted leftovers keep the ordinary badge
        // 出さない理由（ゴースト無し・空・受け入れ外）は 3 つあるが、どれも呼び手は「バッジを足さない」だけなので理由を返さない（判断記録 D-15）
        // Three reasons not to show (no ghost, empty, unaccepted) all mean "add no badge" to the caller, so no reason is returned (D-15)
        public static bool TryGetInputRequirement(BlockItemSlotState slot, BoundItemSlot boundSlot, out MachineInputRequirement requirement)
        {
            if (!(boundSlot is GhostedItemSlot ghosted) || !slot.HasRealItem || !ghosted.Ghost.Accepts(slot.ItemId))
            {
                requirement = default;
                return false;
            }
            requirement = new MachineInputRequirement(slot.Count, ghosted.Ghost.Count, slot.Count < ghosted.Ghost.Count);
            return true;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Client.Ui.State.BlockInventory.Logic.Machine
{
    /// <summary>
    /// 開いている機械に対応するレシピを選択行へ変える（Web machineRecipeSelectionLogic.ts の写し。名前の解決は画面側）
    /// Turns the open machine's recipes into selection rows (port of Web machineRecipeSelectionLogic.ts; name resolution stays in the screen)
    /// </summary>
    public static class MachineRecipeSelectionLogic
    {
        public static List<MachineRecipeSelectionRow> BuildRows(IReadOnlyList<MachineRecipeSummary> recipes, Guid blockGuid, Guid selectedRecipeGuid)
        {
            var hasSelection = HasSelectedRecipe(selectedRecipeGuid);
            var rows = new List<MachineRecipeSelectionRow>();
            foreach (var recipe in recipes)
            {
                // 開いている機械のレシピで、代表出力を持つものだけを行にする（液体だけを出すボイラー等も含む）
                // Rows are the open machine's recipes that have a representative output (fluid-only boilers included)
                if (recipe.BlockGuid != blockGuid) continue;
                if (!TryResolveDisplaySubject(recipe, out var subject)) continue;
                rows.Add(new MachineRecipeSelectionRow(recipe, subject, hasSelection && recipe.RecipeGuid == selectedRecipeGuid));
            }
            return rows;
        }

        public static bool HasSelectedRecipe(Guid selectedRecipeGuid)
        {
            return selectedRecipeGuid != Guid.Empty;
        }

        // 行名の「他 N 件」の N。生産物が複数あるレシピはアイコンと食い違わないよう残りの数を出す（0 以下なら出さない）
        // The N in the row name's "and N others"; multi-product recipes state the remainder so names match icons (hidden at 0 or less)
        public static int OtherOutputCount(MachineRecipeSummary recipe)
        {
            return recipe.OutputItems.Count + recipe.OutputFluids.Count - 1;
        }

        // 代表出力を決める唯一の場所。出力アイテムの先頭を優先し、無ければ出力液体の先頭。両方無ければ行にできない
        // The single place resolving the representative output: first item, else first fluid; neither means no row
        private static bool TryResolveDisplaySubject(MachineRecipeSummary recipe, out RecipeDisplaySubject subject)
        {
            if (recipe.OutputItems.Count > 0)
            {
                subject = new ItemRecipeDisplaySubject(recipe.OutputItems[0].ItemId, recipe.OutputItems[0].Count);
                return true;
            }
            if (recipe.OutputFluids.Count > 0)
            {
                subject = new FluidRecipeDisplaySubject(recipe.OutputFluids[0].FluidId, recipe.OutputFluids[0].Amount);
                return true;
            }
            subject = null;
            return false;
        }
    }
}
```

```csharp
using Client.Ui.State.Text;

namespace Client.Ui.State.BlockInventory.Logic.CleanRoom

    /// <summary>
    /// 部屋欄の各行（Web の CleanRoomDisplay）。空気・清浄機の行は密閉された部屋の派生にだけある（判断記録 D-5）
    /// The room section's rows (Web CleanRoomDisplay); air and purifier rows exist only on the enclosed subtype (D-5)
    /// </summary>
    public abstract class CleanRoomDisplay
    {
        public UiText Headline { get; }
        public bool HeadlineInsufficient { get; }
        private readonly CleanRoomHaltReason _haltReason;

        protected CleanRoomDisplay(UiText headline, bool headlineInsufficient, CleanRoomHaltReason haltReason)
        {
            Headline = headline;
            HeadlineInsufficient = headlineInsufficient;
            _haltReason = haltReason;
        }

        // 停止理由の行。止まっていなければ行なし
        // The halt-reason row; none while not halted
        public bool TryGetHaltReason(out UiText haltReason)
        {
            haltReason = _haltReason switch
            {
                CleanRoomHaltReason.None => null,
                CleanRoomHaltReason.NotEnclosed => new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomHaltNotEnclosed),
                CleanRoomHaltReason.ClassOut => new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomHaltClassOut),
                _ => throw new ArgumentOutOfRangeException(nameof(_haltReason), _haltReason, null),
            };
            return _haltReason != CleanRoomHaltReason.None;
        }
    }

    // 密閉された部屋。清浄機が 1 台も無ければ「清浄機なし」の行を出す
    // An enclosed room; with no purifier at all the "no filter" row appears
    public sealed class EnclosedCleanRoomDisplay : CleanRoomDisplay
    {
        public UiText Air { get; }
        public IReadOnlyList<UiText> Filters { get; }
        public UiText NoFilter { get; }
        public bool ShowsNoFilter => Filters.Count == 0;

        public EnclosedCleanRoomDisplay(UiText headline, bool headlineInsufficient, UiText air, IReadOnlyList<UiText> filters, CleanRoomHaltReason haltReason) : base(headline, headlineInsufficient, haltReason)
        {
            Air = air;
            Filters = filters;
            NoFilter = new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomNoFilter);
        }
    }

    // 密閉されていない・検出中は見出しだけ
    // Unsealed or detecting rooms show the headline only
    public sealed class HeadlineOnlyCleanRoomDisplay : CleanRoomDisplay
    {
        public HeadlineOnlyCleanRoomDisplay(UiText headline, bool headlineInsufficient, CleanRoomHaltReason haltReason) : base(headline, headlineInsufficient, haltReason)
```

```csharp
using System;
using System.Collections.Generic;
using Client.Localization;
using Client.Ui.State.Text;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.State.BlockInventory.Logic.CleanRoom
{
    /// <summary>
    /// 部屋欄の各行のキーと引数を作る。赤（不足）は稼働を妨げている行だけに付ける（Web cleanRoomLogic.ts の写し）
    /// Builds keys and params for each room row; red (insufficient) marks only rows that block operation (port of Web cleanRoomLogic.ts)
    /// </summary>
    public static class CleanRoomLogic
    {
        public static CleanRoomDisplay Display(CleanRoomFetchedDetail cleanRoom)
        {
            var haltReason = cleanRoom.HaltReason;
            switch (cleanRoom)
            {
                case EnclosedCleanRoomDetail enclosed:
                    return EnclosedRows(enclosed.Room, haltReason);
                case OpeningsCleanRoomDetail openings:
                    return new HeadlineOnlyCleanRoomDisplay(Line(LocalizationKeys.Ui.BlockInventory.CleanRoomNotSealed, "count", UiNumberFormat.Integer(openings.OpeningCount)), true, haltReason);
                case NoEnclosureCleanRoomDetail _:
                    return new HeadlineOnlyCleanRoomDisplay(new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomNoEnclosure), true, haltReason);
                case TooLargeCleanRoomDetail tooLarge:
                    return new HeadlineOnlyCleanRoomDisplay(Line(LocalizationKeys.Ui.BlockInventory.CleanRoomTooLarge, "max", UiNumberFormat.Integer(tooLarge.MaxRoomVolume)), true, haltReason);
                case DetectingCleanRoomDetail _:
                    // 検出待ちは数 tick で解消するので赤にしない
                    // Pending detection resolves within ticks, so it is not red
                    return new HeadlineOnlyCleanRoomDisplay(new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomDetecting), false, haltReason);
                default:
                    throw new ArgumentOutOfRangeException(nameof(cleanRoom), cleanRoom.GetType().Name, null);
            }
        }

        private static CleanRoomDisplay EnclosedRows(CleanRoomSummary room, CleanRoomHaltReason haltReason)
        {
            // 清浄機ごとの行。番号は 1 始まり、割合は四捨五入した百分率
            // One row per purifier, numbered from 1, ratios as rounded percentages
            var filters = new List<UiText>();
            for (var i = 0; i < room.AirFilters.Count; i++)
            {
                var filter = room.AirFilters[i];
                filters.Add(new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomFilter, new[]
                {
                    new LocalizationTextParam("index", UiNumberFormat.Integer(i + 1)),
                    new LocalizationTextParam("power", RoundedPercent(filter.PowerRatio)),
                    new LocalizationTextParam("count", UiNumberFormat.Integer(filter.FilterItemCount)),
                    new LocalizationTextParam("remaining", RoundedPercent(filter.FilterRemainingRatio)),
                }));
            }

            // 見出しはサーバーが決めた種別から選ぶ（Out の判定をここで持たない）
            // Pick the headline from the server-decided kind; no Out decision is made here
            var headline = room.Headline is ClassCleanRoomHeadline classHeadline
                ? Line(LocalizationKeys.Ui.BlockInventory.CleanRoomClass, "className", classHeadline.ClassName)
                : new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomClassOut);
            var air = new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomAir, new[]
            {
                new LocalizationTextParam("concentration", UiNumberFormat.Fixed(room.Concentration, 3)),
                new LocalizationTextParam("airChangeRate", UiNumberFormat.Fixed(room.AirChangeRate, 3)),
            });
            return new EnclosedCleanRoomDisplay(headline, room.Headline is ClassOutCleanRoomHeadline, air, filters, haltReason);
        }

        // 0..1 の割合を四捨五入した整数の百分率にする（Web の Math.round(x * 100)）
        // A 0..1 ratio as a rounded whole percentage (Web Math.round(x * 100))
        private static string RoundedPercent(double ratio)
        {
            return UiNumberFormat.Integer(UiNumberFormat.RoundHalfUp(ratio * 100));
        }

        private static UiText Line(LocalizationKey key, string paramName, string paramValue)
        {
            return new UiText(key, new[] { new LocalizationTextParam(paramName, paramValue) });
        }
    }
}
```

```csharp
    {
        public CleanRoomYieldRates Current { get; }

        protected CleanRoomYieldFetched(CleanRoomYieldRates current)
        {
            Current = current;
        }

        public static CleanRoomYieldFetched WithoutPendingDraw(CleanRoomYieldRates current)
        {
            return new SettledCleanRoomYield(current);

        // A fixed draw awaiting output carries only the parent's role; a mismatched pair is refused at construction

        {
            if (current.GetType() != pendingDraw.GetType()) throw new ArgumentException($"pending draw role {pendingDraw.GetType().Name} differs from {current.GetType().Name}");

    {
        public SettledCleanRoomYield(CleanRoomYieldRates current) : base(current)

        }

    {
        public CleanRoomYieldRates PendingDraw { get; }

        public PendingDrawCleanRoomYield(CleanRoomYieldRates current, CleanRoomYieldRates pendingDraw) : base(current)
```

```csharp
using System;
using System.Collections.Generic;
using Client.Localization;
using Client.Ui.State.Text;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.State.BlockInventory.Logic.CleanRoom
{
    public sealed class CleanRoomYieldLine
    {
        public string TestId { get; }
        public UiText Text { get; }

        public CleanRoomYieldLine(string testId, UiText text)
        {
            TestId = testId;
            Text = text;
        }
    }

    /// <summary>
    /// 役割に関係する歩留まりの行だけを作る。割合は小数 1 桁の百分率（Web cleanRoomYieldLogic.ts の写し）
    /// Builds only the yield rows relevant to the role; rates are percentages with one decimal (port of Web cleanRoomYieldLogic.ts)
    /// </summary>
    public static class CleanRoomYieldLogic
    {
        // 出力待ちの確定済み抽選があれば、その値と次の新規抽選の現在値を見出し付きで分けて並べる
        // With a fixed draw awaiting output, list its values and the next draw's current values apart under headings
        public static List<CleanRoomYieldLine> Lines(CleanRoomYieldFetched cleanRoomYield)
        {
            var next = RateLines(cleanRoomYield.Current, "clean-room-yield");
            if (!(cleanRoomYield is PendingDrawCleanRoomYield pending)) return next;

            var lines = new List<CleanRoomYieldLine> { new CleanRoomYieldLine("clean-room-yield-pending-label", new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomYieldPendingDrawLabel)) };
            lines.AddRange(RateLines(pending.PendingDraw, "clean-room-yield-pending"));
            lines.Add(new CleanRoomYieldLine("clean-room-yield-next-label", new UiText(LocalizationKeys.Ui.BlockInventory.CleanRoomYieldNextDrawLabel)));
            lines.AddRange(next);
            return lines;
        }

        private static List<CleanRoomYieldLine> RateLines(CleanRoomYieldRates rates, string testIdPrefix)
        {
            switch (rates)
            {
                case LayerExposureYieldRates layer:
                    return new List<CleanRoomYieldLine> { Row($"{testIdPrefix}-layer-fail-rate", LocalizationKeys.Ui.BlockInventory.CleanRoomYieldLayerFailRate, new[] { Percent(layer.LayerFailRate) }) };
                case DicingYieldRates dicing:
                    return DicingLines(dicing, testIdPrefix);
                default:
                    throw new ArgumentOutOfRangeException(nameof(rates), rates.GetType().Name, null);
            }
        }

        // ダイシングは品質 Tier・損失率・0 でないレベル割合・失敗ダイ割合の順。0 のレベルは出ないので行を作らない
        // Dicing lists tier, loss rate, non-zero level shares, then failed dies; zero levels never appear, so they get no row
        private static List<CleanRoomYieldLine> DicingLines(DicingYieldRates rates, string testIdPrefix)
        {
            var lines = new List<CleanRoomYieldLine>
            {
                Row($"{testIdPrefix}-quality-tier", LocalizationKeys.Ui.BlockInventory.CleanRoomYieldQualityTier, new[] { new LocalizationTextParam("tier", UiNumberFormat.Integer(rates.QualityTier)) }),
                Row($"{testIdPrefix}-die-loss-rate", LocalizationKeys.Ui.BlockInventory.CleanRoomYieldDieLossRate, new[] { Percent(rates.DieLossRate) }),
            };
            for (var i = 0; i < rates.LevelRates.Count; i++)
            {
                if (rates.LevelRates[i] == 0) continue;
                var level = UiNumberFormat.Integer(i + 1);
                lines.Add(Row($"{testIdPrefix}-level-rate-{level}", LocalizationKeys.Ui.BlockInventory.CleanRoomYieldLevelRate, new[] { new LocalizationTextParam("level", level), Percent(rates.LevelRates[i]) }));
            }
            lines.Add(Row($"{testIdPrefix}-failed-rate", LocalizationKeys.Ui.BlockInventory.CleanRoomYieldFailedRate, new[] { Percent(rates.FailedRate) }));
            return lines;
        }

        private static CleanRoomYieldLine Row(string testId, LocalizationKey key, LocalizationTextParam[] textParams)
        {
            return new CleanRoomYieldLine(testId, new UiText(key, textParams));
        }

        private static LocalizationTextParam Percent(double rate)
        {
            return new LocalizationTextParam("percent", UiNumberFormat.Fixed(rate * 100, 1));
        }
    }
}
```

```csharp
        Default,

                _ => throw new ArgumentOutOfRangeException(nameof(mode), mode, null),

        }
```

```csharp
using System;
using Game.Block.Interface.State;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.State.BlockInventory.Logic.Special
{
    // 段階と待ち理由はサーバーの Game.Block.Interface.State の enum をそのまま使う（判断記録 D-1）
    // Phase and wait reason reuse the server enums in Game.Block.Interface.State as they are (D-1)
    public enum RocketCommand
    {
        Build,
        Launch,
    }

    public readonly struct RocketCommandButton
    {
        public RocketCommand Command { get; }
        public LocalizationKey LabelKey { get; }

        public RocketCommandButton(RocketCommand command, LocalizationKey labelKey)
        {
            Command = command;
            LabelKey = labelKey;
        }
    }

    /// <summary>
    /// 発射台の状態文と合図ボタン（Web rocketLaunchPadLogic.ts の写し）
    /// The launch pad's status text and command button (port of Web rocketLaunchPadLogic.ts)
    /// </summary>
    public static class RocketLaunchPadLogic
    {
        // 待ち理由があれば段階より優先する
        // A wait reason, when present, takes precedence over the phase
        public static LocalizationKey StatusKey(RocketLaunchPadPhase phase, RocketLaunchPadWaitReason waitReason)
        {
            return waitReason switch
            {
                RocketLaunchPadWaitReason.None => PhaseStatusKey(phase),
                RocketLaunchPadWaitReason.MissingMaterials => LocalizationKeys.Ui.BlockInventory.RocketWaitMissingMaterials,
                RocketLaunchPadWaitReason.MissingPayload => LocalizationKeys.Ui.BlockInventory.RocketWaitMissingPayload,
                RocketLaunchPadWaitReason.OutputFull => LocalizationKeys.Ui.BlockInventory.RocketWaitOutputFull,
                RocketLaunchPadWaitReason.NoPower => LocalizationKeys.Ui.BlockInventory.RocketWaitNoPower,
                _ => throw new ArgumentOutOfRangeException(nameof(waitReason), waitReason, null),
            };
        }

        // 合図のボタンは初打ち上げ前の合図待ちだけに出る
        // Command buttons appear only while awaiting a command before the first launch
        public static bool TryGetCommandButton(RocketLaunchPadPhase phase, out RocketCommandButton button)
        {
            switch (phase)
            {
                case RocketLaunchPadPhase.AwaitingBuildCommand:
                    button = new RocketCommandButton(RocketCommand.Build, LocalizationKeys.Ui.BlockInventory.RocketBuild);
                    return true;
                case RocketLaunchPadPhase.AwaitingLaunchCommand:
                    button = new RocketCommandButton(RocketCommand.Launch, LocalizationKeys.Ui.BlockInventory.RocketLaunch);
                    return true;
                default:
                    button = default;
                    return false;
            }
        }

        private static LocalizationKey PhaseStatusKey(RocketLaunchPadPhase phase)
        {
            return phase switch
            {
                RocketLaunchPadPhase.Idle => LocalizationKeys.Ui.BlockInventory.RocketStatusIdle,
                RocketLaunchPadPhase.AwaitingBuildCommand => LocalizationKeys.Ui.BlockInventory.RocketStatusAwaitingBuild,
                RocketLaunchPadPhase.Building => LocalizationKeys.Ui.BlockInventory.RocketStatusBuilding,
                RocketLaunchPadPhase.AwaitingLaunchCommand => LocalizationKeys.Ui.BlockInventory.RocketStatusAwaitingLaunch,
                RocketLaunchPadPhase.ReadyToLaunch => LocalizationKeys.Ui.BlockInventory.RocketStatusReadyToLaunch,
                RocketLaunchPadPhase.Launching => LocalizationKeys.Ui.BlockInventory.RocketStatusLaunching,
                _ => throw new ArgumentOutOfRangeException(nameof(phase), phase, null),
            };
        }
    }
```

```csharp
    public enum BlockInventoryBodyKind

        FilterSplitter,
        ElectricToGear,
        TrainPlatform,
        ElectricPole,

        RocketLaunchPad,

    {
        None,
        RocketLaunchPadCommand,
    }

    public readonly struct BlockInventoryComponents
    {
        public BlockInventoryBodyKind Body { get; }
        public BlockInventoryTitleActionKind TitleAction { get; }

        public BlockInventoryComponents(BlockInventoryBodyKind body, BlockInventoryTitleActionKind titleAction)
        {
            Body = body;
            TitleAction = titleAction;
        }
    }

    /// <summary>
    /// マスタの BlockType（PascalCase）→ ビューの種類の静的な表（Web blockComponentRegistry.ts の写し）

    /// </summary>

    {
        // 未登録の種類は共通ビューへ戻す（専用 UI の無い液体ブロック等が落ちないように）
        // Unknown types fall back to the generic view (so blocks without a dedicated UI never break)
        private static readonly BlockInventoryComponents Default = new BlockInventoryComponents(BlockInventoryBodyKind.SectionStack, BlockInventoryTitleActionKind.None);

        public static readonly IReadOnlyDictionary<string, BlockInventoryComponents> Registered = new Dictionary<string, BlockInventoryComponents>
        {
            [BlockMasterElement.BlockTypeConst.FilterSplitter] = Body(BlockInventoryBodyKind.FilterSplitter),
            [BlockMasterElement.BlockTypeConst.Shaft] = Body(BlockInventoryBodyKind.SectionStack),
            [BlockMasterElement.BlockTypeConst.Gear] = Body(BlockInventoryBodyKind.SectionStack),
            [BlockMasterElement.BlockTypeConst.GearBeltConveyor] = Body(BlockInventoryBodyKind.SectionStack),
            [BlockMasterElement.BlockTypeConst.ElectricToGearGenerator] = Body(BlockInventoryBodyKind.ElectricToGear),
            [BlockMasterElement.BlockTypeConst.TrainStation] = Body(BlockInventoryBodyKind.TrainPlatform),
            [BlockMasterElement.BlockTypeConst.TrainItemPlatform] = Body(BlockInventoryBodyKind.TrainPlatform),
            [BlockMasterElement.BlockTypeConst.TrainFluidPlatform] = Body(BlockInventoryBodyKind.TrainPlatform),
            [BlockMasterElement.BlockTypeConst.ElectricPole] = Body(BlockInventoryBodyKind.ElectricPole),
            [BlockMasterElement.BlockTypeConst.SpaceElevatorBase] = Body(BlockInventoryBodyKind.SpaceElevator),
            [BlockMasterElement.BlockTypeConst.RocketLaunchPad] = new BlockInventoryComponents(BlockInventoryBodyKind.RocketLaunchPad, BlockInventoryTitleActionKind.RocketLaunchPadCommand),
        };

        public static BlockInventoryComponents Resolve(string blockType)
        {
            // silent-ok: 未登録は Chest・ElectricMachine 等の標準ビューの正規の経路で縮退ではない / 種類の取りこぼしは registeredBlockTypes.ts 一致テストが CI で知らせ、画面の欠けは S4-alpha の録画で見える
            // silent-ok: unregistered is the regular path for standard views (Chest, ElectricMachine...), not a degradation / missed kinds surface in the CI parity test and S4-alpha recordings
            return Registered.TryGetValue(blockType, out var components) ? components : Default;
        }

        private static BlockInventoryComponents Body(BlockInventoryBodyKind body)
        {
            return new BlockInventoryComponents(body, BlockInventoryTitleActionKind.None);
        }
    }
```

```csharp
using System;
using System.Collections.Generic;
using Core.Master;
using Game.Research;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.State.Research.Logic
{
    /// <summary>
    /// 研究ツリーの判断（Web researchLogic.ts の写し）。state 5 値の意味は Traits の 1 表に畳む
    /// Research tree decisions (port of Web researchLogic.ts); the meaning of the five states is folded into the single Traits table
    /// </summary>
    public static class ResearchLogic
    {
        // サーバーと同じ 2 パス: 単一受付 → 複数受付を、それぞれ宣言順に割り当てる
        // Same two passes as the server: single acceptance, then multiple acceptance, each in declaration order
        public static int[] AllocateResearchItems(IReadOnlyList<ResearchConsumeItem> consumeItems, IReadOnlyDictionary<ItemId, int> owned)
        {
            var remaining = new Dictionary<ItemId, int>();
            foreach (var pair in owned) remaining[pair.Key] = pair.Value;
            var allocated = new int[consumeItems.Count];
            AllocatePass(true);
            AllocatePass(false);
            return allocated;

            #region Internal

            void AllocatePass(bool singleAcceptance)
            {
                for (var index = 0; index < consumeItems.Count; index++)
                {
                    var item = consumeItems[index];
                    if ((item.AcceptedItemIds.Count == 1) != singleAcceptance) continue;
                    foreach (var id in item.AcceptedItemIds)
                    {
                        var have = remaining.TryGetValue(id, out var count) ? count : 0;
                        var amount = Math.Min(have, item.Count - allocated[index]);
                        allocated[index] += amount;
                        remaining[id] = have - amount;
                    }
                }
            }

            #endregion
        }

        // 1 件ごとの不足強調は表示専用。完了と所持数未受信（null）では出さない
        // Per-item shortage is display-only; hidden when completed or while owned counts are unknown (null)
        public static bool IsConsumeItemLacking(ResearchNodeState state, int required, int? allocated)
        {
            return allocated.HasValue && state != ResearchNodeState.Completed && allocated.Value < required;
        }

        // ボタン活性の正本はサーバーの state だけ。所持数は表示にしか使わない
        // Server state is the sole authority for availability; owned counts are display-only
        public static ResearchButtonState DeriveResearchButton(ResearchNodeState state)
        {
            if (state == ResearchNodeState.Completed) return new ResearchButtonState(true, false, LocalizationKeys.Ui.Research.Completed);
            Traits(state, out var preNodeMet, out var itemsSufficient);
            var tooltipKey = preNodeMet
                ? itemsSufficient ? LocalizationKeys.Ui.Research.ClickToResearch : LocalizationKeys.Ui.Research.MissingItems
                : itemsSufficient ? LocalizationKeys.Ui.Research.MissingPrerequisites : LocalizationKeys.Ui.Research.MissingItemsAndPrerequisites;
            return new ResearchButtonState(false, preNodeMet && itemsSufficient, tooltipKey);
        }

        public static NodeCardState DeriveNodeCardState(ResearchNodeState state)
        {
            var completed = state == ResearchNodeState.Completed;
            Traits(state, out var preNodeMet, out _);
            return new NodeCardState(completed, !completed && DeriveResearchButton(state).Interactable, !completed && !preNodeMet);
        }

        // 状態ラベルは枠色と同じ導出から引く。不可の理由は詳細ペインが担うのでカードでは 3 語へ畳む（ADR 0044）
        // The label reads the same derivation as the border; the detail pane explains "unavailable", so cards fold to three words (ADR 0044)
        public static LocalizationKey DeriveNodeStateLabelKey(NodeCardState cardState)
        {
            if (cardState.Completed) return LocalizationKeys.Ui.Research.Completed;
            return cardState.Ready ? LocalizationKeys.Ui.Research.StateAvailable : LocalizationKeys.Ui.Research.StateUnavailable;
        }

        // 初期フォーカス: 研究可能を優先し、無ければ素材待ちの最前線。どちらも無ければ選ばない
        // Initial focus: researchable first, else the item-lacking frontier; none when neither exists
        public static bool TryFindInitialFocusNode(IReadOnlyList<ResearchNodeSummary> nodes, out ResearchNodeSummary node)
        {
            if (TryFindFirst(nodes, ResearchNodeState.Researchable, out node)) return true;
            return TryFindFirst(nodes, ResearchNodeState.UnresearchableNotEnoughItem, out node);
        }

        private static bool TryFindFirst(IReadOnlyList<ResearchNodeSummary> nodes, ResearchNodeState state, out ResearchNodeSummary found)
        {
            foreach (var candidate in nodes)
            {
                if (candidate.State != state) continue;
                found = candidate;
                return true;
            }
            found = default;
            return false;
        }

        // state 5 値の意味の表。state が増えたらここで例外になる
        // The table of what the five states mean; a new state throws right here
        private static void Traits(ResearchNodeState state, out bool preNodeMet, out bool itemsSufficient)
        {
            (preNodeMet, itemsSufficient) = state switch
            {
                ResearchNodeState.Completed => (true, true),
                ResearchNodeState.Researchable => (true, true),
                ResearchNodeState.UnresearchableNotEnoughItem => (true, false),
                ResearchNodeState.UnresearchableNotEnoughPreNode => (false, true),
                ResearchNodeState.UnresearchableAllReasons => (false, false),
                _ => throw new ArgumentOutOfRangeException(nameof(state), state, null),
            };
        }
    }
```

```csharp
namespace Client.Ui.State.Research.Logic
```

```csharp
    /// <summary>
    /// 解放物と報酬を表示用の 1 本の並びへ集める（Web unlockEntries.ts の写し）。並びは Web と同じ種類順
    /// Gathers unlocks and rewards into one display list (port of Web unlockEntries.ts) in the same kind order as the Web
    /// </summary>
    public static class UnlockEntries
    {
        public static List<UnlockEntry> ToUnlockEntries(ResearchUnlockSet unlocks)
        {
            // ブロック → 機械レシピ → レシピ閲覧 → 報酬 → 接続ツール → 車両 → モジュール上限 → レベル継承上限
            // Blocks → machine recipes → recipe views → rewards → connect tools → cars → module limits → inheritance caps
            var entries = new List<UnlockEntry>();
            foreach (var block in unlocks.Blocks) entries.Add(new BlockUnlockEntry(entries.Count, block));
            foreach (var recipe in unlocks.MachineRecipes) entries.Add(new MachineRecipeOutputUnlockEntry(entries.Count, recipe));
            foreach (var itemId in unlocks.ItemRecipeViewItemIds) entries.Add(new ItemRecipeViewUnlockEntry(entries.Count, itemId));
            foreach (var reward in unlocks.RewardItems) entries.Add(new RewardItemUnlockEntry(entries.Count, reward));
            foreach (var guid in unlocks.ConnectToolGuids) entries.Add(new ConnectToolUnlockEntry(entries.Count, guid));
            foreach (var guid in unlocks.TrainCarGuids) entries.Add(new TrainCarUnlockEntry(entries.Count, guid));
            foreach (var limit in unlocks.ModuleSlotStackLimits) entries.Add(new ModuleSlotStackLimitUnlockEntry(entries.Count, limit));
            foreach (var cap in unlocks.LevelInheritanceCraftCaps) entries.Add(new LevelInheritanceCraftCapUnlockEntry(entries.Count, cap));
```

```csharp
using Client.Ui.State.Text;
using Game.Train.RailGraph;

using UnityEngine;

namespace Client.Ui.State.Train.Logic
{
    /// <summary>
    /// 時刻表の各操作から即時送信用の停車駅一覧を組み立てる（Web timetableEditLogic.ts の写し）。元の一覧は変えない
    /// Builds the stop list sent immediately for each operation (port of Web timetableEditLogic.ts); the source list is never mutated
    /// </summary>
    public static class TimetableEditLogic
    {
        // UI は入線方向を固定し、新しい停車駅は常に後端側へ着ける（裁定 2026-09-25）
        // The UI fixes the arrival direction; new stops always use the back side (ruling 2026-09-25)
        private const StationNodeSide UiFixedStopSide = StationNodeSide.Back;

        // 行の識別子。座標の文字列化は UiNumberFormat（InvariantCulture）に通し、ロケールで記号が変わらないようにする
        // Row identifier; coordinates go through UiNumberFormat (InvariantCulture) so locale never changes the signs
        public static string StationKey(Vector3Int position)
        {
            return $"{UiNumberFormat.Integer(position.x)},{UiNumberFormat.Integer(position.y)},{UiNumberFormat.Integer(position.z)}";
        }

        public static IReadOnlyList<TrainTimetableStop> AddStop(IReadOnlyList<TrainTimetableStop> stops, TrainTimetableStation station)
        {
            var added = new List<TrainTimetableStop>(stops) { new TrainTimetableStop(station.Position, station.Name, UiFixedStopSide) };
            return added;
        }

        // 範囲外の番号は無視して同じ一覧を返す。無視した理由は警告ログへ出す
        // Out-of-range indices are ignored and the same list is returned; the reason goes to a warning log
        public static IReadOnlyList<TrainTimetableStop> RemoveStop(IReadOnlyList<TrainTimetableStop> stops, int index)
        {
            if (index < 0 || index >= stops.Count)
            {
                Debug.LogWarning($"[TrainTimetable] remove ignored: invalid stop index {index}");
                return stops;
            }
            var removed = new List<TrainTimetableStop>(stops);
            removed.RemoveAt(index);
            return removed;
        }

        public static IReadOnlyList<TrainTimetableStop> MoveStop(IReadOnlyList<TrainTimetableStop> stops, int index, StopMoveDirection direction)
        {
            var target = index + (int)direction;
            if (index < 0 || index >= stops.Count || target < 0 || target >= stops.Count)
            {
                Debug.LogWarning($"[TrainTimetable] move ignored: stop index outside timetable {index} {direction}");
                return stops;
            }
            var moved = new List<TrainTimetableStop>(stops);
            (moved[index], moved[target]) = (moved[target], moved[index]);
            return moved;
        }

        // 空の時刻表では ON にしない（サーバーは受理して即 OFF に戻すため UI 側で塞ぐ裁定）
        // Never enable on an empty timetable (the server accepts then turns it off at once, so the UI blocks it)
        public static bool CanEnableAutoRun(IReadOnlyList<TrainTimetableStop> stops)
        {
            return stops.Count > 0;
        }

        // 表示名は座標を必ず含め、同名の別駅を見分けられるようにする
        // Labels always carry coordinates so same-named stations stay distinguishable
        public static UiText StationLabel(TrainTimetableStation station, string unnamedStationName)
        {
            var name = station.Name.Length > 0 ? station.Name : unnamedStationName;
            return new UiText(LocalizationKeys.Ui.BlockInventory.TimetableStationLabel, new[]
            {
                new LocalizationTextParam("name", name),
                new LocalizationTextParam("x", UiNumberFormat.Integer(station.Position.x)),
                new LocalizationTextParam("y", UiNumberFormat.Integer(station.Position.y)),
                new LocalizationTextParam("z", UiNumberFormat.Integer(station.Position.z)),
            });
        }

        public static List<TimetableReplaceStop> ToReplacePayload(IReadOnlyList<TrainTimetableStop> stops)
        {
            var payload = new List<TimetableReplaceStop>(stops.Count);
            foreach (var stop in stops) payload.Add(new TimetableReplaceStop(stop.Position, stop.Side));
            return payload;
        }

        // 順序が同じときだけサーバーの行番号を編集リストへ当てる。端の違いも別行、名前の違いは同じ行
        // Apply the server row index only when both orders match; a differing side is a different row, a differing name is not
        public static bool SameStopOrder(IReadOnlyList<TrainTimetableStop> left, IReadOnlyList<TrainTimetableStop> right)
        {
            if (left.Count != right.Count) return false;
            // 行の同一は送信形（座標と端）の等値 1 か所で決める
            // Row identity is decided once by the payload (position and side) equality
            var leftPayload = ToReplacePayload(left);
            var rightPayload = ToReplacePayload(right);
            for (var i = 0; i < leftPayload.Count; i++)
            {
                if (!leftPayload[i].Equals(rightPayload[i])) return false;
            }
            return true;
        }
    }
}
```

```csharp
        public IReadOnlyList<ItemId> AcceptedItemIds { get; }

        public InletSlotBinding(int slot, IReadOnlyList<ItemId> acceptedItemIds)
        {
            Slot = slot;
            AcceptedItemIds = acceptedItemIds;
        }
    }

    /// Shift 配分の判断が読むブロック在庫の部分（Web の BlockInventoryData のうち blockSlotRestriction が読む所）。開いているか・詳細が届いたかを sealed 派生で表す（判断記録 D-5）
    /// The part of the block inventory the Shift-allocation decision reads (what Web blockSlotRestriction reads of BlockInventoryData); open-or-not and detail arrival are sealed subtypes (D-5)
    /// </summary>
    public abstract class BlockSlotRestrictionInput
    {
        public static readonly BlockSlotRestrictionInput NotOpenBlock = new NotOpenBlockSlotInput();

        // 開いたブロック在庫。詳細（入れ口の束縛か機械の束縛）は届いた時点で With* で足す
        // An open block inventory; the detail (inlet or machine bindings) is added with With* once it arrives
        public static OpenBlockSlotInput OpenBlock(string blockType, bool openFailed, int itemSlotCount)
        {
            return new OpenBlockSlotInput(blockType, openFailed, itemSlotCount, BlockSlotDetail.Pending);
        }
    }

    public sealed class NotOpenBlockSlotInput : BlockSlotRestrictionInput
    {
    }

    {
        public string BlockType { get; }
        public bool OpenFailed { get; }
```

```csharp
using System.Collections.Generic;
using Client.Ui.State.BlockInventory.Logic.Common;
using Client.Ui.State.BlockInventory.Logic.Machine;
using Client.Ui.State.ItemMove;
using Core.Master;
using Mooresmaster.Model.BlocksModule;

namespace Client.Ui.State.BlockInventory.Logic.SlotRestriction
{
    /// <summary>
    /// Shift 配分の宛先候補を種別ごとに決める（Web blockSlotRestriction.ts の写し）。ホストが配る束縛が正本で、規則はここで導き直さない
    /// Settles Shift-allocation candidates per block type (port of Web blockSlotRestriction.ts); host-published bindings are the source of truth
    /// </summary>
    public static class BlockSlotRestrictionResolver
    {
        public static BlockSlotRestriction Resolve(BlockSlotRestrictionInput block, ItemId itemId)
        {
            if (!(block is OpenBlockSlotInput open)) return BlockSlotRestriction.Unrestricted;

            // 取得失敗は候補探索の前に返す
            // Propagate fetch failure before looking up candidate slots
            if (open.OpenFailed) return BlockSlotRestriction.InventoryOpenFailed;

            // 入れ口固定の種別は出力枠へ入れない
            // Fixed-inlet types never insert into their output slots
            if (open.BlockType == BlockMasterElement.BlockTypeConst.RocketLaunchPad || open.BlockType == BlockMasterElement.BlockTypeConst.SpaceElevatorBase)
            {
                return BoundByInletBindings(open, itemId);
            }
            return ResolveMachine(open, itemId);
        }

        // 束縛の未着は取得待ちとして確定の拒否と分ける
        // Keep pending bindings distinct from a confirmed refusal
        private static BlockSlotRestriction BoundByInletBindings(OpenBlockSlotInput block, ItemId itemId)
        {
            if (!(block.Detail is InletBlockSlotDetail inlet)) return BlockSlotRestriction.DetailLoading;
            var candidates = new List<int>();
            foreach (var binding in inlet.InletBindings)
            {
                if (AcceptedItems.Contains(binding.AcceptedItemIds, itemId)) candidates.Add(binding.Slot);
            }
            return BlockSlotRestriction.Bound(candidates);
        }

        // 機械は入力帯だけ束縛で絞り、出力帯とモジュール帯は候補に残す。機械以外（チェスト等）は無制限
        // Machines narrow only the input band; output and module bands stay candidates. Non-machines (chests etc.) are unrestricted
        // 詳細の未着も無制限にする（Web の !block.machine と同じ。サーバーが束縛外を拒み既存の拒否通知が出る。判断記録 D-17）
        // A pending detail is unrestricted too (same as Web !block.machine; the server refuses unbound slots with the existing notice, D-17)
        private static BlockSlotRestriction ResolveMachine(OpenBlockSlotInput block, ItemId itemId)
        {
            if (!(block.Detail is MachineBlockSlotDetail detail) || detail.Machine.SlotBindings.Count == 0) return BlockSlotRestriction.Unrestricted;
            var bands = detail.Machine.Layout.Split(block.ItemSlotCount);
            var candidates = MachineSlotGhosts.BoundMachineInputSlotsForItem(detail.Machine, block.ItemSlotCount, itemId);
            candidates.AddRange(bands.Output);
            candidates.AddRange(bands.Module);
            return BlockSlotRestriction.Bound(candidates);
        }
    }
}
```
