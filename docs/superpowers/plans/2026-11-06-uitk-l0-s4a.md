```csharp
using Core.Master;

namespace Client.Ui.State.Items
{
    /// <summary>
    /// スロット 1 つの中身（アイテムと個数）。Web の SlotData の写し
    /// One slot's contents (item and count); a port of the Web SlotData
    /// </summary>
    public readonly struct SlotItem
    {
        public readonly ItemId ItemId;
        public readonly int Count;

        public SlotItem(ItemId itemId, int count)
        {
            ItemId = itemId;
            Count = count;
        }

        // 空スロットの唯一の判定。サーバーは空を（ItemMaster.EmptyItemId, 0 個）の組で送るので個数で見る
        // The single empty-slot predicate; the server sends empty as (ItemMaster.EmptyItemId, 0), so read the count
        public bool IsEmpty => Count <= 0;

        public override string ToString()
        {
            return $"{ItemId.AsPrimitive()}x{Count}";
        }
    }
}
```

```csharp
using System.Collections.Generic;
using Core.Master;

namespace Client.Ui.State.Items
{
    /// <summary>
    /// 所持の判定と集計。recipe・research・inventory が共用する（Web shared/ownedCounts の写し）
    /// Owned predicate and tallies shared by recipe, research and inventory (port of Web shared/ownedCounts)
    /// </summary>
    public static class OwnedItemCounts
    {
        // 所持の唯一の判定。空に加え、wire の不正値（負の ID）も所持外にする（Web の共有契約）
        // The single owned predicate: excludes empties and invalid wire values (negative ids), per the Web contract
        public static bool IsOwnedSlot(SlotItem slot)
        {
            return !slot.IsEmpty && ItemMaster.EmptyItemId.AsPrimitive() < slot.ItemId.AsPrimitive();
        }

        // itemId 別の所持数。空スロット・0 個・負値は数えない
        // Owned count per itemId, skipping empties, zero and negative values
        public static IReadOnlyDictionary<ItemId, int> Build(IReadOnlyList<SlotItem> slots)
        {
            var owned = new Dictionary<ItemId, int>();
            for (var i = 0; i < slots.Count; i++)
            {
                var slot = slots[i];
                if (!IsOwnedSlot(slot)) continue;
                owned[slot.ItemId] = OwnedCountOf(owned, slot.ItemId) + slot.Count;
            }
            return owned;
        }

        // 同じアイテムが複数あるときは先頭の所持スロットだけがアンカーを持つ
        // When an item repeats, only its first owned slot carries the anchor
        public static IReadOnlyDictionary<ItemId, int> FirstSlotIndexByItemId(IReadOnlyList<SlotItem> slots)
        {
            var result = new Dictionary<ItemId, int>();
            for (var i = 0; i < slots.Count; i++)
            {
                var slot = slots[i];
                if (!IsOwnedSlot(slot) || result.ContainsKey(slot.ItemId)) continue;
                result.Add(slot.ItemId, i);
            }
            return result;
        }

        // 全必要アイテムを所持数が満たすか
        // Whether owned counts satisfy every requirement
        public static bool HasEnoughItems(IReadOnlyList<ItemAmount> requirements, IReadOnlyDictionary<ItemId, int> owned)
        {
            for (var i = 0; i < requirements.Count; i++)
            {
                if (OwnedCountOf(owned, requirements[i].ItemId) < requirements[i].Count) return false;
            }
            return true;
        }

        // 未登録は持っていない＝0 個。呼び出し側に同じ分岐を散らさない
        // Missing means not owned, i.e. zero; keeps the same branch out of callers
        public static int OwnedCountOf(IReadOnlyDictionary<ItemId, int> owned, ItemId itemId)
        {
            return owned.TryGetValue(itemId, out var count) ? count : 0;
        }
    }
```

```csharp
using System;
using System.Collections.Generic;
using Client.Ui.State.Items;
using Core.Item.Interface;
using Core.Master;

namespace Client.Ui.State.ItemMove
{
    /// <summary>
    /// 配分 1 件（宛先の番号と個数）
    /// One allocation: destination index and count
    /// </summary>
    public readonly struct PlannedMove : IEquatable<PlannedMove>
    {
        public readonly int Slot;
        public readonly int Count;

        public PlannedMove(int slot, int count)
        {
            Slot = slot;
            Count = count;
        }

        public bool Equals(PlannedMove other) => Slot == other.Slot && Count == other.Count;
        public override bool Equals(object obj) => obj is PlannedMove other && Equals(other);
        public override int GetHashCode() => HashCode.Combine(Slot, Count);
        public override string ToString() => $"{Slot}x{Count}";
    }

    /// <summary>
    /// Shift+クリックの直接移動の配分計画（uGUI PlayerInventoryDirectMover 準拠。Web planDirectMoves の写し）
    /// Direct-move allocation for Shift+click (mirrors uGUI PlayerInventoryDirectMover; port of Web planDirectMoves)
    /// </summary>
    public static class DirectMovePlanner
    {
        public static IReadOnlyList<PlannedMove> Plan(int sourceCount, ItemId itemId, int maxStack, IReadOnlyList<SlotItem> targetSlots)
        {
            // 同種スタック（空きあり）へ順に詰める
            // Fill same-item stacks that have room, in order
            var moves = new List<PlannedMove>();
            var remaining = sourceCount;
            for (var i = 0; i < targetSlots.Count && 0 < remaining; i++)
            {
                var target = targetSlots[i];
                if (target.IsEmpty || target.ItemId != itemId || maxStack <= target.Count) continue;
                var count = Math.Min(remaining, maxStack - target.Count);
                moves.Add(new PlannedMove(i, count));
                remaining -= count;
            }

            // ソースは 1 スタック分なので、残り全量は最初の空スロット 1 つで収まる
            // The source is a single stack, so the first empty slot always fits the remainder
            if (remaining <= 0) return moves;
            // silent-ok: 入り切らない分は計画しないだけで元スロットに残る / プレイヤーは元スロットの残数で画面上すぐ気づく（uGUI と同じ）
            if (TryFindFirstEmpty(targetSlots, out var empty)) moves.Add(new PlannedMove(empty, remaining));
            return moves;
        }

        // 配分→操作列の唯一の変換。候補の番号の順に宛先を並べて配分し、配分の番号を候補の番号へ戻す
        // The single allocation-to-actions conversion: allocate over candidates in order, then map back to their indices
        public static IReadOnlyList<PlannedItemAction> PlanMoveActions(ItemMoveSlotRef from, SlotItem source, ItemMoveArea targetArea, IReadOnlyList<SlotItem> targetSlots, IReadOnlyList<int> candidateIndices, IItemStackLevelLookup stackLevels)
        {
            var candidateSlots = new List<SlotItem>(candidateIndices.Count);
            foreach (var index in candidateIndices) candidateSlots.Add(targetSlots[index]);
            var moves = Plan(source.Count, source.ItemId, stackLevels.GetMaxStack(source.ItemId), candidateSlots);

            var actions = new List<PlannedItemAction>(moves.Count);
            foreach (var move in moves) actions.Add(PlannedItemAction.MoveItem(from, new ItemMoveSlotRef(targetArea, candidateIndices[move.Slot]), move.Count));
            return actions;
        }

        // 制約なしの宛先候補（全スロットの番号）
        // Candidates without a restriction: every slot index
        public static IReadOnlyList<int> AllIndicesOf(IReadOnlyList<SlotItem> slots)
        {
            var all = new int[slots.Count];
            for (var i = 0; i < all.Length; i++) all[i] = i;
            return all;
        }

        private static bool TryFindFirstEmpty(IReadOnlyList<SlotItem> slots, out int index)
        {
            for (index = 0; index < slots.Count; index++)
            {
                if (slots[index].IsEmpty) return true;
            }
            index = 0;
            return false;
        }
    }
}
```

```csharp
using System.Collections.Generic;
using Client.Ui.State.Items;

namespace Client.Ui.State.ItemMove
{
    /// <summary>
    /// 操作の瞬間に読んだプレイヤーの持ち物。描画時の値を混ぜないため計画の入力はすべてここから取る
    /// The player's inventory read at the moment of the gesture; every planner input comes from here, never from render-time values
    /// </summary>
    public sealed class PlayerInventorySnapshot
    {
        public IReadOnlyList<SlotItem> MainSlots { get; }
        public SlotItem Grab { get; }
        public IReadOnlyList<SlotItem> Equipment { get; }

        public PlayerInventorySnapshot(IReadOnlyList<SlotItem> mainSlots, SlotItem grab, IReadOnlyList<SlotItem> equipment)
        {
            MainSlots = mainSlots;
            Grab = grab;
            Equipment = equipment;
        }

        // 枠数が縮んだ直後の古い番地は対象が無いので false を返す
        // A stale ref right after the slot count shrinks has no target, so return false
        public bool TryResolveSlot(ItemMoveSlotRef target, out SlotItem slot)
        {
            switch (target.Area)
            {
                case ItemMoveArea.Grab:
                    slot = Grab;
                    return true;
                case ItemMoveArea.Equipment:
                    return TryGetAt(Equipment, target.Slot, out slot);
                case ItemMoveArea.Main:
                    return TryGetAt(MainSlots, target.Slot, out slot);
                default:
                    // プレイヤー側の操作に block の番地は来ない
                    // Player-side gestures never carry a block ref
                    slot = default;
                    return false;
            }
        }

        private static bool TryGetAt(IReadOnlyList<SlotItem> slots, int index, out SlotItem slot)
        {
            var inRange = 0 <= index && index < slots.Count;
            slot = inRange ? slots[index] : default;
            return inRange;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using Client.Ui.State.Items;
using Core.Item.Interface;

namespace Client.Ui.State.ItemMove
{
    /// <summary>
    /// ブロックスロットのクリック計画（Web shared/itemMove/blockSlotPlan の写し）
    /// Click plans for block slots (port of Web shared/itemMove/blockSlotPlan)
    /// </summary>
    public static class BlockSlotPlanner
    {
        // 左: Shift（空手・中身あり）は main へ配分 / 掴み中は全量置き / 中身ありは全量掴み
        // Left: Shift (empty-handed, filled) allocates into main / place all while holding / pick the whole stack
        public static IReadOnlyList<PlannedItemAction> PlanLeftClick(int index, SlotItem slot, bool shiftKey, int grabCount, IReadOnlyList<SlotItem> mainSlots, IItemStackLevelLookup stackLevels)
        {
            var blockRef = ItemMoveSlotRef.Block(index);
            // uGUI どおりサブ→メインだけ。ホットバーは宛先に含めない
            // Sub to main only, like uGUI; the hotbar is never a destination
            if (shiftKey && grabCount == 0 && !slot.IsEmpty) return DirectMovePlanner.PlanMoveActions(blockRef, slot, ItemMoveArea.Main, mainSlots, DirectMovePlanner.AllIndicesOf(mainSlots), stackLevels);
            if (0 < grabCount) return new[] { PlannedItemAction.BlockInventoryMoveItem(ItemMoveSlotRef.Grab, blockRef, grabCount) };
            if (!slot.IsEmpty) return new[] { PlannedItemAction.BlockInventoryMoveItem(blockRef, ItemMoveSlotRef.Grab, slot.Count) };
            return Array.Empty<PlannedItemAction>();
        }

        // 右: 掴み中は 1 個置き / 空手は split（半分の計算はコマンド側。古い個数に頼らない）
        // Right: place one while holding / split empty-handed (the command computes the half; no stale count)
        public static IReadOnlyList<PlannedItemAction> PlanRightClick(int index, SlotItem slot, int grabCount)
        {
            var blockRef = ItemMoveSlotRef.Block(index);
            if (0 < grabCount) return new[] { PlannedItemAction.BlockInventoryMoveItem(ItemMoveSlotRef.Grab, blockRef, 1) };
            if (slot.IsEmpty) return Array.Empty<PlannedItemAction>();
            return new[] { PlannedItemAction.BlockInventorySplit(blockRef) };
        }

        // ダブル: 集める先（持ち手かクリックしたスロットか）はコマンド側が現在の持ち手で決める
        // Double: the command picks the target (grab or clicked slot) from the current grab
        public static IReadOnlyList<PlannedItemAction> PlanDoubleClick(int index)
        {
            return new[] { PlannedItemAction.BlockInventoryCollect(ItemMoveSlotRef.Block(index)) };
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using Client.Ui.State.Items;

namespace Client.Ui.State.ItemMove
{
    /// <summary>
    /// プレイヤースロット（main・equipment・grab）のクリック計画（Web shared/itemMove/playerSlotPlan の写し）
    /// Click plans for player slots (port of Web shared/itemMove/playerSlotPlan)
    /// </summary>
    public static class PlayerSlotPlanner
    {
        // 左: 掴み中の空スロットは分割ドラッグ開始 / 全量置き / Shift は配分 / 中身ありは全量掴み
        // Left: an empty slot while holding starts split-drag / place all / Shift allocates / pick the whole stack
        public static PlayerSlotActionPlan PlanLeftClick(ItemMoveSlotRef target, SlotItem slot, bool shiftKey, PlayerSlotContext context)
        {
            // 掴み中は中身の有無・同 ID・別 ID によらず全量置き（別 ID の入れ替えはコマンド側）
            // While holding, place all regardless of contents or id (the command swaps a different id)
            var grabCount = context.Inventory.Grab.Count;
            if (0 < grabCount)
            {
                if (!shiftKey && slot.IsEmpty) return new BeginSplitDragPlan(target);
                return ActionsOf(PlannedItemAction.InventoryMoveItem(ItemMoveSlotRef.Grab, target, grabCount));
            }

            // 空手: 空スロットは何もしない / Shift は配分 / それ以外は全量掴み
            // Empty-handed: an empty slot does nothing / Shift allocates / otherwise pick the whole stack
            if (slot.IsEmpty) return ItemActionsPlan.Empty;
            if (shiftKey) return PlanShiftMove(target, slot, context);
            return ActionsOf(PlannedItemAction.InventoryMoveItem(target, ItemMoveSlotRef.Grab, slot.Count));
        }

        // 右: 掴み中は 1 個置き / 空手は split（半分の計算はコマンド側）
        // Right: place one while holding / split empty-handed (the command computes the half)
        public static IReadOnlyList<PlannedItemAction> PlanRightClick(ItemMoveSlotRef target, SlotItem slot, int grabCount)
        {
            if (0 < grabCount) return new[] { PlannedItemAction.InventoryMoveItem(ItemMoveSlotRef.Grab, target, 1) };
            if (slot.IsEmpty) return Array.Empty<PlannedItemAction>();
            return new[] { PlannedItemAction.InventorySplit(target) };
        }

        // ダブル: 集める先はコマンド側が現在の持ち手で決める
        // Double: the command picks the target from the current grab
        public static IReadOnlyList<PlannedItemAction> PlanDoubleClick(ItemMoveSlotRef target)
        {
            return new[] { PlannedItemAction.InventoryCollect(target) };
        }

        // Shift: ブロック UI が開いていればブロックへ、閉じていれば装備からだけ main へ戻す（旧 main⇔hotbar の振り分けは廃止）
        // Shift: into the block while its UI is open; otherwise only equipment returns to main (the old main<->hotbar swap is gone)
        private static PlayerSlotActionPlan PlanShiftMove(ItemMoveSlotRef from, SlotItem slot, PlayerSlotContext context)
        {
            if (context is BlockOpenSlotContext open) return PlanShiftIntoBlock(from, slot, open);
            // silent-ok: ブロック閉時の main からの Shift は配分先が無い仕様（Web・uGUI と同じ） / 何も動かないことはプレイヤーが画面で見て分かる
            if (from.Area != ItemMoveArea.Equipment) return ItemActionsPlan.Empty;

            var mainSlots = context.Inventory.MainSlots;
            return new ItemActionsPlan(DirectMovePlanner.PlanMoveActions(from, slot, ItemMoveArea.Main, mainSlots, DirectMovePlanner.AllIndicesOf(mainSlots), context.StackLevels));
        }

        private static PlayerSlotActionPlan PlanShiftIntoBlock(ItemMoveSlotRef from, SlotItem slot, BlockOpenSlotContext context)
        {
            // 取得状態と「受け付ける枠が無い」は、理由を保って断りとして返す（呼び出し側が通知とログを出す）
            // Fetch states and "no accepting slot" come back as refusals with their reason (the caller notifies and logs)
            switch (context.Restriction)
            {
                case BlockInventoryOpenFailed _:
                    return new ShiftMoveRefusedPlan(ShiftMoveRefusalReason.InventoryOpenFailed);
                case BlockDetailLoading _:
                    return new ShiftMoveRefusedPlan(ShiftMoveRefusalReason.DetailLoading);
                case BoundBlockSlots bound when bound.CandidateIndices.Count == 0:
                    return new ShiftMoveRefusedPlan(ShiftMoveRefusalReason.NoAcceptingBlockSlot);
            }

            // 束縛があれば候補だけを宛先にする。move_item は枠固定の入れ替えで、束縛外はサーバーが無言で拒否するため
            // A bound restriction narrows destinations: move_item swaps a fixed slot and the server silently rejects unbound ones
            var candidateIndices = context.Restriction is BoundBlockSlots bound ? bound.CandidateIndices : DirectMovePlanner.AllIndicesOf(context.BlockItemSlots);
            return new ItemActionsPlan(DirectMovePlanner.PlanMoveActions(from, slot, ItemMoveArea.Block, context.BlockItemSlots, candidateIndices, context.StackLevels));
        }

        private static ItemActionsPlan ActionsOf(PlannedItemAction action)
        {
            return new ItemActionsPlan(new[] { action });
        }
    }
}
```

```csharp
using Client.Ui.State.ItemMove;

namespace Client.Ui.State.Inventory.Logic
{
    /// <summary>
    /// 持ち物・装備・ホットバーの 3 パネル共通のスロット操作を、操作の瞬間の在庫で計画する（Web slotActions の計画部分）
    /// Plans the slot gestures shared by all three panels from the inventory at gesture time (planning half of Web slotActions)
    /// </summary>
    public static class PlayerSlotActionPlanner
    {
        public static PlayerSlotActionPlan PlanLeftDown(PlayerSlotContext context, ItemMoveSlotRef target, bool shiftKey)
        {
            if (!context.Inventory.TryResolveSlot(target, out var slot)) return new StaleSlotIgnoredPlan(target);
            return PlayerSlotPlanner.PlanLeftClick(target, slot, shiftKey, context);
        }

        public static PlayerSlotActionPlan PlanRightDown(PlayerInventorySnapshot inventory, ItemMoveSlotRef target)
        {
            if (!inventory.TryResolveSlot(target, out var slot)) return new StaleSlotIgnoredPlan(target);
            return new ItemActionsPlan(PlayerSlotPlanner.PlanRightClick(target, slot, inventory.Grab.Count));
        }

        // 右ドラッグ進入は掴み中だけ 1 個置く。空手で半分取りを連鎖させない
        // A right-drag enter places one only while holding; never chain split pickups empty-handed
        public static PlayerSlotActionPlan PlanRightEnter(PlayerInventorySnapshot inventory, ItemMoveSlotRef target)
        {
            // silent-ok: 空手の右ドラッグは何もしない仕様（Web・uGUI と同じ） / 持ち手が空なのはプレイヤーに見えている
            if (inventory.Grab.Count <= 0) return ItemActionsPlan.Empty;
            if (!inventory.TryResolveSlot(target, out var slot)) return new StaleSlotIgnoredPlan(target);
            return new ItemActionsPlan(PlayerSlotPlanner.PlanRightClick(target, slot, inventory.Grab.Count));
        }

        // 左ボタンのまま入ったスロットは分割ドラッグの行き先候補（行き先の判定はセッションとコマンド側）
        // A slot entered with the left button held is a split-drag candidate (the session and command decide)
        public static PlayerSlotActionPlan PlanLeftEnter(PlayerInventorySnapshot inventory, ItemMoveSlotRef target)
        {
            if (!inventory.TryResolveSlot(target, out _)) return new StaleSlotIgnoredPlan(target);
            return new EnterSplitDragPlan(target);
        }

        public static PlayerSlotActionPlan PlanDoubleClick(PlayerInventorySnapshot inventory, ItemMoveSlotRef target)
        {
            if (!inventory.TryResolveSlot(target, out _)) return new StaleSlotIgnoredPlan(target);
            return new ItemActionsPlan(PlayerSlotPlanner.PlanDoubleClick(target));
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using Client.Ui.State.ItemMove;

namespace Client.Ui.State.Inventory.Logic
{
    /// <summary>
    /// 分割ドラッグで通ったスロットを重複なしで集め、離したときに 1 回だけ渡す（Web splitDrag の写し）
    /// Collects unique slots crossed during a split drag and hands them over once on release (port of Web splitDrag)
    /// </summary>
    public sealed class SplitDragSession
    {
        private readonly List<ItemMoveSlotRef> _slots = new();

        public bool IsActive { get; private set; }

        public void Begin(ItemMoveSlotRef slot)
        {
            IsActive = true;
            _slots.Clear();
            _slots.Add(slot);
        }

        public void Enter(ItemMoveSlotRef slot)
        {
            // silent-ok: ドラッグ外のホバーと同じスロットへの再進入は行き先に足さない仕様 / 配り先の数は離した後の配分結果で見える
            if (!IsActive || _slots.Contains(slot)) return;
            _slots.Add(slot);
        }
```

```csharp
using System;

namespace Client.Ui.State.Inventory.Logic
{
    /// <summary>
    /// ホイール量の単位（Web WheelEvent.deltaMode の 0・1・2）
    /// Unit of a wheel delta (Web WheelEvent.deltaMode 0, 1, 2)
    /// </summary>
    public enum WheelDeltaUnit
    {
        Pixel,
        Line,
        Page,
    }

    /// <summary>
    /// ホイール累積の結果。進める段数は -1・0・1 のどれか
    /// Result of wheel accumulation; steps is one of -1, 0 or 1
    /// </summary>
    public readonly struct WheelStepResult
    {
        public readonly double Remainder;
        public readonly int Steps;

        public WheelStepResult(double remainder, int steps)
        {
            Remainder = remainder;
            Steps = steps;
        }
    }

    /// <summary>
    /// 装備選択の純ロジック。実スロット間の循環とホイール量の累積（Web equipmentLogic の写し）
    /// Pure equipment-selection logic: cycling across real slots and wheel accumulation (port of Web equipmentLogic)
    /// </summary>
    public static class EquipmentSelectionLogic
    {
        // 0..slotCount-1 を周期 slotCount で循環させる（負の delta は同じ環を逆走）
        // Cycle 0..slotCount-1 with period slotCount; a negative delta walks the same ring backwards
        public static int CycleEquipment(int current, int delta, int slotCount)
        {
            // 装備枠 0 は循環先が無い。呼び出し側が枠 0 で呼ぶのは配線の誤りなので止める
            // Zero equipment slots have nowhere to cycle; calling with zero is a wiring bug, so stop
            if (slotCount <= 0) throw new ArgumentOutOfRangeException(nameof(slotCount), slotCount, "装備枠が 0 のときは循環できない");
            return ((current + delta) % slotCount + slotCount) % slotCount;
        }

        // 累積して閾値を越えたら段数によらず 1 段だけ進める（スクロール加速で 1 ノッチが多段にならない）
        // Accumulate and advance exactly one step past the threshold, so acceleration cannot turn one notch into many
        public static WheelStepResult AccumulateWheelSteps(double remainder, double delta, WheelDeltaUnit unit)
        {
            var normalized = delta / NotchSizeOf(unit);

            // 逆回転では溜まっていた端数を捨て、前の回転の残りで即発火させない
            // On a reversal drop the opposite leftover so the previous rotation cannot fire immediately
            var carried = remainder * normalized < 0 ? 0 : remainder;
            var total = carried + normalized;
            if (1 <= total) return new WheelStepResult(0, 1);
            if (total <= -1) return new WheelStepResult(0, -1);
            return new WheelStepResult(total, 0);
        }

        // 単位ごとの 1 ノッチの量。OS・デバイスで桁が変わるので単位側で正規化する
        // One notch per unit; magnitudes vary by OS and device, so normalize per unit
        private static double NotchSizeOf(WheelDeltaUnit unit)
        {
            switch (unit)
            {
                case WheelDeltaUnit.Pixel: return 100;
                case WheelDeltaUnit.Line: return 3;
                case WheelDeltaUnit.Page: return 1;
                default: throw new ArgumentOutOfRangeException(nameof(unit), unit, "未知のホイール単位");
            }
        }
    }
}
```

```csharp
    /// A recipe's level inheritance (Web "none", "inherit", "inheritUpToUnlocked"); a union where only inheriting carries indices and families
    /// </summary>
    public abstract class RecipeLevelInheritance
    {
        private static readonly RecipeLevelInheritance NoInheritance = new NoLevelInheritance();

        private protected RecipeLevelInheritance()
        {
        }

        public static RecipeLevelInheritance None() => NoInheritance;

        public static RecipeLevelInheritance Inherit(int sourceInputIndex, IReadOnlyList<ItemId> sourceItemIds, int targetOutputIndex, IReadOnlyList<ItemId> targetItemIds, int maxCraftableLevel)
        {
            return new LevelInheritanceMapping(false, sourceInputIndex, sourceItemIds, targetOutputIndex, targetItemIds, maxCraftableLevel);
        }

        public static RecipeLevelInheritance InheritUpToUnlocked(int sourceInputIndex, IReadOnlyList<ItemId> sourceItemIds, int targetOutputIndex, IReadOnlyList<ItemId> targetItemIds, int maxCraftableLevel)
        {
            return new LevelInheritanceMapping(true, sourceInputIndex, sourceItemIds, targetOutputIndex, targetItemIds, maxCraftableLevel);
        }
    }

    // 継承しない。番号も並びも持たない
    // Does not inherit; carries no indices or families
    public sealed class NoLevelInheritance : RecipeLevelInheritance
    {
        internal NoLevelInheritance()
        {
        }
    }

    // 継承元の入力のレベルを継承先の出力へ写す

    {

        // True caps at the researched level (Web inheritUpToUnlocked); false is uncapped (inherit)
        public bool CappedAtUnlockedLevel { get; }
        public int SourceInputIndex { get; }
        public IReadOnlyList<ItemId> SourceItemIds { get; }
        public int TargetOutputIndex { get; }
        public IReadOnlyList<ItemId> TargetItemIds { get; }

        // 今作れる最高レベル（上限つきは研究済みの上限、上限なしは継承元の並びの長さ）
        // Highest craftable level now (the researched cap when capped, the source family length otherwise)
        public int MaxCraftableLevel { get; }

        internal LevelInheritanceMapping(bool cappedAtUnlockedLevel, int sourceInputIndex, IReadOnlyList<ItemId> sourceItemIds, int targetOutputIndex, IReadOnlyList<ItemId> targetItemIds, int maxCraftableLevel)
        {
            CappedAtUnlockedLevel = cappedAtUnlockedLevel;

            SourceItemIds = sourceItemIds;
            TargetOutputIndex = targetOutputIndex;
            TargetItemIds = targetItemIds;

        }
    }
```

```csharp
using System;
using System.Collections.Generic;
using Client.Ui.State.Items;
using Client.Ui.State.Recipe.Data;
using Core.Master;

namespace Client.Ui.State.Recipe.Logic
{
    /// <summary>
    /// レシピ一覧の組み立てとクラフト可否・最大制作数（Web features/recipe/logic/craftLogic の写し）
    /// Builds the recipe list and judges craftability and maximum output (port of Web craftLogic)
    /// </summary>
    public static class CraftLogic
    {
        // 選んだアイテムを作るレシピを、クラフトを先頭・機械を後ろにそれぞれデータ順で並べる
        // Recipes producing the item: crafts first, then machines, each in data order
        public static IReadOnlyList<RecipeEntry> BuildRecipeEntries(IReadOnlyList<CraftRecipeData> craftRecipes, IReadOnlyList<MachineRecipeData> machineRecipes, ItemId itemId)
        {
            var entries = new List<RecipeEntry>();
            foreach (var recipe in craftRecipes)
            {
                if (recipe.ResultItemId == itemId) entries.Add(new CraftRecipeEntry(recipe));
            }
            foreach (var recipe in machineRecipes)
            {
                if (ProducesItem(recipe, itemId)) entries.Add(new MachineRecipeEntry(recipe));
            }
            return entries;
        }

        public static bool Craftable(CraftRecipeData recipe, IReadOnlyDictionary<ItemId, int> counts)
        {
            return OwnedItemCounts.HasEnoughItems(recipe.RequiredItems, counts);
        }

        // 完成品ごとの最大制作数。同じ完成品のレシピが複数あれば多い方を採る（uGUI ItemListView と同じ）
        // Maximum output per result item; the larger wins across recipes of the same result (same as uGUI ItemListView)
        public static IReadOnlyDictionary<ItemId, int> CraftableResultCounts(IReadOnlyList<CraftRecipeData> recipes, IReadOnlyDictionary<ItemId, int> counts)
        {
            var result = new Dictionary<ItemId, int>();
            foreach (var recipe in recipes)
            {
                var outputCount = MaxCraftTimes(recipe, counts) * recipe.ResultCount;
                if (OwnedItemCounts.OwnedCountOf(result, recipe.ResultItemId) < outputCount) result[recipe.ResultItemId] = outputCount;
            }
            return result;
        }

        // 素材ごとの「所持数÷必要数」の最小。素材の無いレシピは 0 回（Web の MAX_SAFE_INTEGER 番兵と同じ帰結）
        // Minimum of owned/required across materials; a recipe without materials yields 0 (same outcome as the Web sentinel)
        private static int MaxCraftTimes(CraftRecipeData recipe, IReadOnlyDictionary<ItemId, int> counts)
        {
            if (recipe.RequiredItems.Count == 0) return 0;
            var times = int.MaxValue;
            foreach (var required in recipe.RequiredItems) times = Math.Min(times, OwnedItemCounts.OwnedCountOf(counts, required.ItemId) / required.Count);
            return times;
        }

        // 継承レシピは基準の生産物に加え、継承先の全レベルも生産物として扱う
        // An inheritance recipe also produces every level of its target besides the base output
        private static bool ProducesItem(MachineRecipeData recipe, ItemId itemId)
        {
            foreach (var output in recipe.OutputItems)
            {
                if (output.ItemId == itemId) return true;
            }
            return recipe.LevelInheritance is LevelInheritanceMapping mapping && ContainsItem(mapping.TargetItemIds, itemId);
        }

        private static bool ContainsItem(IReadOnlyList<ItemId> itemIds, ItemId itemId)
        {
            foreach (var candidate in itemIds)
            {
                if (candidate == itemId) return true;
            }
            return false;
        }
    }
}
```

```csharp
namespace Client.Ui.State.Recipe.Logic
{
    /// <summary>
    /// 長押しクラフトの 1 フレームの結果
    /// Result of one hold-to-craft frame
    /// </summary>
    public readonly struct HoldCraftStep
    {
        public readonly double Elapsed;
        public readonly bool DidCraft;

        public HoldCraftStep(double elapsed, bool didCraft)
        {
            Elapsed = elapsed;
            DidCraft = didCraft;
        }
    }

    /// <summary>
    /// 長押しクラフトの経過時間モデル（uGUI CraftButton.Update の累積を純関数化した Web holdCraftLogic の写し）
    /// Elapsed-time model for hold-to-craft (port of Web holdCraftLogic, itself from uGUI CraftButton.Update)
    /// </summary>
    public static class HoldCraftLogic
    {
        // 復帰直後の巨大な経過で連続発火しないよう、呼び出し側が 1 フレームの進みをこの秒数で丸める
        // Callers cap one frame's progress to this many seconds so a huge resume delta cannot burst-fire
        public const double MaxFrameDeltaSeconds = 0.25;

        // craftTime に届いたら 1 回発火して経過を 0 に戻す（連続クラフト）。1 フレーム最大 1 回
        // On reaching craftTime fire once and reset to 0 (continuous craft); at most once per frame
        public static HoldCraftStep Advance(double elapsed, double deltaSeconds, double craftTime)
        {
            var next = elapsed + deltaSeconds;
            if (craftTime <= next) return new HoldCraftStep(0, true);
            return new HoldCraftStep(next, false);
        }

        // 進捗率 0..1。craftTime<=0 は常に満杯（uGUI SetProgressAllow 相当）、NaN は 0
        // Progress 0..1; craftTime<=0 is always full (uGUI SetProgressAllow), NaN is 0
        public static double Progress(double elapsed, double craftTime)
        {
            if (craftTime <= 0) return 1;
            var ratio = elapsed / craftTime;
            if (double.IsNaN(ratio) || ratio < 0) return 0;
            return 1 < ratio ? 1 : ratio;
        }
    }
}
```

```csharp
        {
            // 0 点でも 1 列を床にし、0 列・0 除算を出さない
            // Floor at one column so zero items never yield zero columns or a divide-by-zero
            var columns = Math.Max(1, Math.Min(count, MaxSlotColumns));

            // 列間の gap を (列数-1) 本引いて等分し、上限で頭打ちにする
            // Subtract (columns-1) gaps, split evenly and cap at the maximum
            var fitted = (columnWidth - (columns - 1) * slotGap) / columns;
            return new RecipeSlotLayoutResult(columns, Math.Min(slotSizeMax, fitted));
        }
    }
}
```

```csharp
using System;
using System.Globalization;
using Client.Localization;
using Client.Ui.State.Recipe.Data;
using Client.Ui.State.Text;

namespace Client.Ui.State.Recipe.Logic

    /// <summary>
    /// 継承レシピのバッジの文（Web levelInheritanceBadge の写し）。継承しないレシピにはバッジを出さない
    /// Badge copy for an inheritance recipe (port of Web levelInheritanceBadge); no badge for a non-inheriting recipe
    /// </summary>
    public static class LevelInheritanceBadgeText
    {
        public static bool TryBuild(RecipeLevelInheritance inheritance, out UiText text)
        {
            switch (inheritance)
            {
                case NoLevelInheritance _:
                    text = null;
                    return false;
                case LevelInheritanceMapping { CappedAtUnlockedLevel: false }:
                    text = new UiText(LocalizationKeys.Ui.Recipe.LevelInheritance);
                    return true;
                case LevelInheritanceMapping capped:
                    // 上限つきは今の上限レベルを差し込む
                    // A capped recipe interpolates its current maximum level
                    var level = capped.MaxCraftableLevel.ToString(CultureInfo.InvariantCulture);
                    text = new UiText(LocalizationKeys.Ui.Recipe.LevelInheritanceCapped, new[] { new LocalizationTextParam("level", level) });
                    return true;
                default:
                    throw new ArgumentOutOfRangeException(nameof(inheritance), inheritance.GetType().Name, "未知のレベル継承の型");
            }
        }
    }
```
