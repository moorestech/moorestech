# UITK 段階 2 部品「面・装飾」（C2-a） Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** Web 版の `shared/ui/GamePanel`（default・craft・hud・skit）・`FadeRule`・Mantine `Paper`・HUD 見出し（チャレンジ HUD と操作モード HUD の同形の組）を、Web と同じ見た目の UITK 部品（`GamePanel`・`GamePanelSurface`・`PaperSurface`・`FadeRule`・`HudSectionHeading`）として `Client.Ui/Components/Panels/`・`Decor/` に作り、L1・部品ギャラリーの見本・参照キャプチャとの並べ比べまで済ませて `feature/uitk` へ取り込む。

**Architecture:** 部品はドメインを知らない `VisualElement` 派生で、表示する値は `SetXxx` で受け、見た目は部品ごとの USS（`GameUiTheme.tss` の族別区画 `Components/Panels`・`Components/Decor` に `@import`）とトークンで決める。Web の局所上書き変数（`--panel-*`・`--title-*`）と、Web に変数の無い差分（`--paper-*`・`--hud-*`）は「つまみ」として `:root` に既定を置き（既定は F1 が `DerivedTokens.uss` に入れる。F1 計画 R-8b）、画面 USS が祖先で上書きする。グラデ・マスク・三角は段階 1 の前例（`GamePanelFace` の頂点カラー格子・Painter2D）で描き、C# は値を持たずトークンを `UssCustomStyleReader` で読む。

**Tech Stack:** Unity UI Toolkit（ランタイム）、USS カスタムプロパティ、`generateVisualContent`（頂点カラー・Painter2D）、NUnit + `UnityEngine.UIElements.TestFramework`（`UiTestHost`）、uloop、ffmpeg（参照の切り抜き）

- **パスの略記**: `S/` = `moorestech_client/Assets/Scripts/`、`W/` = `moorestech_web/webui/src/`。Files 節と各コマンドはリポジトリ相対の完全なパスで書く
- **正本**: `docs/uitk-migration/execution/component-architecture.md`（設計書。§1.4 C7・§2.2・§2.6・§3）、`docs/uitk-migration/execution/execution-plan.md`（レーン C2-a の行・§4.3）、裁定 `origin/master:.decisions/2026-11-06-UITK*.md`、統合計画 `origin/master:docs/uitk-migration/plan.md`。食い違ったら 裁定 → 統合計画 → 設計書 → 実行計画 → この計画 の順で上が勝つ。この計画が設計書と違っていた所（A-22〜A-26）は、全体レビューの回に設計書へ反映済み（実行計画 E-12。判断記録 D-13）

```csharp
using System.Globalization;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;
using Client.Ui.Foundation.Theme;
using NUnit.Framework;
using UnityEngine;

namespace Client.Tests.Ui.Components
{
    /// <summary>
    /// テーマのトークン USS 群をテキストで読み、var() をたどって最終の値を返す（部品 L1 の期待値用）
    /// Reads the theme token USS files as text and follows var() to the final value (expected values for component L1)
    /// </summary>
    public static class ThemeTokenFile
    {
        private const string ThemeDirectory = "Assets/Scripts/Client.Ui/Foundation/Theme";
        private const float ColorTolerance = 0.004f;
        private static readonly Regex VarReference = new(@"^var\((--[\w-]+)\)$");
        private static readonly Regex RgbFunction = new(@"rgba?\(\s*([\d.]+)\s*,\s*([\d.]+)\s*,\s*([\d.]+)\s*(?:,\s*([\d.]+)\s*)?\)");

        public static string Resolve(string tokenName)
        {
            // var(--x) だけの値は参照先をたどる
            // A value that is just var(--x) is followed to its target
            var value = ReadDeclaredValue(tokenName);
            var reference = VarReference.Match(value);
            return reference.Success ? Resolve(reference.Groups[1].Value) : value;
        }

        public static float ResolvePixels(string tokenName)
        {
            var value = Resolve(tokenName);
            Assert.IsTrue(UssTokenValue.TryParsePixels(value, out var pixels), $"{tokenName} = {value}");
            return pixels;
        }

        public static Color ResolveColor(string tokenName)
        {
            // #hex と名前の色は Unity の解釈に任せ、rgb()/rgba() は 0..255 と α で読む
            // Hex and named colors go through Unity's parser; rgb()/rgba() are read as 0..255 channels and alpha
            var value = Resolve(tokenName);
            if (ColorUtility.TryParseHtmlString(value, out var parsed)) return parsed;
            var match = RgbFunction.Match(value);
            Assert.IsTrue(match.Success, $"{tokenName} = {value}");
            var alpha = match.Groups[4].Success ? Channel(4) : 1f;
            return new Color(Channel(1) / 255f, Channel(2) / 255f, Channel(3) / 255f, alpha);

            #region Internal

            float Channel(int group)
            {
                return float.Parse(match.Groups[group].Value, CultureInfo.InvariantCulture);
            }

            #endregion
        }

        public static void AssertSameColor(Color expected, Color actual, string what)
        {
            Assert.AreEqual(expected.r, actual.r, ColorTolerance, $"{what}.r");
            Assert.AreEqual(expected.g, actual.g, ColorTolerance, $"{what}.g");
            Assert.AreEqual(expected.b, actual.b, ColorTolerance, $"{what}.b");
            Assert.AreEqual(expected.a, actual.a, ColorTolerance, $"{what}.a");
        }

        private static string ReadDeclaredValue(string tokenName)
        {
            // 宣言はテーマの USS のどれか 1 枚にある。名前の前に英数とハイフンが続かない所だけを拾う
            // The declaration lives in one of the theme USS files; only matches not preceded by a word char or hyphen count
            var pattern = new Regex(@"(?<![\w-])" + Regex.Escape(tokenName) + @"\s*:\s*([^;]+);");
            var directory = Path.Combine(Path.GetDirectoryName(Application.dataPath), ThemeDirectory);
            foreach (var path in Directory.GetFiles(directory, "*.uss").OrderBy(path => path))
            {
                var match = pattern.Match(File.ReadAllText(path));
                if (match.Success) return Regex.Replace(match.Groups[1].Value, @"/\*.*?\*/", "").Trim();
            }
            Assert.Fail($"{tokenName} is not declared in {ThemeDirectory}");
            return null;
        }
    }
}
```

```csharp
using Client.Ui.Foundation.Theme;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Decor
{
    /// <summary>
    /// 両端を透明へ落とす横グラデの罫線要素。停止位置は Web の 0%・5%・11%・89%・95%・100%（FadeRule の基底で、GamePanel の題の罫線も直に使う）
    /// Horizontal rule element fading both ends; stops are the Web 0/5/11/89/95/100% (base of FadeRule, also used directly for the GamePanel title rules)
    /// </summary>
    /// <remarks>
    /// USS に linear-gradient が無いため Painter2D で塗る。色を受け直す 3 つの変数名は使う側が渡し、3 色は同じ RGB で α だけ違うので RGB は中央の色から取る
    /// USS has no linear-gradient, so Painter2D paints it; callers pass the three re-declared variable names, and since the colors share one RGB the RGB comes from the middle color
    /// </remarks>
    public class FadeGradientRule : VisualElement
    {
        private readonly string _startProperty;
        private readonly string _peakProperty;
        private readonly string _endProperty;
        private readonly Gradient _gradient = new();
        private bool _hasColors;

        public FadeGradientRule(string startProperty, string peakProperty, string endProperty)
        {
            _startProperty = startProperty;
            _peakProperty = peakProperty;
            _endProperty = endProperty;
            pickingMode = PickingMode.Ignore;
            generateVisualContent += DrawRule;
            RegisterCallback<CustomStyleResolvedEvent>(OnCustomStyleResolved);
        }

        private void OnCustomStyleResolved(CustomStyleResolvedEvent evt)
        {
            // 読めなくなったときも描き直し、前回のグラデを残さない
            // Repaint even when unreadable so the previous gradient never lingers
            _hasColors = TryBuildGradient();
            MarkDirtyRepaint();
        }

        private bool TryBuildGradient()
        {
            // 3 色とも読めたときだけ組む。読めない名前は UssCustomStyleReader が LogError で出す
            // Build only when all three resolve; UssCustomStyleReader logs any unresolved name
            var hasStart = UssCustomStyleReader.TryReadColor(customStyle, _startProperty, out var start);
            var hasPeak = UssCustomStyleReader.TryReadColor(customStyle, _peakProperty, out var peak);
            var hasEnd = UssCustomStyleReader.TryReadColor(customStyle, _endProperty, out var end);
            if (!(hasStart && hasPeak && hasEnd)) return false;

            var opaquePeak = new Color(peak.r, peak.g, peak.b, 1f);
            _gradient.SetKeys(
                new[] { new GradientColorKey(opaquePeak, 0f), new GradientColorKey(opaquePeak, 1f) },
                new[]
                {
                    new GradientAlphaKey(0f, 0f), new GradientAlphaKey(start.a, 0.05f), new GradientAlphaKey(peak.a, 0.11f),
                    new GradientAlphaKey(peak.a, 0.89f), new GradientAlphaKey(end.a, 0.95f), new GradientAlphaKey(0f, 1f),
                });
            return true;
        }

        private void DrawRule(MeshGenerationContext context)
        {
            // silent-ok: 色の解決前は描かない / 解決できなければ UssCustomStyleReader が LogError を出す
            if (!_hasColors) return;

            // 要素の矩形いっぱいを左から右への横グラデで塗る
            // Fill the element rect with a left-to-right gradient
            var size = contentRect.size;
            var painter = context.painter2D;
            painter.fillColor = Color.white;
            painter.fillGradient = FillGradient.MakeLinearGradient(_gradient, Vector2.zero, new Vector2(size.x, 0f), AddressMode.Clamp);
            painter.BeginPath();
            painter.MoveTo(Vector2.zero);
            painter.LineTo(new Vector2(size.x, 0f));
            painter.LineTo(size);
            painter.LineTo(new Vector2(0f, size.y));
            painter.ClosePath();
            painter.Fill();
        }
    }
}
```

```csharp
using Client.Ui.Foundation.Theme;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Panels
{
    /// <summary>
    /// パネルの半透明の面。上下左右の縁を透明へ落とすフェードを頂点カラーの格子で描く（Web の ::before とその mask）
    /// Translucent panel face; draws the fades of all four edges into clear as a vertex-colored grid (the Web ::before and its mask)
    /// </summary>
    /// <remarks>
    /// 色とフェード幅は持ち主の USS が --game-panel-face・--game-panel-edge-fade（左右）・--game-panel-top-fade・--game-panel-bottom-fade で渡す。0px の辺はフェードしない
    /// The owner USS passes color and spans as --game-panel-face, --game-panel-edge-fade (sides), --game-panel-top-fade and --game-panel-bottom-fade; a 0px side does not fade
    /// </remarks>
    public sealed class GamePanelFace : VisualElement
    {
        private const int GridLineCount = 4;

        private readonly float[] _gridXs = new float[GridLineCount];
        private readonly float[] _gridYs = new float[GridLineCount];
        private Color _faceColor;
        private float _sideFade;
        private float _topFade;
        private float _bottomFade;
        private bool _hasTokens;

        public GamePanelFace(string ownerClassName)
        {
            AddToClassList("game-panel-face");
            AddToClassList(ownerClassName);
            pickingMode = PickingMode.Ignore;
            generateVisualContent += DrawFace;
            RegisterCallback<CustomStyleResolvedEvent>(OnCustomStyleResolved);
        }

        private void OnCustomStyleResolved(CustomStyleResolvedEvent evt)
        {
            var hasColor = UssCustomStyleReader.TryReadColor(customStyle, "--game-panel-face", out _faceColor);
            var hasSide = UssCustomStyleReader.TryReadPixels(customStyle, "--game-panel-edge-fade", out _sideFade);
            var hasTop = UssCustomStyleReader.TryReadPixels(customStyle, "--game-panel-top-fade", out _topFade);
            var hasBottom = UssCustomStyleReader.TryReadPixels(customStyle, "--game-panel-bottom-fade", out _bottomFade);
            _hasTokens = hasColor && hasSide && hasTop && hasBottom;
            MarkDirtyRepaint();
        }

        private void DrawFace(MeshGenerationContext context)
        {
            // silent-ok: トークンが読めない間は描かない / UssCustomStyleReader が読めなかったトークン名を LogError で出す
            if (!_hasTokens) return;

            // 格子の線を置き、16 頂点に色、3×3 セルに三角 2 枚ずつを張る
            // Place the grid lines, color 16 vertices and span two triangles per 3x3 cell
            var width = contentRect.width;
            var height = contentRect.height;
            SetGridLines();
            var mesh = context.Allocate(GridLineCount * GridLineCount, (GridLineCount - 1) * (GridLineCount - 1) * 6);
            WriteVertices();
            WriteIndices();

            #region Internal

            void SetGridLines()
            {
                var side = Mathf.Min(_sideFade, width / 2f);
                _gridXs[0] = 0f;
                _gridXs[1] = side;
                _gridXs[2] = width - side;
                _gridXs[3] = width;
                _gridYs[0] = 0f;
                _gridYs[1] = Mathf.Min(_topFade, height);
                _gridYs[2] = Mathf.Max(height - _bottomFade, _gridYs[1]);
                _gridYs[3] = height;
            }

            // 4 辺の重みの積を α に掛ける（Web の mask-composite: intersect と同じく積を取る）
            // Multiply alpha by the product of the four edge weights (the same product as the Web mask-composite: intersect)
            void WriteVertices()
            {
                for (var row = 0; row < GridLineCount; row++)
                for (var column = 0; column < GridLineCount; column++)
                {
                    var x = _gridXs[column];
                    var y = _gridYs[row];
                    var tint = _faceColor;
                    tint.a *= Ramp(y, _topFade) * Ramp(height - y, _bottomFade) * Ramp(x, _sideFade) * Ramp(width - x, _sideFade);
                    mesh.SetNextVertex(new Vertex { position = new Vector3(x, y, Vertex.nearZ), tint = tint });
                }
            }

            // 画面座標（y 下向き）で時計回りに張る
            // Wind clockwise in screen space (y down)
            void WriteIndices()
            {
                for (var row = 0; row < GridLineCount - 1; row++)
                for (var column = 0; column < GridLineCount - 1; column++)
                {
                    var topLeft = (ushort)(row * GridLineCount + column);
                    var topRight = (ushort)(topLeft + 1);
                    var bottomLeft = (ushort)(topLeft + GridLineCount);
                    var bottomRight = (ushort)(bottomLeft + 1);
                    mesh.SetNextIndex(topLeft);
                    mesh.SetNextIndex(topRight);
                    mesh.SetNextIndex(bottomRight);
                    mesh.SetNextIndex(topLeft);
                    mesh.SetNextIndex(bottomRight);
                    mesh.SetNextIndex(bottomLeft);
                }
            }

            // 縁からの距離をフェード幅で割った重み。幅 0 の辺は常に 1（mask を掛けない辺）
            // Distance from the edge over the fade span; a zero-span side is always 1 (an unmasked side)
            float Ramp(float distance, float fade)
            {
                return fade <= 0f ? 1f : Mathf.Clamp01(distance / fade);
            }

            #endregion
        }
    }
}
```

```csharp
    /// </remarks>

        private readonly VisualElement _titleAction;
        private readonly VisualElement _body;
        private GamePanelTitle _title;

        public override VisualElement contentContainer => _body;

        public GamePanel(string testid, GamePanelFrame frame)
        {
            name = testid;
            AddToClassList("game-panel");
            _header = new VisualElement();
            _header.AddToClassList("game-panel__header");
            _titleAction = new VisualElement { pickingMode = PickingMode.Ignore };
            _titleAction.AddToClassList("game-panel__title-action");
            _header.Add(_titleAction);
            _body = new VisualElement();
            _body.AddToClassList("game-panel__body");

            // 額縁ごとの子の並びを 1 か所で決める。背後の面と装飾（Web の z-index 0）を先に、craft のグリップ（z-index 2）を最後に置く
            // Decide each frame's child order in one place: backdrop (Web z-index 0) first, the craft grip (z-index 2) last
            switch (frame)
            {
                case GamePanelFrame.Default:
                    AddToClassList("game-panel--default");
                    hierarchy.Add(new GamePanelFace("game-panel__face"));
                    hierarchy.Add(new GamePanelBottomDeco());
                    AddTitleRowsAndBody();
                    break;
                case GamePanelFrame.Craft:
                    AddToClassList("game-panel--craft");
                    hierarchy.Add(Decoration("game-panel__craft-face"));
                    hierarchy.Add(Decoration("game-panel__craft-rim"));
                    hierarchy.Add(Decoration("game-panel__craft-line"));
                    AddTitleRowsAndBody();
                    hierarchy.Add(new GamePanelCraftGrip());
                    break;
                default:
                    throw new ArgumentOutOfRangeException(nameof(frame), frame, null);
            }

            #region Internal

            // 題の行と上下の罫線は SetTitle で --titled が付くまで USS が隠す
            // USS hides the title row and its rules until SetTitle adds --titled
            void AddTitleRowsAndBody()
            {
                hierarchy.Add(DecoLine("game-panel__deco-line--top"));
                hierarchy.Add(_header);
                hierarchy.Add(DecoLine("game-panel__deco-line--bottom"));
                hierarchy.Add(_body);
            }

            // 題の罫線は FadeRule と同じ要素で、色は位置のクラスが --game-panel-rule-* で渡す（Web の .decoLine）
            // Title rules reuse the FadeRule element; the position class passes colors via --game-panel-rule-* (the Web .decoLine)
            VisualElement DecoLine(string positionClassName)
            {
                var line = new FadeGradientRule("--game-panel-rule-start", "--game-panel-rule-peak", "--game-panel-rule-end");
                line.AddToClassList("game-panel__deco-line");
                line.AddToClassList(positionClassName);
                return line;
            }

            VisualElement Decoration(string className)
            {
                var decoration = new VisualElement { pickingMode = PickingMode.Ignore };
                decoration.AddToClassList(className);
                return decoration;
            }

        }

        {

            // Create the title only once; never attach a keyless LocalizedLabel (that logs an error)
            if (_title == null)
            {
                _title = new GamePanelTitle();
                _header.Insert(0, _title);
            }
            _title.SetText(title);
            AddToClassList("game-panel--titled");
        }

        public void SetTitleAction(VisualElement action)
        {
            // 副次アクションは 1 つだけ。差し替えは前の要素を外してから足す
            // Only one title action; replacing removes the previous element first
            _titleAction.Clear();
            _titleAction.Add(action);
        }
    }
```

```csharp
using System;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Panels
{
    /// <summary>
    /// 枠を持たない面だけのパネル（hud・skit）。題を持たないことを型で分ける（Web の GamePanel の FaceOnlyProps）
    /// Face-only panel without a frame (hud, skit); the type keeps titles unreachable (the Web GamePanel FaceOnlyProps)
    /// </summary>
    public sealed class GamePanelSurface : VisualElement
    {
        private readonly VisualElement _body;

        public override VisualElement contentContainer => _body;

        public GamePanelSurface(string testid, GamePanelSurfaceKind kind)
        {
            name = testid;
            AddToClassList("game-panel-surface");
            AddToClassList(KindClassName(kind));

            // 面は本文の背後。色とフェードは種類のクラスが USS で渡す
            // The face sits behind the body; the kind class passes its color and fades through USS
            hierarchy.Add(new GamePanelFace("game-panel-surface__face"));
            _body = new VisualElement();
            _body.AddToClassList("game-panel-surface__body");
            hierarchy.Add(_body);
        }

        private static string KindClassName(GamePanelSurfaceKind kind)
        {
            return kind switch
            {
                GamePanelSurfaceKind.Hud => "game-panel-surface--hud",
                GamePanelSurfaceKind.Skit => "game-panel-surface--skit",
                _ => throw new ArgumentOutOfRangeException(nameof(kind), kind, null),
            };
        }
    }
}
```
