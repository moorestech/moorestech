# UITK 段階 3 部品「スロット族 1」（C3-a） Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** Web 版の `shared/ui` のスロット族（`SlotFrame`・`useSlotMouse`・`slotContent.module.css`・`GameIcon`・`ItemSlot`・`BlockSlot`・`PlacementTargetFace`・`SlotGrid`）を、Web と同じ見た目の UITK 部品として `Client.Ui/Components/Slots/` に作り、ID から表示素材を引く Content 層（`GameContentPresentation`）と表示素材の値型 `UiIcon` を足して、L1・部品ギャラリー・参照キャプチャとの並べ比べ（720p・2560×1080）まで済ませて `feature/uitk` へ取り込む。

**Architecture:** 部品はドメインを知らない `VisualElement` 派生で、アイコンは `UiIcon`（テクスチャ・ID の文字・名前）、数は `int`、文言は `UiText` で受ける。ID → `UiIcon` の解決は `Client.Ui/Content/` の `GameContentPresentation` 1 か所に置く（設計書 A-4）。枠の 3 層のベベルは Web の `box-shadow` の重なり順（先頭が最前）をそのまま `border` の帯 3 本＋面際の帯 1 本で写し、帯が箱の外へ育つか内へ食い込むかだけを `--bevel-inset` のキーワードで C# が決める。Web の局所上書き変数（`--icon-pad`・`--count-*`・`--shortage-count-*`・`--slot-grid-*`・`--sel-c*`・`--selected-face-*`・`--filled-face-inset`・`--face-inset-color`・`--bevel-inset`）は同名のつまみとして既定を `DerivedTokens.uss` の `:root` に置き、画面 USS が祖先で上書きする。

**Tech Stack:** Unity 6000.3.8f1 UI Toolkit（ランタイムパネル・USS カスタムプロパティ）、UniRx、VContainer、NUnit＋`com.unity.ui.test-framework`（`UiTestHost`・`PanelSimulator`）、uloop CLI、プレイテスト DSL（`run-scenario.sh`）、ffmpeg（参照の切り抜き）、gh CLI、bd（Beads）

- **パスの略記**: `S/` = `moorestech_client/Assets/Scripts/`、`W/` = `moorestech_web/webui/src/`。Files 節と各コマンドはリポジトリ相対の完全なパスで書く
- **正本**: 裁定 `.decisions/2026-11-06-UITK*.md`、統合計画 `docs/uitk-migration/plan.md`、設計書 `docs/uitk-migration/execution/component-architecture.md`（§2.1・§2.3・§3・§7.1。この計画で A-50〜A-54 を足した）、実行計画 `docs/uitk-migration/execution/execution-plan.md`（レーン C3-a の行・§4.3・§6）。食い違ったら 裁定 → 統合計画 → 設計書 → 実行計画 → この計画 の順で上が勝つ
- **ゴール（ユーザーの原文）**: 「WebUIと同じ見た目で、適切な設計、コンポーネント化されたUITKベースのUIを開発すること」

```csharp
namespace Client.Ui.Components.Slots
```

```csharp
    /// <summary>
    /// テクスチャ 1 枚の表示と、欠けたときの代わりの文字（Web の GameIcon）
    /// Shows one texture, or fallback text when it is missing (the Web GameIcon)
    /// </summary>
    public sealed class GameIcon : VisualElement
    {
        private const string ImageStateClass = "game-icon--image";
        private const string FallbackStateClass = "game-icon--fallback";

        private readonly VisualElement _image;
        private LocalizedLabel _fallbackLabel;

        public GameIcon()
        {
            pickingMode = PickingMode.Ignore;
            AddToClassList("game-icon");
            _image = new VisualElement { pickingMode = PickingMode.Ignore };
            _image.AddToClassList("game-icon__image");
            Add(_image);
        }

        public void SetIcon(UiIcon icon, GameIconFallback fallback)
        {
            // テクスチャがあれば絵だけを出す
            // With a texture, show only the picture
            if (icon.Texture != null)
            {
                _image.style.backgroundImage = new StyleBackground(icon.Texture);
                EnableInClassList(ImageStateClass, true);
                EnableInClassList(FallbackStateClass, false);
                return;
            }

            // 欠けたら Web の onError と同じく fallback の種類で代わりを決める
            // When missing, pick the substitute by fallback kind as the Web onError does
            _image.style.backgroundImage = StyleKeyword.Null;
            EnableInClassList(ImageStateClass, false);
            switch (fallback)
            {
                case GameIconFallback.IdText:
                    ShowFallback(SlotTexts.IconIdFallback(icon.IdText));
                    break;
                case GameIconFallback.Label:
                    ShowFallback(icon.Label);
                    break;
                case GameIconFallback.None:
                    EnableInClassList(FallbackStateClass, false);
                    break;
            }
        }

        public void ClearIcon()
        {
            _image.style.backgroundImage = StyleKeyword.Null;
            EnableInClassList(ImageStateClass, false);
            EnableInClassList(FallbackStateClass, false);
        }

        private void ShowFallback(UiText text)
        {
            // キーの無い LocalizedLabel をパネルに載せない（LogError になる）ので初回に作る

            if (_fallbackLabel == null)

                _fallbackLabel.AddToClassList("game-icon__fallback");
                Add(_fallbackLabel);
            }
            _fallbackLabel.SetText(text);
            EnableInClassList(FallbackStateClass, true);
        }
```

```csharp
using System;
using System.Collections.Generic;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// ボタンの番号と押下中のビットから身振りを決める（Web の useSlotMouse）
    /// Decides gestures from the button index and held-button bits (the Web useSlotMouse)
    /// </summary>
    public static class SlotGestureRules
    {
        // UITK の button は 0 左・1 右、pressedButtons は bit0 左・bit1 右（Web の button 0/2・buttons 1/2 と同じ意味）
        // UITK button is 0 left / 1 right and pressedButtons is bit0 left / bit1 right (same meaning as Web button 0/2 and buttons 1/2)
        private const int LeftButton = 0;
        private const int RightButton = 1;
        private const int LeftHeldBit = 1;
        private const int RightHeldBit = 2;
        private const int DoubleClickCount = 2;

        private static readonly SlotGesture[] LeftEnterOnly = { new(SlotGestureKind.LeftEnter, false) };
        private static readonly SlotGesture[] RightEnterOnly = { new(SlotGestureKind.RightEnter, false) };
        private static readonly SlotGesture[] BothEnters = { new(SlotGestureKind.LeftEnter, false), new(SlotGestureKind.RightEnter, false) };

        public static bool TryFromPointerDown(int button, bool shift, out SlotGesture gesture)
        {
            switch (button)
            {
                case LeftButton:
                    gesture = new SlotGesture(SlotGestureKind.LeftDown, shift);
                    return true;
                case RightButton:
                    gesture = new SlotGesture(SlotGestureKind.RightDown, false);
                    return true;
            }

            // silent-ok: 中ボタン等は Web でも身振りにならない（契約どおりの無反応） / 契約は SlotGestureRulesTest が固定し、変えれば開発者がテストで気づく
            // silent-ok: middle and other buttons are no gesture on the Web either (the contract) / SlotGestureRulesTest pins it, so a change surfaces in tests
            gesture = default;
            return false;
        }

        public static IReadOnlyList<SlotGesture> FromPointerEnter(int pressedButtons)
        {
            // 左の進入を先に、右を後に出す（Web の onMouseEnter の判定順）
            // Emit the left enter first and the right second (the Web onMouseEnter order)
            var left = (pressedButtons & LeftHeldBit) != 0;
            var right = (pressedButtons & RightHeldBit) != 0;
            if (left && right) return BothEnters;
            if (left) return LeftEnterOnly;
            if (right) return RightEnterOnly;
            return Array.Empty<SlotGesture>();
        }

        // Web の dblclick は左ボタンの 2 回目のクリックで 1 回出る
        // The Web dblclick fires once on the second left click
        public static bool IsDoubleClick(int button, int clickCount)
        {
            return button == LeftButton && clickCount == DoubleClickCount;
```

```csharp
using Client.Ui.Foundation.Theme;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// SlotFrame の box-shadow 4 層（面際 1・ベベル 3）を border の帯で描く装飾子。寸法に効かない
    /// Decorator drawing the SlotFrame four box-shadow layers (face inset and three bevels) as border bands; it never affects layout
    /// </summary>
    public sealed class SlotFrameRing : VisualElement
    {
        private const string BevelInsetProperty = "--slot-frame-bevel-inset";
        private static readonly string[] BevelWidthProperties = { "--slot-frame-bevel-1", "--slot-frame-bevel-2", "--slot-frame-bevel-3" };

        private readonly VisualElement[] _bands = new VisualElement[3];

        public SlotFrameRing()
        {
            pickingMode = PickingMode.Ignore;
            AddToClassList("slot-frame__ring");

            // box-shadow は先に書いた層が上に描かれる。c3 → c2 → c1 → 面際の順に足し、後の子を上にする
            // box-shadow paints earlier layers on top; add c3, c2, c1 then the face inset so later children draw above
            for (var layer = 3; layer >= 1; layer--)
            {
                var band = new VisualElement { pickingMode = PickingMode.Ignore };
                band.AddToClassList("slot-frame__band");
                band.AddToClassList($"slot-frame__band--{layer}");
                _bands[layer - 1] = band;
                Add(band);
            }
            var faceInset = new VisualElement { pickingMode = PickingMode.Ignore };
            faceInset.AddToClassList("slot-frame__face-inset");
            Add(faceInset);

            RegisterCallback<CustomStyleResolvedEvent>(_ => ApplyBevelGeometry());
        }

        private void ApplyBevelGeometry()
        {
            // 読めないときは前の形のまま（LogError で開発者が気づく）
            // When unreadable keep the previous shape (the LogError tells the developer)
            if (!TryReadGrowsInward(out var growsInward)) return;
            for (var i = 0; i < _bands.Length; i++)
            {
                if (!UssCustomStyleReader.TryReadPixels(customStyle, BevelWidthProperties[i], out var width)) return;

                // outset は幅ぶん箱の外へ、inset は箱の内側に帯を置く（Web の spread と同じ形）
                // outset puts the band one width outside the box, inset keeps it inside (the same shape as the Web spread)
                var offset = growsInward ? 0f : -width;
                var band = _bands[i];
                band.style.left = offset;
                band.style.top = offset;
                band.style.right = offset;
                band.style.bottom = offset;
                band.style.borderLeftWidth = width;
                band.style.borderTopWidth = width;
                band.style.borderRightWidth = width;
                band.style.borderBottomWidth = width;
            }
        }

        private bool TryReadGrowsInward(out bool growsInward)
        {
            // 読めないときの LogError は UssCustomStyleReader が出す
            // UssCustomStyleReader logs the error when it cannot read the value
            growsInward = false;
            if (!UssCustomStyleReader.TryReadKeyword(customStyle, BevelInsetProperty, out var keyword)) return false;
            switch (keyword)
            {
                case "inset":
                    growsInward = true;
                    return true;
                case "outset":
                    return true;
            }
            Debug.LogError($"[SlotFrameRing] '{BevelInsetProperty}' must be inset or outset but was '{keyword}'");
            return false;
        }
    }
```

```csharp
using System;
using UniRx;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// 1 マスの枠・ベベル・面と 5 状態、マウスの身振り（Web の SlotFrame と useSlotMouse）
    /// One cell's frame, bevel, face, five states and mouse gestures (the Web SlotFrame and useSlotMouse)
    /// </summary>
    public sealed class SlotFrame : VisualElement
    {
        private readonly Subject<SlotGesture> _slotGestured = new();
        private readonly VisualElement _content;

        public IObservable<SlotGesture> OnSlotGestured => _slotGestured;
        public override VisualElement contentContainer => _content;

        public SlotFrame(string testid)
        {
            name = testid;
            AddToClassList("slot-frame");

            // 下端の帯 → リング → 中身の順に重ね、Web の背景・box-shadow・子の重なりにそろえる
            // Stack the bottom trim, ring and content in this order to match the Web background, box-shadow and child layering
            var selectedTrim = new VisualElement { pickingMode = PickingMode.Ignore };
            selectedTrim.AddToClassList("slot-frame__selected-trim");
            hierarchy.Add(selectedTrim);
            hierarchy.Add(new SlotFrameRing());
            _content = new VisualElement { pickingMode = PickingMode.Ignore };
            _content.AddToClassList("slot-frame__content");
            hierarchy.Add(_content);

            // 押下・進入・離脱・クリックを身振りへ訳す
            // Translate press, enter, leave and click into gestures
            RegisterCallback<PointerDownEvent>(OnPointerDown);
            RegisterCallback<PointerMoveEvent>(OnPointerMove);
            RegisterCallback<PointerEnterEvent>(OnPointerEnter);
            RegisterCallback<PointerLeaveEvent>(_ => _slotGestured.OnNext(new SlotGesture(SlotGestureKind.HoverLeave, false)));
            RegisterCallback<ClickEvent>(OnClick);
        }

        public void SetState(SlotFrameState state)
        {
            EnableInClassList("slot-frame--selected", state.Selected);
            EnableInClassList("slot-frame--filled", state.Filled);
            EnableInClassList("slot-frame--catalog", state.Catalog);
            EnableInClassList("slot-frame--insufficient", state.Insufficient);
            EnableInClassList("slot-frame--ghost", state.Ghost);
        }

        private void OnPointerDown(PointerDownEvent evt)
        {
            if (SlotGestureRules.TryFromPointerDown(evt.button, evt.shiftKey, out var gesture)) _slotGestured.OnNext(gesture);
        }

        private void OnPointerMove(PointerMoveEvent evt)
        {
            // 別ボタンを押したまま追加で押すと UITK は PointerDownEvent でなく PointerMoveEvent で届ける（Web の onMouseDown は毎回出る）
            // While another button is held, an extra press arrives as a PointerMoveEvent, not a PointerDownEvent (Web onMouseDown fires every time)
            if (evt.button < 0 || (evt.pressedButtons & (1 << evt.button)) == 0) return;
            if (SlotGestureRules.TryFromPointerDown(evt.button, evt.shiftKey, out var gesture)) _slotGestured.OnNext(gesture);
        }

        private void OnPointerEnter(PointerEnterEvent evt)
        {
            // 押したままの進入を先に、ホバーの開始を後に出す（Web の onMouseEnter の順）
            // Emit held-button enters first and the hover start after (the Web onMouseEnter order)
            foreach (var gesture in SlotGestureRules.FromPointerEnter(evt.pressedButtons)) _slotGestured.OnNext(gesture);
            _slotGestured.OnNext(new SlotGesture(SlotGestureKind.HoverEnter, false));
        }

        private void OnClick(ClickEvent evt)
        {
            // ボタンの番号の意味は SlotGestureRules だけが持つ
            // Only SlotGestureRules knows what the button numbers mean
            if (SlotGestureRules.IsDoubleClick(evt.button, evt.clickCount)) _slotGestured.OnNext(new SlotGesture(SlotGestureKind.DoubleClick, false));
        }
    }
}
```

```csharp
using Client.Ui.Foundation.Icons;
using Client.Ui.State.Text;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// ItemSlotContent から面・中身・バッジ・ツールチップを決める（Web の ItemSlot 本体の式）
    /// Derives face, content, badge and tooltip from ItemSlotContent (the Web ItemSlot body formulas)
    /// </summary>
    public sealed class ItemSlotPresentation
    {
        public bool Filled { get; }
        public bool GhostShown { get; }
        public UiIcon ShownIcon { get; }
        public bool ShowsBadge { get; }
        public int BadgeCount { get; }
        public UiText Tooltip { get; }

        private ItemSlotPresentation(bool filled, bool ghostShown, UiIcon shownIcon, bool showsBadge, int badgeCount, UiText tooltip)
        {
            Filled = filled;
            GhostShown = ghostShown;
            ShownIcon = shownIcon;
            ShowsBadge = showsBadge;
            BadgeCount = badgeCount;
            Tooltip = tooltip;
        }

        public static ItemSlotPresentation From(ItemSlotContent content)
        {
            // カタログは常に絵を出し、白面は所持数がある時だけ（Web の owned・hasItem・filled）
            // A catalog cell always shows the icon and turns white only when owned (the Web owned, hasItem and filled)
            var count = content.Count;
            var owned = count.HasValue && count.Value > 0;
            var hasItem = content.Icon != null && (content.Marks.Catalog || !count.HasValue || count.Value > 0);
            var filled = content.Marks.Catalog ? owned : hasItem;

            // 実物の無い空きマスに限りゴーストを描く。白面は実物だけで決める
            // The ghost draws only on a genuinely empty cell; the white face is decided by the real item alone
            var ghostShown = content.Ghost != null && !hasItem;
            var shownIcon = ghostShown ? content.Ghost.Icon : hasItem ? content.Icon : null;
            var shownCount = ghostShown ? content.Ghost.Count : count;

            // 所持/必要がある間は個数を出さず、ツールチップは不足の文 → 上書き → 表示中の名前の順
            // While owned/required shows the count stays hidden; the tooltip prefers the shortage text, then the override, then the shown name
            var showsBadge = content.Shortage == null && (owned || ghostShown) && shownCount.HasValue && shownCount.Value > 0;
            var tooltip = shownIcon == null ? null : content.Shortage?.Tooltip ?? content.TooltipOverride ?? shownIcon.Label;
            return new ItemSlotPresentation(filled, ghostShown, shownIcon, showsBadge, showsBadge ? shownCount.Value : 0, tooltip);
        }
    }
}
```

```csharp
using System;
using Client.Ui.Foundation.Text;
using Client.Ui.Foundation.Tooltip;
using Client.Ui.State.Text;
using Mooresmaster.Localization.Generated;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// アイテム 1 マス（個数・所持/必要・ゴースト・ツールチップ）。Web の ItemSlot
    /// One item cell (count, owned/required, ghost and tooltip); the Web ItemSlot
    /// </summary>
    public sealed class ItemSlot : VisualElement
    {
        private const string ShortageStateClass = "item-slot--shortage";
        private const string LackStateClass = "item-slot__shortage--lack";

        private readonly SlotFrame _frame;
        private readonly SlotContent _content;
        private LocalizedLabel _shortageLabel;

        public IObservable<SlotGesture> OnSlotGestured => _frame.OnSlotGestured;

        public ItemSlot(string frameTestid)
        {
            // 所持/必要は枠の外へはみ出し減光も受けないので、枠を包む外側の層に置く（Web の .shortageSlot）。testid は Web と同じく枠へ
            // Owned/required overflows the frame and escapes its dimming, so it lives in this wrapping layer (the Web .shortageSlot); the testid goes on the frame as on the Web
            pickingMode = PickingMode.Ignore;
            AddToClassList("item-slot");
            _frame = new SlotFrame(frameTestid);
            _content = new SlotContent();
            _frame.Add(_content);
            Add(_frame);
        }

        public void SetContent(ItemSlotContent content)
        {
            var presentation = ItemSlotPresentation.From(content);
            _frame.SetState(new SlotFrameState(content.Marks.Selected, presentation.Filled, content.Marks.Catalog, content.Marks.Insufficient, presentation.GhostShown));
            ApplyShownContent(presentation);
            ApplyShortage(content.Shortage);
        }

        private void ApplyShownContent(ItemSlotPresentation presentation)
        {
            // 中身の無いマスは絵も個数もツールチップも出さない（Web の disabled）
            // An empty cell shows no icon, count or tooltip (the Web disabled)
            if (presentation.ShownIcon == null)
            {
                _content.ClearIcon();
                _content.ClearCount();
                HoverTooltipMark.Unmark(_frame);
                return;
            }
            _content.SetIcon(presentation.ShownIcon, GameIconFallback.IdText);
            if (presentation.ShowsBadge) _content.SetCount(presentation.BadgeCount);
            else _content.ClearCount();
            HoverTooltipMark.Mark(_frame, presentation.Tooltip);
        }

        private void ApplyShortage(ItemSlotShortage shortage)
        {
            // 包みは常に同じで、所持/必要の有無はラベルの出し入れだけで切り替える（Web の再マウント防止）
            // The wrapper never changes; owned/required toggles only the label (the Web remount guard)
            EnableInClassList(ShortageStateClass, shortage != null);
            if (shortage == null) return;

            // キーの無い LocalizedLabel をパネルに載せないよう初回に作る
            // Create on first use so a key-less LocalizedLabel never sits on the panel
            if (_shortageLabel == null)
            {
                _shortageLabel = new LocalizedLabel { pickingMode = PickingMode.Ignore };
                _shortageLabel.AddToClassList("item-slot__shortage");
                Add(_shortageLabel);
            }
            _shortageLabel.SetText(SlotTexts.ItemCountSummary(shortage.OwnedCount, shortage.RequiredCount));
            _shortageLabel.EnableInClassList(LackStateClass, shortage.Lacking);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// 列数固定のスロットの格子。USS に grid が無いので列数ごとの行に並べ、間隔は USS のつまみで付ける（Web の SlotGrid）
    /// Fixed-column slot grid; USS has no grid, so items go into rows of the column count and the gaps come from USS knobs (the Web SlotGrid)
    /// </summary>
    public sealed class SlotGrid : VisualElement
    {
        private const string CellClass = "slot-grid__cell";
        private const string RowStartClass = "slot-grid__cell--row-start";

        private readonly List<VisualElement> _items = new();
        private readonly VisualElement _tracks;
        private int _columns;
        private VisualElement _lastRow;

        public SlotGrid(string testid, int columns)
        {
            name = testid;
            pickingMode = PickingMode.Ignore;
            AddToClassList("slot-grid");
            _columns = RequirePositive(columns);
            _tracks = new VisualElement { pickingMode = PickingMode.Ignore };
            _tracks.AddToClassList("slot-grid__tracks");
            Add(_tracks);
            RebuildTracks();
        }

        public void SetColumns(int columns)
        {
            _columns = RequirePositive(columns);

            // 列の幅の行と行を作り直し、マスを前から詰め直す
            // Rebuild the track row and the rows, then refill the cells from the front
            ClearRows();
            RebuildTracks();
            for (var i = 0; i < _items.Count; i++) PlaceAt(i);
        }

        public void AddItem(VisualElement item)
        {
            item.AddToClassList(CellClass);
            _items.Add(item);
            PlaceAt(_items.Count - 1);
        }

        public void ClearItems()
        {
            foreach (var item in _items)
            {
                item.RemoveFromClassList(CellClass);
                item.RemoveFromClassList(RowStartClass);
            }
            _items.Clear();
            ClearRows();
        }

        private void ClearRows()
        {
            // 列の幅の行だけ残し、マスの行を外す
            // Keep only the track row and drop the cell rows
            Clear();
            Add(_tracks);
            _lastRow = null;
        }

        private void RebuildTracks()
        {
            // マスが列数より少なくても Web の repeat(cols, --slot-size) と同じ幅を保つ、高さ 0 の列の行
            // A zero-height track row that keeps the Web repeat(cols, --slot-size) width even with fewer cells than columns
            _tracks.Clear();
            for (var i = 0; i < _columns; i++)
            {
                var track = new VisualElement { pickingMode = PickingMode.Ignore };
                track.AddToClassList("slot-grid__track");
                track.EnableInClassList("slot-grid__track--row-start", i == 0);
                _tracks.Add(track);
            }
        }

        private void PlaceAt(int index)
        {
            // 行の頭のマスで新しい行を作る。最初の行だけ上の間隔を持たない
            // A row-start cell opens a new row; only the first row has no top gap
            var column = index % _columns;
            if (column == 0)
            {
                _lastRow = new VisualElement { pickingMode = PickingMode.Ignore };
                _lastRow.AddToClassList("slot-grid__row");
                _lastRow.EnableInClassList("slot-grid__row--first", index == 0);
                Add(_lastRow);
            }
            var item = _items[index];
            item.EnableInClassList(RowStartClass, column == 0);
            _lastRow.Add(item);
        }

        private static int RequirePositive(int columns)
        {
            if (columns < 1) throw new ArgumentOutOfRangeException(nameof(columns), columns, "SlotGrid needs at least one column");
            return columns;
        }
    }
}
```

```csharp
using System;
using Client.Game.InGame.Context;
using Core.Master;
using UnityEngine;

namespace Client.Ui.Content
{
    /// <summary>
    /// 本番のテクスチャの出所。ClientContext の画像コンテナを読む（作る側は全部 Texture2D）
    /// The production texture source, reading ClientContext's image containers (all built from Texture2D)
    /// </summary>
    public sealed class ClientContextContentTextureSource : IGameContentTextureSource
    {
        // 空の ID はコンテナが LogError 無しで null を返すので、引く前に止める（GameContentPresentation と同じ扱い）
        // Containers return null for an empty ID without logging, so stop before the lookup (same as GameContentPresentation)
        // silent-ok: 空でない欠けた ID は各コンテナが LogError を出して null を返す / 画面では GameIcon が #id を描き、開発者は Console の LogError で気づく
        // silent-ok: each container logs an error and returns null for a missing non-empty ID / on screen GameIcon draws #id and the developer sees the LogError
        public Texture2D ItemTexture(ItemId id)
        {
            if (id == ItemMaster.EmptyItemId) throw new ArgumentException("The empty item has no texture; pass a null icon to ItemSlotContent instead", nameof(id));
            return ClientContext.ItemImageContainer.GetItemView(id)?.ItemTexture as Texture2D;
        }

        public Texture2D BlockTexture(BlockId id)
        {
            return ClientContext.BlockImageContainer.GetBlockView(id)?.ItemTexture as Texture2D;
        }

        public Texture2D FluidTexture(FluidId id)
        {
            if (id == FluidMaster.EmptyFluidId) throw new ArgumentException("The empty fluid has no texture; the screen must not ask for it", nameof(id));
            return ClientContext.FluidImageContainer.GetItemView(id)?.FluidTexture as Texture2D;
        }
    }
}
```

```csharp
using System;
using Client.Ui.Foundation.Icons;
using Client.Ui.State.Text;
using Core.Master;
using Mooresmaster.Localization.Generated;

namespace Client.Ui.Content
{
    /// <summary>
    /// ID → 表示素材（UiIcon・名前）を 1 か所で解く Content 層（設計書 A-4）
    /// The Content layer resolving IDs into display material (UiIcon and names) in one place (design A-4)
    /// </summary>
    public sealed class GameContentPresentation
    {
        private readonly IGameContentTextureSource _textures;

        public GameContentPresentation(IGameContentTextureSource textures)
        {
            _textures = textures;
        }

        public UiIcon Item(ItemId id)
        {
            // 名前を先に解き、空の ID はテクスチャを引く前に止める（素材の側の LogError を先に出さない）
            // Resolve the name first so an empty ID stops before the texture lookup logs anything
            var name = ItemName(id);
            return new UiIcon(_textures.ItemTexture(id), id.AsPrimitive().ToString(), name);
        }

        public UiIcon Block(BlockId id)
        {
            var blockGuid = MasterHolder.BlockMaster.GetBlockMaster(id).BlockGuid;
            return new UiIcon(_textures.BlockTexture(id), id.AsPrimitive().ToString(), UiText.FromContent(ContentLocalizationKeys.BlockName(blockGuid)));
        }

        public UiIcon Fluid(FluidId id)
        {
            // 空の液体（空タンク）も表示素材にしない。マスタは Guid.Empty を黙って返すのでここで止める
            // An empty fluid (empty tank) is never display material; the master silently returns Guid.Empty, so stop here
            if (id == FluidMaster.EmptyFluidId) throw new ArgumentException("The empty fluid has no display material; the screen must not ask for it", nameof(id));
            var fluidGuid = MasterHolder.FluidMaster.GetFluidGuid(id);
            return new UiIcon(_textures.FluidTexture(id), id.AsPrimitive().ToString(), UiText.FromContent(ContentLocalizationKeys.FluidName(fluidGuid)));
        }

        public UiText ItemName(ItemId id)
        {
            // 空きは表示素材にしない。画面が ItemSlotContent の Icon を null にして表す
            // An empty ID is never display material; the screen expresses it with a null ItemSlotContent icon
            if (id == ItemMaster.EmptyItemId) throw new ArgumentException("The empty item has no display material; pass a null icon to ItemSlotContent instead", nameof(id));
            return UiText.FromContent(ContentLocalizationKeys.ItemName(MasterHolder.ItemMaster.GetItemGuid(id)));
        }
    }
}
```

```csharp
// Runtime panels do not tick in Edit Mode, so call update, repaint and offscreen render through reflection three times
var utility = typeof(VisualElement).Assembly.GetType("UnityEngine.UIElements.UIElementsRuntimeUtility");
var invoked = new System.Collections.Generic.List<string>();
for (var pass = 0; pass < 3; pass++)
{

        if (method == null) return $"missing {name}";
        method.Invoke(null, method.GetParameters().Select(p => p.ParameterType == typeof(bool) ? (object)false : null).ToArray());

}

// 読み出して PNG にし、作ったものを全部片付ける

Object.DestroyImmediate(settings);
target.Release();
Object.DestroyImmediate(target);
Object.DestroyImmediate(pixels);
return $"{output} from {source} via {string.Join(",", invoked)}";
```
