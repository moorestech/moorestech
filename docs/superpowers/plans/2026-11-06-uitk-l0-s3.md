# UITK L0-s3 段階 3 の純ロジック移植 Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** Web 版の段階 3（ビルドメニュー・ホットバー・モーダル・配置対象・押下ジェスチャ）の純ロジックと、その vitest を C# の純ロジックと NUnit へ 1 対 1 で写し、画面レーン（S3-buildmenu・S3-hotbar-modal）と部品レーン（C3-a・C3-b）が使う型を先に `feature/uitk` へ置く。

**Architecture:** ロジックは `Client.Ui.State/<機能>/Logic/`（UnityEngine に触れない static 関数と不変の値型）に置き、DOM に依存していた Web フックの計算部分は「視口のインターフェース（`IBuildMenuScrollViewport`）を受ける追跡器」に写し、Web と同じく表示層（`Client.Ui/Screens/BuildMenu/CategoryScroll/`）に置く。ViewModel・Model・コマンド・DI 登録は作らない（画面レーンの仕事）。先頭で Unity の外の `dotnet test`（S12）を試し、回ればエンジン非依存の L0 を各タスクで Unity 無しに回し、最後に Editor 枠を 1 回だけ借りて `uloop compile` と全 L0 を回す。

**Tech Stack:** C# 9（Unity 6000.3.8f1）、UniRx（`ReactiveProperty`）、NUnit（Unity Test Framework の EditMode）、検証用に `~/.dotnet/dotnet`（SDK 8.0.424）と NUnit 3.14.0、Node（`tools/uitk/test-name-parity.mjs`。W0-tools が作る）。

**パスの略記:** `S/` = `moorestech_client/Assets/Scripts/`、`W/` = `moorestech_web/webui/src/`。Files 節と git コマンドは repo 相対の完全パスで書く。

```csharp
using System;

namespace Client.Ui.State.Hotbar.Logic
{
    /// <summary>
    /// ドラッグ元と落とし先の組を割当・入替・解除へ写す。Web hotbarDnd.ts の写し
    /// Maps a drag source and drop target pair to assign, swap or clear; a port of the Web hotbarDnd.ts
    /// </summary>
    public static class HotbarDropResolver
    {
        public static HotbarDropAction Resolve(HotbarDragSource source, HotbarDropTarget target)
        {
            switch (source)
            {
                // メニューのエントリは枠へ落としたときだけ割当になる
                // A menu entry only assigns when it lands on a slot
                case BuildMenuEntryDragSource entry:
                    return target is HotbarSlotDropTarget slot
                        ? new HotbarAssignAction(slot.SlotIndex, entry.TargetId)
                        : HotbarNoDropAction.Instance;
                case HotbarSlotDragSource slotSource:
                    return ResolveFromSlot(slotSource.SlotIndex, target);
                default:
                    throw new ArgumentOutOfRangeException(nameof(source), source, "未知のホットバーのドラッグ元");
            }
        }

        private static HotbarDropAction ResolveFromSlot(int sourceSlotIndex, HotbarDropTarget target)
        {
            switch (target)
            {
                // 自分自身への落としは何も変えない
                // A drop onto itself changes nothing
                case HotbarSlotDropTarget slot:
                    return slot.SlotIndex == sourceSlotIndex
                        ? HotbarNoDropAction.Instance
                        : new HotbarSwapAction(sourceSlotIndex, slot.SlotIndex);
                case OutsideHotbarDropTarget:
                    return new HotbarClearAction(sourceSlotIndex);
                default:
                    throw new ArgumentOutOfRangeException(nameof(target), target, "未知のホットバーの落とし先");
            }
        }
    }
}
```

```csharp
namespace Client.Ui.State.Modal.Logic
{
    /// <summary>
    /// モーダルへの応答。入力文を運ぶのは入力付きモーダルの確定だけ（Web respondPayload の写し）
    /// A reply to a modal; only an input modal's confirm carries text (port of the Web respondPayload)
    /// </summary>
    public sealed class ModalResponse
    {
        public string RequestId { get; }
        public ModalResult Result { get; }

        // null は文なし。ModalRequestService.Respond の text へそのまま渡す
        // Null means no text; passed straight to ModalRequestService.Respond's text
        public string Text { get; }

        private ModalResponse(string requestId, ModalResult result, string text)
        {
            RequestId = requestId;
            Result = result;
            Text = text;
        }

        public static ModalResponse Confirm(ModalRequest request, string enteredText)
        {
            // 入力なしのモーダルでは入力欄の残り文を運ばない
            // A modal without input never carries leftover field text
            return new ModalResponse(request.Id, ModalResult.Confirm, request.RequiresInput ? enteredText : null);
        }

        public static ModalResponse Cancel(ModalRequest request)
        {
            return new ModalResponse(request.Id, ModalResult.Cancel, null);
        }
    }
}
```

```csharp
namespace Client.Ui.State.Modal.Logic
{
    /// <summary>
    /// モーダルの確定可否と色の判定（Web modalLogic.ts の canConfirm・buttonColor の写し）
    /// Modal confirm eligibility and colour (port of the Web modalLogic.ts canConfirm and buttonColor)
    /// </summary>
    public static class ModalLogic
    {
        // 入力付きモーダルは空白だけの文を確定させない（uGUI BlueprintNameInputView と同じ検証）
        // An input modal rejects whitespace-only text (same validation as uGUI BlueprintNameInputView)
        public static bool CanConfirm(ModalRequest request, string enteredText)
        {
            if (!request.RequiresInput) return true;
            return enteredText.Trim().Length > 0;
        }

        // エラーは赤、確認は青（uGUI の色分けと同じ）
        // Error is red and confirm is blue, mirroring the uGUI styling
        public static ModalButtonTone ConfirmButtonTone(ModalVariant variant)
        {
            return variant == ModalVariant.Error ? ModalButtonTone.Red : ModalButtonTone.Blue;
        }
    }
}
```

```csharp
using System;

namespace Client.Ui.Foundation.Input.Gesture
{
    /// <summary>
    /// 押下のジェスチャの純粋計算（Web shared/pointerGesture/dragThreshold.ts の写し）
    /// Pure pointer-gesture math (port of the Web shared/pointerGesture/dragThreshold.ts)
    /// </summary>
    public static class UiDragThreshold
    {
        // タップとドラッグを分ける閾値。5px 未満はタップ・選択のまま
        // Movement threshold separating tap from drag; under 5px stays a tap or selection
        public const float ThresholdPx = 5f;

        // 押下点からの移動が閾値以上になったらドラッグ確定
        // A move of at least the threshold from the press point commits to a drag
        public static bool Exceeded(float dx, float dy)
        {
            return Math.Sqrt((double)dx * dx + (double)dy * dy) >= ThresholdPx;
        }

        // 掴んだ点を基準に、ポインタを下へ動かすと内容も下へ流れるパン量
        // Natural pan: moving the pointer down from the grabbed point slides the content down
        public static float NextScrollTop(float startScrollTop, float movedY)
        {
            return startScrollTop - movedY;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

namespace Client.Ui.State.BuildMenu.Logic
{
    /// <summary>
    /// ビルドメニューの群分けと検索（Web buildMenuGrouping.ts の写し）
    /// Build-menu grouping and search (port of the Web buildMenuGrouping.ts)
    /// </summary>
    public static class BuildMenuGrouping
    {
        // カテゴリ定義順→サブカテゴリ定義順で群にし、空の群は落とす。エントリの並びは配信順のまま
        // Group by category then sub-category definition order, dropping empty groups; entry order stays as delivered
        public static IReadOnlyList<BuildMenuCategoryGroup> Group(IReadOnlyList<BuildMenuCategory> categories, IReadOnlyList<BuildMenuCatalogEntry> entries)
        {
            var buckets = BucketByCategory();
            var groups = new List<BuildMenuCategoryGroup>();
            foreach (var category in categories)
            {
                // silent-ok: カテゴリとサブカテゴリの整合はマスタのロード時に検証済みで、ここで落ちるのは検索で空になった群だけ / 不整合はマスタロードの例外で開発者が気づく
                // Category consistency is validated at master load, so only search-emptied groups drop here; an inconsistency surfaces as a master-load exception
                if (!buckets.TryGetValue(category.CategoryGuid, out var subBuckets)) continue;
                var sections = new List<BuildMenuSection>();
                foreach (var subCategoryGuid in category.SubCategoryGuids)
                {
                    if (!subBuckets.TryGetValue(subCategoryGuid, out var sectionEntries)) continue;
                    sections.Add(new BuildMenuSection(category.CategoryGuid, subCategoryGuid, sectionEntries));
                }
                if (sections.Count == 0) continue;
                groups.Add(new BuildMenuCategoryGroup(category.CategoryGuid, sections));
            }
            return groups;

            #region Internal

            // 全カテゴリを常に載せるので、総当たりの再走査を避けて 1 パスでバケツに分ける
            // Every category stays mounted, so bucket the entries in one pass instead of rescanning per category
            Dictionary<Guid, Dictionary<Guid, List<BuildMenuCatalogEntry>>> BucketByCategory()
            {
                var byCategory = new Dictionary<Guid, Dictionary<Guid, List<BuildMenuCatalogEntry>>>();
                foreach (var entry in entries)
                {
                    if (!byCategory.TryGetValue(entry.CategoryGuid, out var bySubCategory))
                    {
                        bySubCategory = new Dictionary<Guid, List<BuildMenuCatalogEntry>>();
                        byCategory.Add(entry.CategoryGuid, bySubCategory);
                    }
                    if (!bySubCategory.TryGetValue(entry.SubCategoryGuid, out var bucket))
                    {
                        bucket = new List<BuildMenuCatalogEntry>();
                        bySubCategory.Add(entry.SubCategoryGuid, bucket);
                    }
                    bucket.Add(entry);
                }
                return byCategory;
            }

            #endregion
        }

        // 表示名の部分一致（大文字小文字を無視）。空の検索語は全件
        // Case-insensitive substring search on display labels; an empty query returns everything
        public static IReadOnlyList<BuildMenuCatalogEntry> Search(string query, IReadOnlyList<BuildMenuCatalogEntry> entries)
        {
            if (query.Length == 0) return entries;
            var lowered = query.ToLowerInvariant();
            return entries.Where(entry => entry.DisplayLabel.ToLowerInvariant().Contains(lowered)).ToList();
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Client.Ui.State.BuildMenu.Logic
{
    /// <summary>
    /// 1 本スクロールの判定（Web buildMenuScrollSpy.ts の写し）。視口に依存しない
    /// Judgements for the single scroll (port of the Web buildMenuScrollSpy.ts); viewport-free
    /// </summary>
    public static class BuildMenuScrollSpy
    {
        // スムーズスクロールの停止位置は小数で揺れるので ±1px を同値とみなす（ジャンプ到達専用）
        // Smooth scrolling settles on fractional positions, so ±1px counts as equal (jump arrival only)
        private const float JumpSettleTolerancePx = 1f;

        // scroll-spy が見出しを活性にするしきい値。到達判定とは別に調整できるよう分ける
        // The scroll-spy activation threshold, kept separate from the arrival tolerance so each can be tuned
        private const float ScrollSpyActivationTolerancePx = 1f;

        // 視口上端以上にある見出しのうち上端が最大のもの。無ければ最も上の見出し
        // Among headings at or above the viewport top, the one with the largest top; otherwise the topmost heading
        public static Guid? ActiveCategoryAtScroll(IReadOnlyList<CategoryHeadingOffset> offsets, float scrollTop)
        {
            if (offsets.Count == 0) return null;
            CategoryHeadingOffset? reached = null;
            var topmost = offsets[0];
            foreach (var offset in offsets)
            {
                var isReached = offset.Top - ScrollSpyActivationTolerancePx <= scrollTop;
                if (isReached && (reached == null || offset.Top > reached.Value.Top)) reached = offset;
                if (offset.Top < topmost.Top) topmost = offset;
            }
            return (reached ?? topmost).CategoryGuid;
        }

        public static bool IsJumpSettled(float scrollTop, float targetTop)
        {
            return Math.Abs(scrollTop - targetTop) <= JumpSettleTolerancePx;
        }

        // 末尾カテゴリの見出しを視口上端まで持ち上げられるよう、足りない分を埋める
        // Fill the shortfall so the last category heading can still reach the viewport top
        public static float TrailingSpacerHeight(float viewportHeight, float lastGroupHeight)
        {
            return Math.Max(0f, viewportHeight - lastGroupHeight);
        }
    }
}
```

```csharp
    /// Decides the current category, jump pin, trailing spacer and restore of the single scroll (port of the math in the Web useBuildMenuCategoryScroll)
    /// </summary>
    public sealed class BuildMenuCategoryScrollTracker
    {
        public IReadOnlyReactiveProperty<Guid?> ActiveCategory => _activeCategory;
        public IReadOnlyReactiveProperty<float> SpacerHeight => _spacerHeight;

        // Whether the saved position was applied; reset on a viewport swap
        private bool _scrollRestored;

        public BuildMenuCategoryScrollTracker(BuildMenuSessionStore session)
        {
            _session = session;
        }

        public void AttachViewport(IBuildMenuScrollViewport viewport)
        {
            // 張り替え前の実効位置を保存し、新しい視口で復元をやり直す
            // Persist the outgoing viewport's position and redo the restore on the new one
            if (_viewport != null) _session.SetScrollTop(_viewport.ScrollTop);
            _viewport = viewport;
            _scrollRestored = false;
            SyncLayout();
        }

        public void SetVisibleCategories(IReadOnlyList<Guid> categoryGuids)
        {
            // 同じ並びの渡し直しではジャンプの固定を保つ
            // Passing the same sequence again keeps the jump pin
            if (_visibleCategories.SequenceEqual(categoryGuids)) return;
            _visibleCategories = categoryGuids.ToArray();
            _jumpPin = null;
            SyncLayout();
        }

        public void JumpTo(Guid categoryGuid)
        {
            if (_viewport == null || !_viewport.TryGetHeadingTop(categoryGuid, out var headingTop))
            {
                Debug.LogWarning($"[BuildMenuCategoryScroll] jump to {categoryGuid} ignored: the viewport or its heading is not attached");
                return;
            }
            _activeCategory.Value = categoryGuid;
            PinOrSettleJump(categoryGuid, ReachableTop(headingTop));
        }

        public void OnScrolled(float scrollTop)
        {
            _session.SetScrollTop(scrollTop);
            if (_jumpPin is { } pin)
            {
                // 未到達の間は固定を保ち、到達したら追従ではなく目標自身を確定させる
                // Keep the pin until arrival, then settle on the target itself rather than on scroll-spy
                if (!BuildMenuScrollSpy.IsJumpSettled(scrollTop, pin.Top)) return;
                SettleOnJumpTarget(pin.CategoryGuid);
                return;
            }
            RecomputeActiveCategory(scrollTop);
        }

        // ホイール・押下・キー入力でジャンプの固定をすぐ解く
        // Wheel, press or key input releases the jump pin immediately
        public void OnUserIntervened()
        {
            // silent-ok: 固定が無いときの解除は冪等 / 現在地は ActiveCategory の購読者が見る
            // Releasing with no pin is idempotent; ActiveCategory subscribers see the current category
            if (_jumpPin == null) return;
            _jumpPin = null;
            RecomputeActiveCategory(_viewport.ScrollTop);
        }

        // 視口・見出し・群・スペーサの寸法が変わったら View が呼ぶ
        // The view calls this whenever the viewport, a heading, a group or the spacer changes size
        public void OnLayoutChanged() => SyncLayout();

        public void PersistScrollPosition()
        {
            _session.SetScrollTop(_viewport.ScrollTop);
        }

        private void SyncLayout()
        {
            // silent-ok: 視口の装着前は測れない / AttachViewport が同じ処理をやり直す
            // Nothing is measurable before a viewport attaches; AttachViewport reruns this
            if (_viewport == null) return;
            var spacer = RemeasureSpacer();
            if (spacer == BuildMenuSpacerMeasure.Settled) RestoreScrollOnce();
            if (_jumpPin is not { } pin)
            {
                RecomputeActiveCategory(_viewport.ScrollTop);
                return;
            }

            // 反映前の scrollHeight で引き直すと末尾ジャンプが手前で到達扱いになるので、反映後の通知を待つ
            // Re-targeting on the pre-apply scrollHeight settles a trailing jump short, so wait for the applied notification
            if (spacer == BuildMenuSpacerMeasure.Changed) return;
            RetargetJump(pin);
        }

        private BuildMenuSpacerMeasure RemeasureSpacer()
        {
            // 末尾群が未配置の間は実測できない。0 と確定させると復元位置まで 0 へ潰れる
            // The last group is unmeasurable until laid out; settling it as 0 would flatten the restore target too
            if (_visibleCategories.Count == 0) return BuildMenuSpacerMeasure.Unmeasurable;
            var lastCategory = _visibleCategories[_visibleCategories.Count - 1];
            if (!_viewport.TryGetGroupHeight(lastCategory, out var lastGroupHeight)) return BuildMenuSpacerMeasure.Unmeasurable;

            var desired = BuildMenuScrollSpy.TrailingSpacerHeight(_viewport.ClientHeight, lastGroupHeight);
            if (desired != _spacerHeight.Value) { _spacerHeight.Value = desired; return BuildMenuSpacerMeasure.Changed; }
            // 要求済みでも View が当てるまでは反映待ち（1px 許容）。古い scrollHeight で復元を確定しない
            // Requested is not applied: wait for the view (1px tolerance) so the restore never settles on a stale scrollHeight
            return BuildMenuScrollSpy.IsSpacerApplied(_viewport.AppliedSpacerHeight, desired) ? BuildMenuSpacerMeasure.Settled : BuildMenuSpacerMeasure.Changed;
        }

        private void RestoreScrollOnce()
        {
            if (_scrollRestored) return;
            _scrollRestored = true;
            _viewport.SetScrollTopImmediately(_session.Current.ScrollTop);

            // クランプ後の実効値へ保存を揃える
            // Realign the store with the clamped effective value
            _session.SetScrollTop(_viewport.ScrollTop);
        }

        private void RetargetJump(BuildMenuJumpPin pin)
        {
            if (!_viewport.TryGetHeadingTop(pin.CategoryGuid, out var headingTop))
            {
                Debug.LogWarning($"[BuildMenuCategoryScroll] jump target {pin.CategoryGuid} lost its heading; the pin waits for user input or a visible-set change");
                return;
            }
            var top = ReachableTop(headingTop);
            if (top == pin.Top) return;
            PinOrSettleJump(pin.CategoryGuid, top);
        }

        // スペーサ込みでも届かないときは最大スクロール位置を目標にし、到達判定を成立させる
        // When even the spacer cannot reach, target the max scroll so the settle check can succeed
        private float ReachableTop(float headingTop) => Math.Min(headingTop, _viewport.ScrollHeight - _viewport.ClientHeight);

        // 目標位置にすでに居れば目標自身で確定し、居なければ固定してスムーズに動かす
        // Settle on the target if already there, otherwise pin it and scroll smoothly
        private void PinOrSettleJump(Guid categoryGuid, float top)
        {
            if (BuildMenuScrollSpy.IsJumpSettled(_viewport.ScrollTop, top))
            {
                SettleOnJumpTarget(categoryGuid);
                return;
            }
            _jumpPin = new BuildMenuJumpPin(categoryGuid, top);
            _viewport.ScrollSmoothlyTo(top);
        }

        // 到達時は scroll-spy ではなく目標自身を現在地にし、固定を解く
        // On arrival the target itself becomes current (not scroll-spy) and the pin is released
        private void SettleOnJumpTarget(Guid categoryGuid)
        {
            _jumpPin = null;
            _activeCategory.Value = categoryGuid;
        }

        private void RecomputeActiveCategory(float scrollTop)
        {
            var offsets = new List<CategoryHeadingOffset>();
            foreach (var categoryGuid in _visibleCategories)
            {
                if (_viewport.TryGetHeadingTop(categoryGuid, out var top)) offsets.Add(new CategoryHeadingOffset(categoryGuid, top));
            }
            _activeCategory.Value = BuildMenuScrollSpy.ActiveCategoryAtScroll(offsets, scrollTop);
        }
    }
}
```
