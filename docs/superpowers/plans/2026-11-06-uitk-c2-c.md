# UITK C2-c（段階 2 部品: 文字・表示・スクロール ― GaugeBar・TooltipFace・GameScrollView・KeyCap・ShortcutHintLabel・Typography・UiStack/UiGroup・albof.16.2）Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** 文字・表示・スクロールの共有部品（`GaugeBar`・`TooltipFace`・`GameScrollView`・`KeyCap`・`ShortcutHintLabel`）と、Mantine の `Text`/`Title`/`Stack`/`Group` の写し（Typography のクラス・`UiStack`/`UiGroup`）を Web 版と同じ見た目で作り、ポーズの子画面と設定画面の直書き（ScrollView・余白の足し算・見出しの寸法）をこれらへ置き換え、albof.16.2（言語選択の結果型の統合・画質プリセット導出の重複）を片付ける。

**Architecture:** 部品レーン（単位エージェント・Editor 要）。部品は 1 部品 1 ディレクトリ・1 USS で、F1 が作った `GameUiTheme.tss` の族別区画（Foundation/Theme・Components/Controls・Components/Gauges・Components/Text）へ `@import` を 1 行ずつ足す（設計書 C1・実行計画 E-1）。文言は `UiText`／`LocalizedLabel.SetText`（C9）、押下は `OnClicked`（C8）、色と寸法は Web と同名のトークン（`Tokens.uss`・`SlotTokens.uss`・F1 の `MantineTokens.uss`）だけを読み、比率（ゲージの高さ `slot-size/6`、ScrollArea のバーの内側 `size/5`）だけを C# が USS のつまみから計算する。見た目の正本は Web の CSS（`GaugeBar/style.module.css`・`panelBodyScrollArea.module.css`・`HoverTooltip`・`app/tokens.css` の `keyHintText`・`LocalizedShortcutHint.tsx`）と Mantine 8.3.18（`ScrollArea`・`Text`・`Title`・`Stack`・`Group`）で、数値はこの計画に逐語で写した。L1 は部品ごとに状態・寸法・つまみを確かめ、Web の `pauseMenuSettingsLayout.spec.ts`(2) を 2 つの論理解像度の L1 に写し、ギャラリーと参照キャプチャで並べ比べ、`uitk-lifeline` と `graphics-quality/quality-ui-switching` を UITK 側で録画して締める。最後は moores-code-review → `base=feature/uitk` の PR → エージェントがマージコミットで取り込む（実行計画 §4.3）。

**Tech Stack:** Unity 6000.3.8f1 UI Toolkit（ランタイムパネル・USS・`CustomStyleResolvedEvent`）、UniRx、NUnit＋`com.unity.ui.test-framework`（`UITestFixture`・`PanelSimulator`）、uloop CLI、プレイテスト DSL（`run-scenario.sh`）、node（Web の辞書キー生成）、gh CLI、bd（Beads）

```csharp
// moorestech_client/Assets/Scripts/Client.Ui/Foundation/Layout/UiGapContainer.cs
using System.Collections.Generic;
using Client.Ui.Foundation.Theme;
using UnityEngine.UIElements;

namespace Client.Ui.Foundation.Layout
{
    /// <summary>
    /// 表示中の先頭以外の子へ USS の --ui-gap を余白として配る並びの土台（USS に gap が無いため）
    /// Base of a row/column that hands USS --ui-gap to every shown child except the first (USS has no gap)
    /// </summary>
    /// <remarks>
    /// 余白はインラインで当てる。部品が 2 クラスで当てる margin: 0（PanelActionButton）に負けないため
    /// The space is applied inline so it beats the two-class margin: 0 of components such as PanelActionButton
    /// 子の表示切替は SetItemShown で行う。USS で隠した子は数えたまま残る
    /// Toggle children with SetItemShown; a child hidden by USS is still counted
    /// </remarks>
    public abstract class UiGapContainer : VisualElement
    {
        private const string GapProperty = "--ui-gap";
        private readonly string _rootClass;
        private readonly List<VisualElement> _items = new();
        private string _gapClass;
        private float _gapPixels;
        private bool _gapResolved;

        protected UiGapContainer(string rootClass, UiGap gap)
        {
            _rootClass = rootClass;
            AddToClassList(rootClass);
            SetGap(gap);
            RegisterCallback<CustomStyleResolvedEvent>(_ => OnCustomStyleResolved());
        }

        public void SetGap(UiGap gap)
        {
            if (_gapClass != null) RemoveFromClassList(_gapClass);
            _gapClass = $"{_rootClass}--gap-{UiGapClass.Suffix(gap)}";
            AddToClassList(_gapClass);
        }

        public void AddItem(VisualElement item)
        {
            _items.Add(item);
            Add(item);
            ApplyLeadingSpaces();
        }

        public void SetItemShown(VisualElement item, bool shown)
        {
            item.style.display = shown ? DisplayStyle.Flex : DisplayStyle.None;
            ApplyLeadingSpaces();
        }

        public void ClearItems()
        {
            _items.Clear();
            Clear();
        }

        protected abstract void SetLeadingSpace(VisualElement item, StyleLength space);

        private void OnCustomStyleResolved()
        {
            // 読めないときは UssCustomStyleReader が LogError を出す。間隔は前の値のまま
            // When unreadable, UssCustomStyleReader logs an error and the previous gap stays
            if (!UssCustomStyleReader.TryReadPixels(customStyle, GapProperty, out var pixels)) return;
            _gapPixels = pixels;
            _gapResolved = true;
            ApplyLeadingSpaces();
        }

        private void ApplyLeadingSpaces()
        {
            // silent-ok: スタイル解決の前は間隔が未定 / パネルに載ると CustomStyleResolvedEvent が必ず来て置き直す（L1 UiLayoutTest）
            if (!_gapResolved) return;

            // 表示中の先頭は USS の余白に任せ、それ以降に gap を当てる
            // The first shown child keeps its USS margin; later shown children get the gap
            var foundFirstShown = false;
            foreach (var item in _items)
            {
                if (item.style.display == DisplayStyle.None) continue;
                SetLeadingSpace(item, foundFirstShown ? new StyleLength(_gapPixels) : new StyleLength(StyleKeyword.Null));
                foundFirstShown = true;
            }
        }
    }
}
```

```csharp
// moorestech_client/Assets/Scripts/Client.Tests/Ui/Text/MultiLineLeadingTest.cs
using Client.Tests.Ui.TestHost;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Tests.Ui.Text
{
    /// <summary>
    /// 2 行の文の行送りがリッチテキストの line-height で行高トークンと 1px 以内になることを確かめる（設計書 A-48）
    /// Verifies rich-text line-height brings a two-line pitch within 1px of the line-height token (design A-48)
    /// </summary>
    public class MultiLineLeadingTest : UiTestHost
    {
        // Web のチャレンジ HUD の目標（--challenge-hud-objective-font-size 14px・--challenge-hud-objective-line-height 20px）
        // The Web challenge HUD objective (font 14px, line-height token 20px)
        private const float FontSize = 14f;
        private const float TargetLineHeight = 20f;

        [Test]
        public void 改行した2行の行送りが行高と1px以内になる()
        {
            var tag = $"<line-height={TargetLineHeight}px>";
            var plainPitch = Measure("Objective\nObjective", null) - Measure("Objective", null);
            var taggedPitch = Measure(tag + "Objective\nObjective", null) - Measure(tag + "Objective", null);

            Debug.Log($"[MultiLineLeading] newline: default pitch {plainPitch:F2}px, tagged pitch {taggedPitch:F2}px");
            Assert.AreEqual(TargetLineHeight, taggedPitch, 1f);
        }

        [Test]
        public void 折り返した2行の行送りが行高と1px以内になる()
        {
            const string sentence = "Objective Objective";
            var tag = $"<line-height={TargetLineHeight}px>";
            var oneLineWidth = MeasureWidth(sentence);
            var wrapWidth = oneLineWidth * 0.6f;
            var plainPitch = Measure(sentence, wrapWidth) - Measure("Objective", null);
            var taggedPitch = Measure(tag + sentence, wrapWidth) - Measure(tag + "Objective", null);

            Debug.Log($"[MultiLineLeading] wrap: default pitch {plainPitch:F2}px, tagged pitch {taggedPitch:F2}px");
            Assert.AreEqual(TargetLineHeight, taggedPitch, 1f);
        }

        private float Measure(string text, float? wrapWidth)
        {
            var label = Mount(text);
            if (wrapWidth.HasValue)
            {
                label.style.width = wrapWidth.Value;
                label.style.whiteSpace = WhiteSpace.Normal;
            }
            simulate.FrameUpdate();
            var height = label.layout.height;
            label.RemoveFromHierarchy();
            return height;
        }

        private float MeasureWidth(string text)
        {
            var label = Mount(text);
            simulate.FrameUpdate();
            var width = label.layout.width;
            label.RemoveFromHierarchy();
            return width;
        }

        private Label Mount(string text)
        {
            // 内容の寸法で測るため絶対配置にし、Typography の min-height を付けない
            // Absolute so the label sizes to its content; no Typography class, so no min-height
            var label = new Label(text) { enableRichText = true };
            label.style.position = Position.Absolute;
            label.style.fontSize = FontSize;
            label.style.whiteSpace = WhiteSpace.NoWrap;
            rootVisualElement.Add(label);
            return label;
        }
    }
}
```

```csharp
// moorestech_client/Assets/Scripts/Client.Ui/Components/Gauges/GaugeBar/GaugeBar.cs
using Client.Ui.Foundation.Theme;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Gauges
{
    /// <summary>
    /// 0 から 1 の値だけを描く読み取り専用の横ゲージ（Web の shared/ui/GaugeBar）
    /// Read-only horizontal gauge that draws only a zero-to-one value (Web shared/ui/GaugeBar)
    /// </summary>
    public sealed class GaugeBar : VisualElement
    {
        private const string SlotSizeProperty = "--gauge-bar-slot-size";

        // Web の height: calc(var(--slot-size) / 6)。USS に calc が無いので C# で割る
        // Web height: calc(var(--slot-size) / 6); USS has no calc, so C# divides
        private const float SlotSizeToHeight = 1f / 6f;

        private readonly VisualElement _fill;

        public GaugeBar(string testid)
        {
            name = testid;
            pickingMode = PickingMode.Ignore;
            AddToClassList("gauge-bar");

            // 線は充填より先に置き、充填が線を覆う Web の重なりにそろえる
            // The outline goes before the fill so the fill covers it, matching the Web layering
            var outline = new VisualElement { pickingMode = PickingMode.Ignore };
            outline.AddToClassList("gauge-bar__outline");
            Add(outline);
            _fill = new VisualElement { pickingMode = PickingMode.Ignore };
            _fill.AddToClassList("gauge-bar__fill");
            Add(_fill);

            SetValue(0f);
            RegisterCallback<CustomStyleResolvedEvent>(_ => OnCustomStyleResolved());
        }

        public void SetValue(float ratio01)
        {
            // NaN は 0 として描く（Web の Number.isNaN の分岐）
            // NaN draws as 0 (the Web's Number.isNaN branch)
            var clamped = float.IsNaN(ratio01) ? 0f : Mathf.Clamp01(ratio01);
            _fill.style.width = Length.Percent(clamped * 100f);
        }

        private void OnCustomStyleResolved()
        {
            // 読めないときは UssCustomStyleReader が LogError を出し、高さは前の値のまま
            // When unreadable, UssCustomStyleReader logs an error and the height keeps its previous value
            if (!UssCustomStyleReader.TryReadPixels(customStyle, SlotSizeProperty, out var slotSize)) return;
            style.height = slotSize * SlotSizeToHeight;
        }
    }
}
```

```csharp
    /// Tooltip face and lines (the face of Web shared/ui/HoverTooltip and shared/tooltip/CursorTooltip); placement and visibility belong to the host
    /// </summary>
    public sealed class TooltipFace : VisualElement
    {
        private readonly string _lineName;

        public TooltipFace(string testid, TooltipFaceLook look)
        {
            name = testid;
            _lineName = $"{testid}-line";
            pickingMode = PickingMode.Ignore;
            AddToClassList("tooltip-face");
            AddToClassList(LookClass(look));
        }

        public void SetText(UiText text)
        {
            SetLines(new[] { text });
        }

        public void SetLines(IReadOnlyList<UiText> lines)
        {

            // One label per line; the Web CursorTooltip also tags each line with a testid
            Clear();
            foreach (var line in lines)
            {

                label.AddToClassList("tooltip-face__line");
                label.SetText(line);
                Add(label);
            }
        }

        {
            return look switch
            {
                TooltipFaceLook.Cursor => "tooltip-face--cursor",

                _ => throw new ArgumentOutOfRangeException(nameof(look), look, null),
```

```csharp
// moorestech_client/Assets/Scripts/Client.Ui/Components/Text/ShortcutHintLabel/ShortcutHintSegments.cs
using System;

namespace Client.Ui.Components.Text
{
    /// <summary>
    /// 解いた文を kbd の前と後ろへ割った結果（Web の LocalizedShortcutHint の markerIndex の分岐を種類で表す）
    /// A resolved sentence split into the parts before and after the kbd (the markerIndex branch of Web LocalizedShortcutHint, as a kind)
    /// </summary>
    public readonly struct ShortcutHintSegments
    {
        // Web の SHORTCUT_MARKER と引数名。翻訳文の {shortcut} にこの印を入れて位置を探す
        // Web SHORTCUT_MARKER and param name; this marker fills {shortcut} in the sentence to locate the kbd
        public const string Marker = "";
        public const string ParamName = "shortcut";

        public readonly ShortcutHintSplitKind Kind;
        public readonly string Before;
        public readonly string After;

        private ShortcutHintSegments(ShortcutHintSplitKind kind, string before, string after)
        {
            Kind = kind;
            Before = before;
            After = after;
        }

        public static ShortcutHintSegments Split(string resolved, ShortcutHintLayout layout)
        {
            // prefix は割らず kbd を先頭にする
            // Prefix never splits; the kbd goes first
            if (layout == ShortcutHintLayout.Prefix) return new ShortcutHintSegments(ShortcutHintSplitKind.KeyFirst, "", resolved);

            // inline は印の位置で割る。印が無いのは辞書の破損なので kbd を先頭へ退避し、その種類で返す
            // Inline splits at the marker; a missing marker means a broken dictionary, so fall back to key-first and say so
            var markerIndex = resolved.IndexOf(Marker, StringComparison.Ordinal);
            if (markerIndex < 0) return new ShortcutHintSegments(ShortcutHintSplitKind.KeyFirstBecauseMarkerMissing, "", resolved);
            return new ShortcutHintSegments(ShortcutHintSplitKind.Inline, resolved.Substring(0, markerIndex), resolved.Substring(markerIndex + Marker.Length));
        }
    }
}
```

```csharp
using Client.Ui.State.Text;

namespace Client.Ui.Foundation.Text

    /// <summary>
    /// 宿主がパネルにいる間 UiText を解いて流し、言語が変わったら解き直す（LocalizedLabel と同じ購読の、文字列を流す版。D-18）
    /// While the host is on a panel, resolves a UiText and emits it, re-resolving on language change (LocalizedLabel's subscription, emitting the string; D-18)
    /// </summary>
    /// <remarks>
    /// 解いた文を割るなど Label 1 つに入れない部品が使う。Components は Client.Localization を直接購読しない（設計書 §1.1）
    /// For components that do not put the text into a single Label (e.g. split it); Components never subscribe to Client.Localization (design §1.1)
    /// </remarks>
    public sealed class LocalizedTextBinding
    {
        private readonly VisualElement _host;
        private readonly Subject<string> _onResolved = new();
        private UiText _text;
        private IDisposable _languageSubscription;

        public IObservable<string> OnResolved => _onResolved;

        public LocalizedTextBinding(VisualElement host)
        {
            _host = host;
            host.RegisterCallback<AttachToPanelEvent>(_ => OnAttached());
            host.RegisterCallback<DetachFromPanelEvent>(_ => OnDetached());
        }

        public void SetText(UiText text)
        {
            _text = text;
            // 宿主が載った後に作られた束縛（C2-b の置き文字）は AttachToPanelEvent を受けないので、ここで購読を始める
            // A binding created after its host attached (C2-b placeholder) never sees AttachToPanelEvent, so subscribe here
            if (_host.panel == null) return;
            EnsureSubscribed();
            Resolve();
        }

        private void OnAttached()
        {
            // 文無しで載ったら空のまま出さず、開発者に知らせる
            // Report a host attached without a text instead of leaving it empty
            if (_text == null) Debug.LogError($"[LocalizedTextBinding] attached without a text: {_host.name}");
            EnsureSubscribed();
            Resolve();
        }

        private void EnsureSubscribed()
        {
            if (_languageSubscription != null) return;
            _languageSubscription = Localize.OnLanguageChanged.Subscribe(_ => Resolve());
        }

        private void OnDetached()
        {
            _languageSubscription?.Dispose();
            _languageSubscription = null;
        }

        private void Resolve()
        {
            // silent-ok: 文が来るまで流さない / 載った時点で無ければ OnAttached が LogError を出す
            if (_text == null) return;
            _onResolved.OnNext(_text.Resolve());
        }
    }
}
```

```csharp
using Client.Ui.Foundation.Text;

namespace Client.Ui.Components.Text

    /// <summary>
    /// 翻訳の語順を保ったまま、文中に KeyCap を 1 つ差し込む（Web の shared/i18n/LocalizedShortcutHint）
    /// Inserts one KeyCap into a sentence while keeping the translated word order (Web shared/i18n/LocalizedShortcutHint)
    /// </summary>
    /// <remarks>
    /// 文の解決と言語変更の解き直しは Foundation の LocalizedTextBinding が持ち、ここは解いた文を割って並べるだけ（D-18）
    /// Resolving and re-resolving on language change belong to Foundation's LocalizedTextBinding; this only splits and lays out the result (D-18)
    /// </remarks>
    public sealed class ShortcutHintLabel : VisualElement
    {
        private const string FollowingClass = "shortcut-hint-label__item--following";
        private readonly ShortcutHintLayout _layout;
        private readonly Label _before;
        private readonly KeyCap _keyCap;
        private readonly Label _after;
        private readonly LocalizedTextBinding _sentence;

        public ShortcutHintLabel(string testid, ShortcutHintLayout layout)
        {
            name = testid;
            _layout = layout;
            pickingMode = PickingMode.Ignore;
            AddToClassList("shortcut-hint-label");
            _before = CreateText();
            _keyCap = new KeyCap("");
            _after = CreateText();
            Add(_before);
            Add(_keyCap);
            Add(_after);
            _sentence = new LocalizedTextBinding(this);
            _sentence.OnResolved.Subscribe(ShowResolved);
        }

        public void SetHint(UiText sentence, string shortcutText)
        {
            // {shortcut} に印を入れた文を渡す。印を先頭に置くので、呼び出し側が shortcut を渡しても印が勝つ
            // Hand over the sentence with the marker in {shortcut}; the marker goes first, so it wins over a caller-supplied shortcut
            _keyCap.SetKeyText(shortcutText);
            _sentence.SetText(sentence.WithLeadingParam(ShortcutHintSegments.ParamName, ShortcutHintSegments.Marker));
        }

        private void ShowResolved(string resolved)
        {
            // 印の位置で前後に割る。訳の欠けは Localize が [!key] の可視文字列で返すので空の分岐は持たない（D-20）
            // Split at the marker; a missing translation comes back as a visible [!key], so there is no empty branch (D-20)
            var segments = ShortcutHintSegments.Split(resolved, _layout);
            if (segments.Kind == ShortcutHintSplitKind.KeyFirstBecauseMarkerMissing) Debug.LogError($"[ShortcutHintLabel] {name}: the sentence has no shortcut marker; showing the key first");
            ShowText(_before, segments.Before);
            ShowText(_after, segments.After);
            MarkFollowingItems();

            #region Internal

            void ShowText(Label label, string value)
            {
                label.text = value;
                label.style.display = value.Length == 0 ? DisplayStyle.None : DisplayStyle.Flex;
            }

            void MarkFollowingItems()
            {
                // 表示中の先頭以外に間隔のクラスを付ける（Web は置く側の flex の gap）
                // Mark every shown item except the first for the gap (the Web uses the host's flex gap)
                var foundFirstShown = false;
                foreach (var item in Children())
                {
                    if (item.style.display == DisplayStyle.None) continue;
                    item.EnableInClassList(FollowingClass, foundFirstShown);
                    foundFirstShown = true;
                }
            }

            #endregion
        }

        private static Label CreateText()
        {
            var label = new Label { pickingMode = PickingMode.Ignore };
            label.AddToClassList("shortcut-hint-label__text");
            return label;
        }
    }
```

```csharp
using System;
using Client.Ui.Foundation.Theme;
using UniRx;
using UnityEngine;

namespace Client.Ui.Components.Controls
{
    /// <summary>
    /// 本文いっぱいに伸びる縦スクロール領域。バーは Mantine ScrollArea と同じく本文に重ね、溝とつまみは panelBodyScrollArea の色
    /// Vertical scroll area that fills the body; the bar overlays the content like Mantine ScrollArea with panelBodyScrollArea colors
    /// </summary>
    public sealed class GameScrollView : ScrollView
    {
        private const string ScrollbarSizeProperty = "--game-scroll-view-scrollbar-size";

        // Mantine ScrollArea のバーは padding = size / 5 で、つまみは size − 2 × padding
        // Mantine ScrollArea pads the bar by size / 5, so the thumb is size - 2 x padding
        private const float ScrollbarPaddingRatio = 0.2f;

        private readonly Subject<float> _onScrollOffsetChanged = new();

        public IObservable<float> OnScrollOffsetChanged => _onScrollOffsetChanged;

        public GameScrollView(string testid) : base(ScrollViewMode.Vertical)
        {
            name = testid;
            AddToClassList("game-scroll-view");
            horizontalScrollerVisibility = ScrollerVisibility.Hidden;
            verticalScrollerVisibility = ScrollerVisibility.Auto;

            // スクロール量の変化はバーの値の変化で拾う（ホイール・ドラッグ・ScrollTo・SetScrollOffset のどれでも来る）
            // Offset changes come from the bar value (wheel, drag, ScrollTo and SetScrollOffset all move it)
            verticalScroller.valueChanged += offset => _onScrollOffsetChanged.OnNext(offset);
            RegisterCallback<CustomStyleResolvedEvent>(_ => OnCustomStyleResolved());
        }

        public void SetScrollOffset(float y)
        {
            scrollOffset = new Vector2(scrollOffset.x, y);
        }

        private void OnCustomStyleResolved()
        {
            // 読めないときは UssCustomStyleReader が LogError を出し、バーは Theme.uss の既定の幅のまま
            // When unreadable, UssCustomStyleReader logs an error and the bar keeps the Theme.uss default width
            if (!UssCustomStyleReader.TryReadPixels(customStyle, ScrollbarSizeProperty, out var size)) return;
            var padding = size * ScrollbarPaddingRatio;
            verticalScroller.style.width = size;
            verticalScroller.style.paddingLeft = padding;
            verticalScroller.style.paddingRight = padding;
            verticalScroller.style.paddingTop = padding;
            verticalScroller.style.paddingBottom = padding;
        }
```

```csharp
    /// The single rule that derives the selected preset from graphics settings; shared by the UITK view model and the Web topic
    /// </summary>

                GraphicsQualityPreset.Low => GraphicsQualityPresetChoice.Low,
                _ => throw new System.ArgumentOutOfRangeException(nameof(preset), preset, null),
            };
        }
    }
```
