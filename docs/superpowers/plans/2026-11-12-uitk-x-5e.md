```csharp
using Mooresmaster.Localization.Generated;
```

```python
# .github/scripts/uitk_no_web/allowlist.py
"""Allowlist of the UITK no-Web gate (G4): path entries and line entries, each with a reason."""

import re
from dataclasses import dataclass
from fnmatch import fnmatchcase


@dataclass(frozen=True)
class PathEntry:
    glob: str
    reason: str
    line_no: int


@dataclass(frozen=True)
class LineEntry:
    glob: str
    pattern: re.Pattern
    reason: str
    line_no: int


@dataclass(frozen=True)
class Allowlist:
    paths: tuple
    lines: tuple


def parse_allowlist(text: str) -> tuple:
    """Return (Allowlist, errors). Each entry is `path <glob>  # 理由` or `line <glob> <regex>  # 理由`."""
    paths, lines, errors = [], [], []
    for line_no, raw in enumerate(text.splitlines(), 1):
        stripped = raw.strip()
        if not stripped or stripped.startswith("#"):
            continue
        # 理由の無い行は誤り。黙って許すと許可の根拠が残らない
        # An entry without a reason is an error; silently accepting it leaves no rationale
        body, separator, reason = stripped.partition(" # ")
        if not separator or not reason.strip():
            errors.append(f"{line_no}: 理由（' # 理由'）が無い")
            continue
        kind, _, rest = body.strip().partition(" ")
        rest = rest.strip()
        if kind == "path" and rest and " " not in rest:
            paths.append(PathEntry(rest, reason.strip(), line_no))
        elif kind == "line":
            glob, _, regex = rest.partition(" ")
            if not glob or not regex.strip():
                errors.append(f"{line_no}: line は '<glob> <regex>' の 2 つが要る")
                continue
            lines.append(LineEntry(glob, re.compile(regex.strip()), reason.strip(), line_no))
        else:
            errors.append(f"{line_no}: 種類は 'path <glob>' か 'line <glob> <regex>' だけ")
    return Allowlist(tuple(paths), tuple(lines)), errors


def path_allowed(entry: PathEntry, path: str) -> bool:
    return fnmatchcase(path, entry.glob)


def line_allowed(entry: LineEntry, path: str, text: str) -> bool:
    return fnmatchcase(path, entry.glob) and entry.pattern.search(text) is not None
```

```python
# .github/scripts/uitk_no_web/scan.py
"""Find G4 tokens in tracked file names and tracked text contents."""

import re
import subprocess
import sys
from dataclasses import dataclass
from pathlib import Path

# git grep で粗く拾い、Python で CEF の語の境界を見る（BSD の正規表現に \b が無く、和文の隣の CEF も拾うため）
# git grep catches candidates; Python checks CEF word edges (BSD regex lacks \b, and CEF next to Japanese must still hit)
GREP_PATTERN = "WebUi|webui|Cef|CEF|Kestrel|moorestech_web"
TOKEN_PATTERN = re.compile(r"WebUi|webui|Cef|(?<![A-Za-z0-9_])CEF(?![A-Za-z0-9_])|Kestrel|moorestech_web")


@dataclass(frozen=True)
class Hit:
    path: str
    line_no: int
    token: str
    text: str


def tracked_files(root: Path) -> list:
    result = _git(root, ["ls-files", "-z"], (0,))
    return [path for path in result.decode("utf-8", errors="replace").split("\0") if path]


def path_hits(files: list) -> list:
    hits = []
    for path in files:
        match = TOKEN_PATTERN.search(path)
        if match is not None:
            hits.append(Hit(path, 0, match.group(0), path))
    return hits


def content_hits(root: Path) -> list:
    # -I でバイナリを飛ばす。終了コード 1 は「当たり無し」
    # -I skips binaries; exit code 1 means "no match"
    output = _git(root, ["grep", "-I", "-n", "-z", "-E", GREP_PATTERN, "--", "."], (0, 1))
    hits = []
    for record in output.decode("utf-8", errors="replace").splitlines():
        parts = record.split("\0", 2)
        if len(parts) != 3:
            continue
        path, line_no, text = parts
        match = TOKEN_PATTERN.search(text)
        # 英数字に挟まれた CEF（素材の 16 進など）は語ではないので落とす
        # A CEF wedged between ASCII word characters (hex in assets etc.) is not the word, so it is dropped
        if match is None:
            continue
        hits.append(Hit(path, int(line_no), match.group(0), text.strip()))
    return hits


def _git(root: Path, args: list, ok_codes: tuple) -> bytes:
    completed = subprocess.run(["git", "-C", str(root), *args], capture_output=True)
    if completed.returncode not in ok_codes:
        print(f"G4: git {' '.join(args[:2])} が失敗: {completed.stderr.decode('utf-8', errors='replace').strip()}", file=sys.stderr)
        sys.exit(2)
    return completed.stdout
```

```python
# .github/scripts/uitk_no_web/cli.py
"""Command line entry of the UITK no-Web gate (G4)."""

import argparse
import sys
from collections import Counter
from pathlib import Path

from .allowlist import line_allowed, parse_allowlist, path_allowed
from .scan import content_hits, path_hits, tracked_files

DEFAULT_ALLOWLIST = ".github/scripts/uitk_no_web/allowlist.txt"


def main(argv: list) -> int:
    parser = argparse.ArgumentParser(description="UITK no-Web gate (G4)")
    parser.add_argument("--root", default=".")
    parser.add_argument("--allowlist", default=DEFAULT_ALLOWLIST)
    parser.add_argument("--report", action="store_true")
    args = parser.parse_args(argv)
    root = Path(args.root).resolve()

    # 許可リストは必須。読めなければ何も許さず止める
    # The allowlist is required; if it cannot be read, nothing is allowed and the gate stops
    allowlist_path = root / args.allowlist
    if not allowlist_path.is_file():
        print(f"G4: 許可リスト {args.allowlist} が無い", file=sys.stderr)
        return 2
    allowlist, errors = parse_allowlist(allowlist_path.read_text(encoding="utf-8"))
    if errors:
        for error in errors:
            print(f"G4: 許可リストを読めない: {error}", file=sys.stderr)
        return 2

    # 語の当たりを集め、許可リストで落とす。使われた行を数えて腐った行を出す
    # Collect hits, drop allowed ones, and count which entries were used to surface stale ones
    files = tracked_files(root)
    hits = path_hits(files) + content_hits(root)
    used = set()
    remaining = []
    for hit in hits:
        entry = _allowing_entry(allowlist, hit)
        if entry is None:
            remaining.append(hit)
        else:
            used.add(entry.line_no)
    stale = [entry for entry in (*allowlist.paths, *allowlist.lines) if entry.line_no not in used]

    if args.report:
        _print_report(remaining, stale)
        return 0
    if stale:
        for entry in stale:
            print(f"G4: 何も許していない許可リストの行 {entry.line_no}: {entry.glob}（{entry.reason}）", file=sys.stderr)
        return 2
    if remaining:
        for hit in remaining:
            print(f"G4: {hit.path}:{hit.line_no}: {hit.token}: {hit.text[:160]}")
        print(f"G4: 許可リスト外に {len(remaining)} 件")
        return 1
    print("G4: 許可リスト外 0 件")
    return 0


def _allowing_entry(allowlist, hit):
    for entry in allowlist.paths:
        if path_allowed(entry, hit.path):
            return entry
    if hit.line_no == 0:
        return None
    for entry in allowlist.lines:
        if line_allowed(entry, hit.path, hit.text):
            return entry
    return None


def _print_report(remaining: list, stale: list) -> None:
    # 置き場の上 3 段ごとの件数。残りの作業量の見積もりに使う
    # Counts per top three path segments, used to size the remaining work
    counts = Counter("/".join(hit.path.split("/")[:3]) for hit in remaining)
    for area, count in counts.most_common():
        print(f"{count}\t{area}")
    print(f"G4(report): 許可リスト外 {len(remaining)} 件・何も許していない行 {len(stale)} 件")
```

```python
# .github/scripts/tests/test_uitk_no_web.py
"""Unit tests of the UITK no-Web gate (G4) on throwaway git repositories."""

import contextlib
import io
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from uitk_no_web.cli import main  # noqa: E402

ALLOWLIST = ".github/scripts/uitk_no_web/allowlist.txt"


class UitkNoWebGateTest(unittest.TestCase):
    def setUp(self):
        self._dir = tempfile.TemporaryDirectory()
        self.root = Path(self._dir.name)
        subprocess.run(["git", "init", "-q", str(self.root)], check=True)

    def tearDown(self):
        self._dir.cleanup()

    def write(self, path: str, text: str) -> None:
        target = self.root / path
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(text, encoding="utf-8")
        subprocess.run(["git", "-C", str(self.root), "add", path], check=True)

    def run_gate(self, *extra: str) -> tuple:
        out, err = io.StringIO(), io.StringIO()
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            code = main(["--root", str(self.root), *extra])
        return code, out.getvalue() + err.getvalue()

    def test_empty_repo_with_empty_allowlist_passes(self):
        self.write(ALLOWLIST, "# empty\n")
        self.assertEqual(0, self.run_gate()[0])

    def test_missing_allowlist_stops(self):
        self.write("a.cs", "class A {}\n")
        self.assertEqual(2, self.run_gate()[0])

    def test_token_outside_allowlist_fails_with_location(self):
        self.write(ALLOWLIST, "# empty\n")
        self.write("src/A.cs", "// WebUi の名残\n")
        code, output = self.run_gate()
        self.assertEqual(1, code)
        self.assertIn("src/A.cs:1: WebUi", output)

    def test_path_entry_allows_and_is_used(self):
        self.write(ALLOWLIST, "path docs/adr/* # 経緯\n")
        self.write("docs/adr/0001.md", "CefUnity を使う\n")
        self.assertEqual(0, self.run_gate()[0])

    def test_line_entry_allows_only_matching_line(self):
        self.write(ALLOWLIST, "line skill.md git\\ show # 撤去前の読み方\n")
        self.write("skill.md", "git show abc:moorestech_web/webui/x\nmoorestech_web の残り\n")
        code, output = self.run_gate()
        self.assertEqual(1, code)
        self.assertIn("skill.md:2:", output)

    def test_cef_inside_identifier_is_not_a_token(self):
        self.write(ALLOWLIST, "# empty\n")
        self.write("Mat.mat", "_Color: Vector1_5CEF3A\n")
        self.assertEqual(0, self.run_gate()[0])

    def test_cef_next_to_japanese_is_a_token(self):
```

```markdown
# 見た目のホワイトリスト（UITK）

撤去前の Web 版で守っていた見た目の約束を、UITK の部品・トークン・層の名前へ写したもの。既存の画面はこの約束で作られている。新しい面・部品を足すときは、ここに無い表現・部品・パターンを使わない。新しい表現が要るなら、作る前にここへ書いて裁定を取る。

- stage 族: 1280×720 の舞台で一様に拡縮する。パネル・グリッド・詳細。`UiFamily.Stage`
- viewport 族: 実画面の辺に付き、中身の寸法だけが舞台の倍率に従う。常時 HUD（ホットバー・装備 HUD・キーヒント・採掘の進捗・目標 HUD・操作モード HUD）。`UiFamily.Viewport`。舞台の絶対配置を式で補正しない

- 背面の viewport 族（舞台の裏）は通知だけ（ADR 0017）

- 中身の量で寸法が決まるパネル（チェスト等）は、余白がフェード幅より狭い辺を安全帯のトークン（`--block-panel-right-safe-area`・`--block-panel-bottom-safe-area`）で補う

- 題の行の右端の副次アクションは `GamePanel.SetTitleAction` の置き場だけ。本文の先頭や末尾へ置かない
- Mantine の枠で描いていた画面（ポーズ・設定・バグ報告・トースト・モーダル・チャレンジ）は `DialogFrame` 等の写した枠を使う。面を `GamePanel` に統一しない
- モーダルは中央に置き、幕は `ModalScreen` の 1 枚だけ（画面の幕と二重にしない）。確認・入力の定型モーダルは `ModalScreen` を通す

- 状態は USS のクラス（`--selected`・`--filled`・`--catalog`・`--insufficient`）。足すときもクラス
- 用途の違うスロット群を同じパネルに並べるときは、ラベル（`--text-muted`）か `FadeRule` で必ず分ける
- 左右の数が違いうる行の中央の要素（進捗の矢印等）は、左右を同じ幅に取って中央に固定する
- アイコンに重なる文字は文字色の反対色の縁で浮かせる（`OutlinedCountLabel`。ADR 0033）。ぼかしの影は使わない
- 素材の「所持/必要」は `ItemSlot` の不足の表示だけが描く。画面に数字の要素を複製しない

- パレットは半透明の紺（#0a0e1b／#070912 系、α 0.8）と寒色の灰

## 6. 装飾
- 装飾を画像にしない。枠・罫線・グリップ等は USS と描画（`generateVisualContent`）で描く
- 装飾の語彙は 6 つだけ: ① 両端がフェードする水平の罫線（`FadeRule`・題の上下）② 下向き三角の底面（default の枠の下部）③ 右下の三角のグリップ（craft の枠）④ 両端の菱形（スキットの選択肢だけ）⑤ シアンの下向きシェブロン＝送り待ちの印（スキットの会話窓だけ・光彩なし）⑥ 黄黒の斜線の警告帯（削除モードの画面の上下端だけ）
```

```python
#!/usr/bin/env python3
"""Rewrite `web: moorestech_web/webui/...` and `web: W/...` source comments to `web: web@<sha10>/...` on comment lines only."""
import re
import subprocess
import sys
from pathlib import Path

sha = sys.argv[1]
root = Path(sys.argv[2])
files = subprocess.run(["git", "-C", str(root), "ls-files", "moorestech_client/Assets/Scripts", ".agents/skills/ref-uitk-design"],
                       capture_output=True, text=True, check=True).stdout.split()
targets = [f for f in files if f.endswith((".cs", ".uss", ".tss", ".uxml", ".md"))]
comment = re.compile(r"^\s*(//|/\*|\*|<!--|-|\|)")
changed_lines = 0
for rel in targets:
    path = root / rel
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    out = []
    for line in lines:
        new = line
        if comment.match(line) or "/*" in line or "//" in line:
            new = new.replace("moorestech_web/webui/", f"web@{sha}/")
            new = re.sub(r"(?<![A-Za-z0-9_@/])W/", f"web@{sha}/src/", new)
        if new != line:
            changed_lines += 1
        out.append(new)
    path.write_text("".join(out), encoding="utf-8")
print(f"changed_lines={changed_lines}")
```
