// The delete bands' layer (mode HUD, OverlayPanel) precedes the hotbar's (Chrome) under the root, so the hotbar draws over them
public static void AssertDeleteBandsBeforeHotbar(VisualElement root)
{
    var hotbar = root.Q<VisualElement>(name: "hotbar-hud");
    var bands = root.Q<VisualElement>(name: Stage2HudSurfaces.DeleteModeWarning.Name);
    Assert.IsNotNull(hotbar, "hotbar-hud が木に無い");
    Assert.IsNotNull(bands, "削除の帯が木に無い");
    var hotbarTop = TopUnderRoot(root, hotbar);
    var bandsTop = TopUnderRoot(root, bands);
    Assert.Less(root.IndexOf(bandsTop), root.IndexOf(hotbarTop), "削除の帯の層がホットバーの層より後ろにある（帯がホットバーの上に描かれる）");

    #region Internal
    VisualElement TopUnderRoot(VisualElement r, VisualElement e)
    {
        while (e.parent != r) e = e.parent;
        return e;
    }
    #endregion
}
