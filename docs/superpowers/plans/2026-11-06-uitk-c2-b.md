# UITK C2-b（段階 2 部品: 操作・入力 ― IconButton・FormButton・GameTextField・albof.12）Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** 操作・入力の共有部品 `IconButton`・`FormButton`・`GameTextField`（見た目 2 種と置き文字）を `Client.Ui/Components/Controls/` に Web 版と同じ見た目で作り、バグ報告画面の USS 直書き（Mantine Button 相当・textAreaField 相当）をこれらへ置き換え、albof.12（TextField の IME・ESC 要件 8 項目）を Mac の合成キーで現物照合して閉じる。

**Architecture:** 部品レーン（単位エージェント・Editor 要）。部品は 1 部品 1 ディレクトリ・1 USS で、USS は F1 が作った `GameUiTheme.tss` の操作部品の区画に `@import` を足して束ねる（設計書 C1・[A-2]）。押下は `IObservable<Unit> OnClicked`（C8）、文言は `UiText`／`LocalizationKey`（C9）、色は F1 が生成する `MantineTokens.uss`（Web と同名の `--mantine-*`）と既存の Web トークンだけを使う。見た目の正本は Web の CSS（`textAreaField.module.css`・`IconButton/style.module.css`）と Mantine 8.3.18 の `Button.css`・`Input.css`（dark）で、数値はこの計画に逐語で写した。L1 は部品ごとに状態・イベント・寸法（つまみ）を確かめ、バグ報告の L1 3 クラスを緑に保ち、Editor の部品ギャラリーと参照キャプチャの切り抜きで並べ比べ、`uitk-lifeline` の録画と Mac の OS 段の合成キーによる IME 1 回で締める。最後は moores-code-review → `base=feature/uitk` の PR → agent がマージコミットで取り込む。

**Tech Stack:** Unity 6000.3.8f1 UI Toolkit（ランタイムパネル・USS・Painter2D）、UniRx、NUnit＋`com.unity.ui.test-framework`（`UITestFixture`・`PanelSimulator`）、uloop CLI、プレイテスト DSL（`run-scenario.sh`）、osascript（System Events の `key code`）、gh CLI、bd（Beads）

```csharp
using System;
using UniRx;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Controls
{
    /// <summary>
    /// 面を持たない小さいアイコンボタン（Web の shared/ui/IconButton）。中の図形は呼び出し側が渡し、閉じるには IconButtonCloseGlyph を渡す
    /// Small faceless icon button (Web shared/ui/IconButton); callers pass the glyph, and IconButtonCloseGlyph for a close button
    /// </summary>
    /// <remarks>
    /// 拡張点: 閉じる以外の図形は画面が VisualElement で作って渡す（Web が children に SVG を渡すのと同じ）。図形は CustomStyleResolvedEvent で GlyphColorProperty と GlyphStrokeProperty を読んで描く
    /// Extension: screens build non-close glyphs as VisualElements and pass them in (as the Web passes an SVG child); a glyph reads GlyphColorProperty and GlyphStrokeProperty in CustomStyleResolvedEvent and draws with them
    /// </remarks>
    public sealed class IconButton : Button
    {
        public const string UssClassName = "icon-button";
        public const string GlyphUssClassName = "icon-button__glyph";
        public const string GlyphColorProperty = "--icon-button-glyph-color";
        public const string GlyphStrokeProperty = "--icon-button-glyph-stroke";

        private readonly Subject<Unit> _clicked = new();

        public IObservable<Unit> OnClicked => _clicked;

        public IconButton(string testid, VisualElement glyph)
        {
            name = testid;
            AddToClassList(UssClassName);

            // GameUi は入力欄以外にフォーカスを持たせない
            // GameUi never lets anything but text fields hold focus
            focusable = false;

            // 図形は押下を横取りしないようポインタを受けず、色と線幅は図形のクラスで受け取る
            // The glyph takes no pointer so it never steals the press, and gets colour and stroke via its class
            glyph.pickingMode = PickingMode.Ignore;
            glyph.AddToClassList(GlyphUssClassName);
            Add(glyph);

            // 押下は UniRx で外へ出す
            // Presses leave through UniRx
            clicked += () => _clicked.OnNext(Unit.Default);
        }
    }
}
```

```csharp
    /// IconButton の既定の ×。Web の viewBox 0 0 16 16 の 2 本線（M3 3L13 13M13 3L3 13・stroke-linecap square）を Painter2D で描く
    /// The IconButton default cross: Web's two strokes in a 16x16 viewBox (M3 3L13 13M13 3L3 13, square caps) drawn with Painter2D
    /// </summary>
    public sealed class IconButtonCloseGlyph : VisualElement
    {
        public const string UssClassName = "icon-button__close-glyph";

        // Web の CloseIcon の viewBox と線の端点（2 本ぶん）
        // Web CloseIcon viewBox and the stroke endpoints (two strokes)
        private const float ViewBoxSize = 16f;
        private static readonly Vector2[] StrokeEnds = { new(3f, 3f), new(13f, 13f), new(13f, 3f), new(3f, 13f) };

        private Color _strokeColor;
        private float _strokeWidth;
        private bool _hasStyle;

        public IconButtonCloseGlyph()
        {
            AddToClassList(UssClassName);
            pickingMode = PickingMode.Ignore;
            generateVisualContent += DrawCross;
            RegisterCallback<CustomStyleResolvedEvent>(_ => OnCustomStyleResolved());
        }

        private void OnCustomStyleResolved()
        {
            // ホバーで色が変わるたびに解決し直されるので、読み直して描き直す
            // Hover re-resolves the colour each time, so re-read and redraw
            var hasColor = UssCustomStyleReader.TryReadColor(customStyle, IconButton.GlyphColorProperty, out _strokeColor);
            var hasWidth = UssCustomStyleReader.TryReadPixels(customStyle, IconButton.GlyphStrokeProperty, out _strokeWidth);
            _hasStyle = hasColor && hasWidth;
            MarkDirtyRepaint();
        }

        private void DrawCross(MeshGenerationContext context)
        {
            // silent-ok: 色と線幅の解決前は描かない / 解決できなければ UssCustomStyleReader が LogError を出す
            // silent-ok: draw nothing until colour and width resolve / UssCustomStyleReader logs an error if they never do
            if (!_hasStyle) return;

            // viewBox の 16 を要素の幅へ拡大し、SVG と同じく線幅も同じ倍率で太らせる
            // Scale the 16-unit viewBox to the element width, and the stroke width with it as SVG does
            var scale = contentRect.width / ViewBoxSize;
            var painter = context.painter2D;
            painter.strokeColor = _strokeColor;
            painter.lineWidth = _strokeWidth * scale;
            painter.lineCap = LineCap.Butt;
            painter.BeginPath();
            for (var i = 0; i < StrokeEnds.Length; i += 2) AddSquareCappedSegment(painter, StrokeEnds[i] * scale, StrokeEnds[i + 1] * scale, painter.lineWidth / 2f);
            painter.Stroke();

            #region Internal

            // Painter2D に square の端が無いので、両端を線幅の半分だけ線の向きへ延ばして同じ形にする
            // Painter2D has no square cap, so extend both ends by half the stroke width along the line to match
            void AddSquareCappedSegment(Painter2D target, Vector2 from, Vector2 to, float halfWidth)
            {
                var direction = (to - from).normalized;
                target.MoveTo(from - direction * halfWidth);
                target.LineTo(to + direction * halfWidth);
            }

            #endregion
        }
    }
}
```

```csharp
namespace Client.Ui.Components.Controls
{
    /// <summary>
    /// 送信・OK・キャンセル用の押しボタン（Mantine Button の写し）。文言・色の系統・大きさを Set で受け、押下を OnClicked で出す
    /// Push button for send / OK / cancel (copy of Mantine Button); takes label, colour family and size via Set and emits presses via OnClicked
    /// </summary>
    /// <remarks>
    /// 拡張点: Web で新しい variant・size を使い始めたら、列挙とクラスを 1 組ずつ足す。外側の余白・幅（fullWidth）は置く側の画面 USS が付ける
    /// Extension: when the Web starts using a new variant or size, add one enum value and class pair; outer margin and width (fullWidth) belong to the placing screen's USS
    /// </remarks>
    public sealed class FormButton : Button
    {
        public const string UssClassName = "form-button";
        public const string LabelUssClassName = "form-button__label";
        public const string FilledUssClassName = "form-button--filled";
        public const string DangerUssClassName = "form-button--danger";
        public const string SubtleUssClassName = "form-button--subtle";
        public const string ExtraLargeUssClassName = "form-button--xl";

        private readonly Subject<Unit> _clicked = new();
        private readonly LocalizedLabel _label;

        public IObservable<Unit> OnClicked => _clicked;

        public FormButton(string testid, UiText label, FormButtonVariant variant)
        {
            name = testid;
            AddToClassList(UssClassName);

            // GameUi は入力欄以外にフォーカスを持たせない
            // GameUi never lets anything but text fields hold focus
            focusable = false;

            // 文言は言語変更で解決し直すラベルに持たせ、押下を横取りさせない
            // The label re-resolves on language change and never steals the press
            _label = new LocalizedLabel { pickingMode = PickingMode.Ignore };
            _label.AddToClassList(LabelUssClassName);
            Add(_label);
            SetLabel(label);
            SetVariant(variant);

            // 押下は UniRx で外へ出す
            // Presses leave through UniRx
            clicked += () => _clicked.OnNext(Unit.Default);
        }

        public void SetLabel(UiText label)
        {
            _label.SetText(label);
        }

        public void SetVariant(FormButtonVariant variant)
        {
            // 色の系統はいつも 1 つだけ付ける（Web の variant と color の組と同じ）
            // Exactly one colour family at a time, matching the Web variant/color pair
            EnableInClassList(FilledUssClassName, variant == FormButtonVariant.Filled);
            EnableInClassList(DangerUssClassName, variant == FormButtonVariant.Danger);
            EnableInClassList(SubtleUssClassName, variant == FormButtonVariant.Subtle);
        }

        public void SetSize(FormButtonSize size)
        {
            EnableInClassList(ExtraLargeUssClassName, size == FormButtonSize.ExtraLarge);
        }
    }
}
```

```csharp
using System;
using Client.Ui.Foundation.Text;
using Client.Ui.State.Text;
using UniRx;
using UnityEngine;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Controls
{
    /// <summary>
    /// GameUi 共通の入力欄。見た目 2 種・置き文字・フォーカスの輪と、全選択の抑止・IME の未確定文字列の下線・候補窓のキャレット追従を持つ TextField の包み
    /// Shared GameUi text field: wraps a TextField with two looks, a placeholder and a focus ring, plus no select-all, an IME composition underline and caret-following candidate window
    /// </summary>
    /// <remarks>
    /// 拡張点: 入力欄に共通する見た目・IME の振る舞いはここへ足す。寸法は置く側が外側の箱に付ける。文字数の検証や絞り込みなど画面固有の処理は OnValueChanged の購読側が持つ。ESC・フォーカス同期はルートの GameUiInputRules が受け持つ
    /// Extension: add text-field-wide visuals and IME behaviour here; callers size the outer box; screen-specific logic (length checks, filtering) belongs to OnValueChanged subscribers, and ESC / focus sync belong to the root GameUiInputRules
    /// </remarks>
    public sealed class GameTextField : VisualElement
    {
        public const string UssClassName = "game-text-field";
        public const string FocusedUssClassName = "game-text-field--focused";
        public const string MultilineUssClassName = "game-text-field--multiline";
        public const string FocusRingUssClassName = "game-text-field__focus-ring";

        private readonly Subject<string> _valueChanged = new();
        private readonly TextElement _textElement;
        private readonly ImeCompositionUnderline _underline;
        private readonly IVisualElementScheduledItem _focusedTick;
        private LocalizedTextBinding _placeholder;

        public UnityEngine.UIElements.TextField Field { get; }

        /// <summary>
        /// 値の変化。IME の変換中の未確定文字列も値として流れる。購読側（文字数の検証・即時の絞り込み）は未確定の文字列でも壊れないこと（§4.9）
        /// Value changes. Uncommitted IME composition text also flows as a value; subscribers (length checks, live filtering) must not break on it (§4.9)
        /// </summary>
        public IObservable<string> OnValueChanged => _valueChanged;

        public GameTextField(string testid, bool multiline, GameTextFieldLook look)
        {
            // 外側の箱はフォーカスもポインタも受けず、testid は DSL の SetText が引く内側の TextField に付ける
            // The outer box takes neither focus nor pointer; the testid goes on the inner TextField that the DSL's SetText looks up
            focusable = false;
            pickingMode = PickingMode.Ignore;
            AddToClassList(UssClassName);
            AddToClassList(LookUssClassName(look));
            EnableInClassList(MultilineUssClassName, multiline);

            // クリックでのフォーカスで本文を全選択しない（1 打で置き換わるのを防ぐ）
            // Do not select everything on click-focus, so one keystroke cannot replace the whole text
            Field = new UnityEngine.UIElements.TextField { name = testid, multiline = multiline };
            Field.selectAllOnFocus = false;
            Field.selectAllOnMouseUp = false;
            Field.RegisterValueChangedCallback(evt => _valueChanged.OnNext(evt.newValue));
            Add(Field);

            // Web の :focus-visible の outline を外側の箱の子で描く（出す・出さないは見た目のクラスと --focused で USS が決める）
            // Draw the Web :focus-visible outline as a child of the outer box (USS shows it from the look class and --focused)
            var focusRing = new VisualElement { pickingMode = PickingMode.Ignore };
            focusRing.AddToClassList(FocusRingUssClassName);
            Add(focusRing);

            // IME の状態はイベントが無いので、フォーカス中だけパネルのスケジューラで毎フレーム見る
            // IME state has no events, so watch it every frame on the panel scheduler only while focused
            _textElement = Field.Q<TextElement>();
            _underline = new ImeCompositionUnderline(Field, _textElement);
            _focusedTick = Field.schedule.Execute(OnFocusedTick).Every(0);
            _focusedTick.Pause();
            Field.RegisterCallback<FocusInEvent>(_ => OnFocusGained());
            Field.RegisterCallback<FocusOutEvent>(_ => OnFocusLost());

        }

        public void SetValueWithoutNotify(string value)
        {
            Field.SetValueWithoutNotify(value);
        }

        public void SetPlaceholder(UiText placeholder)
        {
            // 置き文字の束縛は初めて置き文字を持つときに作る（置き文字の無い欄は束縛を持たず、文無しの LogError も出ない）
            // Create the placeholder binding on first use (fields without a placeholder hold no binding and log nothing)
            if (_placeholder == null)
            {
                _placeholder = new LocalizedTextBinding(Field);
                _placeholder.OnResolved.Subscribe(text => Field.textEdition.placeholder = text);
            }
            _placeholder.SetText(placeholder);
        }

        // EditMode では未確定文字列を作れないので、L1 から直接呼べるよう下線の更新を分けている
        // Composition text cannot be produced in EditMode, so the underline update is split out for L1 to call directly
        internal void ApplyComposition(string compositionString)
        {
            _underline.Apply(compositionString);
        }

        private void OnFocusGained()
        {
            // フォーカスの状態は輪と枠の色のためにクラスで USS へ渡す
            // Hand focus to USS as a class for the ring and border colour
            AddToClassList(FocusedUssClassName);
            _focusedTick.Resume();
        }

        private void OnFocusedTick()
        {
            _underline.Refresh(UnityEngine.Input.compositionString);

            // キャレットは行の下端を指すので候補窓は行のすぐ下に出る。UITK 自身が書くパネル座標の値を、スクリーンのピクセルへ換算した値で上書きする
            // The caret points at the line bottom so the window sits just below the line; overwrite UITK's own panel-space value with the screen-pixel conversion
            var caretPanelPoint = _textElement.LocalToWorld(Field.textSelection.cursorPosition);
            var panelSize = panel.visualTree.layout.size;
            var screenSize = new Vector2(Screen.width, Screen.height);
            UnityEngine.Input.compositionCursorPos = ImeCaretScreenPosition.FromPanel(caretPanelPoint, panelSize, screenSize, ImeCaretScreenPosition.PlatformOrigin);
        }

        private void OnFocusLost()
        {
            // フォーカスが外れると変換は終わるので、見張りを止めて下線と輪を消す
            // Losing focus ends the composition, so stop watching and hide the underline and ring
            RemoveFromClassList(FocusedUssClassName);
            _focusedTick.Pause();
            _underline.Reset();
        }

        private static string LookUssClassName(GameTextFieldLook look)
        {
            return look switch
            {
                GameTextFieldLook.GaugeInset => "game-text-field--gauge-inset",
                GameTextFieldLook.MantineInput => "game-text-field--mantine-input",
                _ => throw new ArgumentOutOfRangeException(nameof(look), look, "unknown GameTextFieldLook"),
            };
        }
    }
}
```

```csharp
    /// Resend and export buttons rescuing unsent reports plus the export status line (Web BugReportRecoveryControls)
    /// </summary>
    public sealed class BugReportRecoveryControlsView : VisualElement
    {
        private readonly FormButton _resendButton;
        private readonly FormButton _exportButton;
        private readonly Label _exportStatus;

        public BugReportRecoveryControlsView(IPauseMenuCommands commands)
        {
            AddToClassList("bug-report__recovery");

            // 再送と書き出しは Web と同じ Mantine の filled ボタン
            // Resend and export are Mantine filled buttons, as on the Web
            _resendButton = CreateButton("bug-report-resend-failed", LocalizationKeys.Ui.BugReport.ResendFailed);
            _resendButton.OnClicked.Subscribe(_ => commands.ResendFailedReports());
            _exportButton = CreateButton("bug-report-export-unsent", LocalizationKeys.Ui.BugReport.ExportUnsent);
            _exportButton.OnClicked.Subscribe(_ => commands.ExportUnsentReports());
            _exportStatus = new Label { name = "bug-report-export-status" };
            _exportStatus.AddToClassList("bug-report__status");
            Add(_resendButton);
            Add(_exportButton);
            Add(_exportStatus);
        }

        public void Apply(PlaytestDeliveryStatus delivery, PlaytestExportStatus export)
        {
            // 書き出し中は二重に走らせないよう両方とも押せなくする
            // Both buttons are disabled while exporting so neither starts a second run
            var exporting = export is PlaytestExportRunning;
            ShowIf(_resendButton, BugReportFormText.ShowResend(delivery));
            ShowIf(_exportButton, BugReportFormText.ShowExport(delivery));
            _resendButton.SetEnabled(!exporting);
            _exportButton.SetEnabled(!exporting);

            var statusLine = BugReportFormText.DescribeExport(export);
            ShowIf(_exportStatus, statusLine != null);
            _exportStatus.text = statusLine ?? "";
        }

        private static FormButton CreateButton(string testid, LocalizationKey label)
        {
            var button = new FormButton(testid, new UiText(label), FormButtonVariant.Filled);
            button.AddToClassList("bug-report__button");
            return button;
        }

        private static void ShowIf(VisualElement element, bool visible)
        {
            element.style.display = visible ? DisplayStyle.Flex : DisplayStyle.None;
        }
```

```bash
#!/bin/bash
# uitk-c2b-ime-mac.cs の運転役: run-scenario.sh でシナリオを起動し、シナリオが書いた step-<n>.keys の key code を osascript で Unity へ送る
# Driver for uitk-c2b-ime-mac.cs: boots the scenario via run-scenario.sh and sends the key codes in step-<n>.keys to Unity via osascript
# osascript を Unity の中から呼ぶと Unity にアクセシビリティの許可が要るので、許可のある端末側のこのスクリプトから送る
# Calling osascript from inside Unity would need Accessibility permission for Unity, so keys are sent from this terminal-side script
#
# usage: uitk-c2b-ime-mac-keys.sh <unity-project-path> <scenario.cs>
set -u
PROJECT="$1"
SCENARIO="$2"
SKILL_DIR="$(cd "$(dirname "$0")/../.." && pwd)"
MARKERS="$PROJECT/PlaytestResults/scratch/ime-mac"
LOG="$PROJECT/PlaytestResults/scratch/ime-mac-runner.log"
rm -rf "$MARKERS"
mkdir -p "$MARKERS"

# 終わり方によらず入力ソースを英数へ戻す
# Always return the input source to eisu (ABC), however this ends
restore_input_source() { osascript -e 'tell application "System Events" to key code 102' >/dev/null 2>&1; }
trap restore_input_source EXIT

# シナリオは子プロセスで走らせ、このスクリプトが前景で鍵を送る（wait で終わりを待つ）
# Run the scenario as a child while this script sends keys in the foreground (wait collects it)
bash "$SKILL_DIR/scripts/run-scenario.sh" "$PROJECT" "$SCENARIO" >"$LOG" 2>&1 &
RUNNER=$!
step=1
deadline=$((SECONDS + 540))
while [ $SECONDS -lt $deadline ]; do
  [ -f "$MARKERS/finish" ] && break
  kill -0 "$RUNNER" 2>/dev/null || { echo "runner exited before finish"; break; }
  KEYS="$MARKERS/step-$step.keys"
  if [ -f "$KEYS" ]; then
    pid=$(head -1 "$KEYS")
    osascript -e "tell application \"System Events\" to set frontmost of (first process whose unix id is $pid) to true" >/dev/null
    outcome=sent
    for code in $(tail -n +2 "$KEYS"); do
      # 他のセッションが前面を取っていたら送らずに止める（別アプリへ打たない）
      # Stop without sending if another session took the front (never type into another app)
      front=$(osascript -e 'tell application "System Events" to unix id of first process whose frontmost is true')
      if [ "$front" != "$pid" ]; then outcome="aborted: frontmost pid $front is not unity $pid"; break; fi
      osascript -e "tell application \"System Events\" to key code $code"
      sleep 0.25
    done
    screencapture -x "$MARKERS/step-$step.png"
    echo "$outcome" >"$MARKERS/step-$step.done"
    echo "step $step: $outcome"
    step=$((step + 1))
  fi
  sleep 0.2
done
wait "$RUNNER"
status=$?
tail -40 "$LOG"
exit $status
```
