# UITK 段階 4 部品「スロット族 2: 液体」（C4-a） Implementation Plan

> **For the controller session (実装を担うsubagentはこのブロックを無視してよい):** このplanの実行は moores-subagent-driven-development スキルが担う。実行モード（規模ゲート未満の単一subagent実装モード／閾値超のタスクごと派遣）は同スキルの規模ゲートに従って決める。ステップはチェックボックス（`- [ ]`）記法で書く。

**Goal:** Web 版の液体のスロット族（`shared/ui/FluidSlot`・`FluidAmountSlot`・`FluidSlotRow`）を、Web と同じ見た目の UITK 部品として `Client.Ui/Components/Slots/Fluid/` に作り、液体の背面の色を Content 層（`GameContentPresentation.FluidFace`）に足して、L0・L1・部品ギャラリー・参照キャプチャとの並べ比べ（720p・2560×1080）まで済ませて `feature/uitk` へ取り込む。

**Architecture:** 部品はドメインを知らない `VisualElement` 派生で、液体の絵は `UiIcon`、量は `double`、背面の色は `Color?`（null＝色が無いのでフィルを描かない）で受ける。ID → `UiIcon`・色の解決は Content の `GameContentPresentation` 1 か所（設計書 A-4・A-53）。`FluidSlot` は Web と同じく SlotFrame を使わない暗い四角（Mantine の dark-4 の線・dark-8 の面）に、下から量の割合で液体の色を満たすフィル・絵・白字に黒縁の量を重ねる。`FluidAmountSlot` は C3-a の `SlotFrame`（白面）＋`SlotContent`（絵）に、白縁の黒字の量の帯を足す。量の縁はどちらも C3-a の `OutlinedCountLabel`（縁だけの Label を後ろに敷く 2 層）を使い、黒縁のために縁の色のつまみ `--outlined-count-label-stroke` を 1 つ足す。`FluidSlotRow` は C2-c の `UiGroup(UiGap.Xs)` に `FluidSlot` を並べ、C4-b の `ProgressArrowBar` を末尾に出し入れする。

**Tech Stack:** Unity 6000.3.8f1 UI Toolkit（ランタイムパネル・USS カスタムプロパティ）、UniRx、NUnit＋`com.unity.ui.test-framework`（`UiTestHost`）、uloop CLI、ffmpeg（参照の切り抜き）、gh CLI、bd（Beads）

- **パスの略記**: `S/` = `moorestech_client/Assets/Scripts/`、`W/` = `moorestech_web/webui/src/`。Files 節と各コマンドはリポジトリ相対の完全なパスで書く
- **正本**: 裁定 `.decisions/2026-11-06-UITK*.md`、統合計画 `docs/uitk-migration/plan.md`、設計書 `docs/uitk-migration/execution/component-architecture.md`（§2.1・§2.3・§3・§4.3・§7.1。この計画で A-95〜A-99 を足す。末尾の「設計書・実行計画へ反映する行」）、実行計画 `docs/uitk-migration/execution/execution-plan.md`（レーン C4-a の行・§4.3・§6）。食い違ったら 裁定 → 統合計画 → 設計書 → 実行計画 → この計画 の順で上が勝つ
- **ゴール（ユーザーの原文）**: 「WebUIと同じ見た目で、適切な設計、コンポーネント化されたUITKベースのUIを開発すること」

```csharp
using System;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// 量と容量からフィルの割合 0..1 を出す（Web の fillRatio と clamp01）
    /// Fill ratio 0..1 from amount and capacity (the Web fillRatio and clamp01)
    /// </summary>
    public static class FluidFillRatio
    {
        public static float Of(double amount, double capacity)
        {
            // 容量 0 以下は 0、NaN は 0、範囲外は 0..1 へ丸める
            // Capacity at or below 0 gives 0, NaN gives 0, and out-of-range values clamp to 0..1
            if (capacity <= 0) return 0f;
            var ratio = amount / capacity;
            if (double.IsNaN(ratio)) return 0f;
            return (float)Math.Clamp(ratio, 0d, 1d);
        }
    }
}
```

```csharp
using Client.Ui.Foundation.Text;
using Client.Ui.Foundation.Tooltip;
using Client.Ui.State.Text;
using UnityEngine.UIElements;

namespace Client.Ui.Components.Slots
{
    /// <summary>
    /// 液体 1 マス。背面のフィル・絵・量・名前のツールチップ（Web の FluidSlot）
    /// One fluid cell: back fill, picture, amount and name tooltip (the Web FluidSlot)
    /// </summary>
    public sealed class FluidSlot : VisualElement
    {
        private const string FillShownStateClass = "fluid-slot--fill-shown";
        private const string AmountShownStateClass = "fluid-slot--amount-shown";
        private const string GhostStateClass = "fluid-slot--ghost";

        private readonly VisualElement _fill;
        private readonly GameIcon _icon;
        private readonly OutlinedCountLabel _amount;

        public FluidSlot(string testid)
        {
            name = testid;
            AddToClassList("fluid-slot");

            // フィル → 絵 → 量の順に重ね、量を最前面に残す（Web の子の順）
            // Stack fill, picture, then amount so the amount stays frontmost (the Web child order)
            _fill = new VisualElement { pickingMode = PickingMode.Ignore };
            _fill.AddToClassList("fluid-slot__fill");
            Add(_fill);
            _icon = new GameIcon();
            _icon.AddToClassList("fluid-slot__icon");
            Add(_icon);

            // 量は白字に黒縁（Web の .iconTextOutlineDark）。縁の色は置き場のクラスがつまみで黒にする
            // The amount is white with a dark outline (the Web .iconTextOutlineDark); the placement class turns the outline knob black
            _amount = new OutlinedCountLabel(this, "fluid-slot__amount", UiTypographyClass.TextBold);
        }

        public void SetFluid(FluidSlotContent content)
        {
            switch (content.Kind)
            {
                case FluidSlotContentKind.Empty:
                    ShowEmpty();
                    break;
                case FluidSlotContentKind.Ghost:
                    ShowGhost();
                    break;
                case FluidSlotContentKind.Filled:
                    ShowFilled();
                    break;
            }

            #region Internal

            void ShowEmpty()
            {
                // 空タンクは素の枠だけ。ツールチップの印も外す（Web は Tooltip ごと描かない）
                // An empty tank is the bare frame only; the tooltip mark goes too (the Web renders no Tooltip)
                EnableInClassList(GhostStateClass, false);
                EnableInClassList(FillShownStateClass, false);
                EnableInClassList(AmountShownStateClass, false);
                _icon.ClearIcon();
                HoverTooltipMark.Unmark(this);
            }

            void ShowGhost()
            {
                // フィル無しで絵と必要量だけを透かす
                // Fade only the picture and required amount, without a fill
                EnableInClassList(GhostStateClass, true);
                EnableInClassList(FillShownStateClass, false);
                ShowPictureAndAmount();
            }

            void ShowFilled()
            {
                // 色の無い液体はフィルを描かない（フォールバック色でごまかさない）
                // A fluid without a color draws no fill (no fallback color)
                // silent-ok: 色が無いのはマスタの色が読めないときだけ / GameContentPresentation.FluidFace が LogError を出し、開発者は Console で気づく
                EnableInClassList(GhostStateClass, false);
                EnableInClassList(FillShownStateClass, content.Face.HasValue);
                if (content.Face.HasValue)
                {
                    // 高さと色は Web と同じく動的な値をインラインで当てる
                    // Height and color are dynamic values applied inline, as on the Web
                    _fill.style.height = Length.Percent(FluidFillRatio.Of(content.Amount, content.Capacity) * 100f);
                    _fill.style.backgroundColor = content.Face.Value;
                }
                ShowPictureAndAmount();
            }

            void ShowPictureAndAmount()
            {
                // 絵が欠けたら何も代わりに出さない（フィルが残るので。Web の fallback none）
                // A missing picture shows no substitute, since the fill remains (the Web fallback none)
                _icon.SetIcon(content.Icon, GameIconFallback.None);
                _amount.SetText(UiText.FromVerbatim(SlotAmountFormat.Format(content.Amount)));
                EnableInClassList(AmountShownStateClass, true);
                HoverTooltipMark.Mark(this, content.Icon.Label);
            }

            #endregion
        }
    }
}
```
